Struct Manifest 

Source 
pub struct Manifest { /* private fields */ }
Expand description

Fluent builder for a DriverManifest — the manifest verb’s payload.

Defaults match the common case: verbs is ["manifest","serve"], capabilities is ["pavo"], and version is the kit’s crate version — call Manifest::version with env!("CARGO_PKG_VERSION") (evaluated in your crate) when the driver versions independently.

let m = kit::Manifest::new("demo", "Demo")
    .version(env!("CARGO_PKG_VERSION"))
    .icon("🦚")
    .capability("config")
    .optional("config")
    .view(kit::view("board", "Board", "/demo/board",
        kit::markdown(kit::base().title("Hi"), "hello")))
    .sidebar(kit::contribution("home", "Home", "/demo/board"))
    .build();
assert_eq!(m.name, "demo");
assert_eq!(m.deps.unwrap().optional, vec!["config".to_string()]);
assert_eq!(m.contributes.sidebar.len(), 1);
