3//! tamias-driver-config — the common config API driver for the 9aia
4//! suite. Graduated from `examples/drivers/config` (the Python
5//! prototype); now a first-party Rust driver in the pavo workspace.
6//!
7//! Answers pavo's wire ops (`manifest`/`views`/`view`/`query`/`render`/
8//! `submit`)
9//! over `serve`, plus the sibling CLI verbs `manifest` and
10//! `capabilities`. The `query` surface is the product-facing API:
11//!
12//! | query      | params                    | result                          |
13//! |------------|---------------------------|---------------------------------|
14//! | `get`      | `key` (dotted)            | `{key, value, found}`           |
15//! | `set`      | `key`, `value`            | `{key, value, path, written}`   |
16//! | `resolve`  | —                         | merged config tree (nested)     |
17//! | `config`   | —                         | merged config flattened to KV  |
18//! | `sources`  | —                         | every contributing layer        |
19//!
20//! Other drivers read config by calling `tamias-driver-config serve`
21//! `query get/resolve` — namespaced keys (`myco.theme`), never raw
22//! files. Resolution order and write-back rules live in
23//! [`config`]'s module docs; the user-facing contract is the crate
24//! README.

41fn main() -> ExitCode {
42    let args: Vec<String> = std::env::args().skip(1).collect();
43    // --config may appear anywhere in argv (serve included); strip it
44    // before verb dispatch. `--config=PATH` and `--config PATH` both
45    // work; last one wins.
46    let mut explicit: Option<PathBuf> = None;
47    let mut positional: Vec<String> = Vec::new();
48    let mut i = 0;
49    while i < args.len() {
50        let arg = &args[i];
51        if arg == "--config" {
52            if let Some(v) = args.get(i + 1) {
53                explicit = Some(PathBuf::from(v));
54                i += 2;
55                continue;
56            }
57            eprintln!("tamias-driver-config: --config requires a path");
58            return ExitCode::from(2);
59        }
60        if let Some(v) = arg.strip_prefix("--config=") {
61            explicit = Some(PathBuf::from(v));
62            i += 1;
63            continue;
64        }
65        positional.push(arg.clone());
66        i += 1;
67    }
68
69    match positional.first().map(String::as_str) {
70        Some("manifest") => {
71            match serde_json::to_string_pretty(&view::manifest()) {
72                Ok(s) => println!("{s}"),
73                Err(e) => {
74                    eprintln!("tamias-driver-config: manifest serialization failed: {e}");
75                    return ExitCode::FAILURE;
76                }
77            }
78            ExitCode::SUCCESS
79        }
80        Some("capabilities") => {
81            let m = view::manifest();
82            println!(
83                "{}",
84                json!({"capabilities": m.capabilities, "verbs": m.verbs})
85            );
86            ExitCode::SUCCESS
87        }
88        Some("serve") => serve(explicit),
89        _ => {
90            eprintln!(
91                "usage: tamias-driver-config manifest|capabilities|serve [--config PATH]\n\
92                 \n\
93                 The common config API: `query get/set/resolve/config/sources` over \
94                 the pavo wire. Config layers: defaults < ~/.config/pavo/ < pavo.toml/\
95                 pavo.json in cwd ancestors < --config/PAVO_CONFIG < PAVO_CONFIG_* env."
96            );
97            ExitCode::from(2)
98        }
99    }
100}

102/// ndjson serve loop — one request per line on stdin, one response
103/// per line on stdout. Nothing else may touch stdout.
104fn serve(explicit: Option<PathBuf>) -> ExitCode {
105    let opts = Options {
106        explicit,
107        ..Options::from_process()
108    };
109    let stdin = std::io::stdin();
110    let mut out = std::io::stdout().lock();
111    for line in stdin.lock().lines() {
112        let Ok(line) = line else { break };
113        let line = line.trim();
114        if line.is_empty() {
115            continue;
116        }
117        let response = match parse_request(line) {
118            Ok(req) => dispatch(&req, &opts),
119            Err(e) => Response::err(format!("{e}")),
120        };
121        let text = pavo_protocol::wire::response_line(&response).unwrap_or_else(|_| {
122            "{\"ok\":false,\"error\":\"response serialization failed\"}".to_string()
123        });
124        if writeln!(out, "{text}").is_err() || out.flush().is_err() {
125            return ExitCode::FAILURE; // client went away
126        }
127    }
128    ExitCode::SUCCESS
129}

131fn dispatch(req: &Request, opts: &Options) -> Response {
132    match req {
133        Request::Manifest => Response::ok_typed(&view::manifest()),
134        Request::Views => Response::ok_typed(&view::view_summaries()),
135        Request::View { view, params } => view_result(view, params, opts),
136        Request::Query { query, params } => query_op(query, params, opts),
137        Request::Submit { view, form, values } => submit_op(view, form.as_deref(), values, opts),
138        Request::Render { .. } => {
139            Response::err("config driver ships no renderers — built-in rendering applies")
140        }
141    }
142}

144/// `view` op — resolve the spec, feed every query binding.
145fn view_result(id: &str, params: &Map<String, Value>, opts: &Options) -> Response {
146    let Some(spec) = view::manifest().views.into_iter().find(|v| v.id == id) else {
147        return Response::err(format!("unknown view: {id}"));
148    };
149    let mut data = Map::new();
150    for binding in &spec.queries {
151        // The binding's static params carry the real query name under
152        // "query"; caller params merge on top (they win).
153        let real = binding
154            .params
155            .get("query")
156            .and_then(Value::as_str)
157            .unwrap_or(&binding.name)
158            .to_string();
159        let mut merged = binding.params.clone();
160        for (k, v) in params {
161            merged.insert(k.clone(), v.clone());
162        }
163        if let Some(payload) = query_data(&real, &merged, opts) {
164            data.insert(binding.name.clone(), payload);
165        } else {
166            data.insert(
167                binding.name.clone(),
168                json!({"error": format!("unknown query: {real}")}),
169            );
170        }
171    }
172    Response::ok_typed(&ViewResult { view: spec, data })
173}

175/// `query` op — one named query. Unknown names are wire errors
176/// (`ok:false`); known-but-failed queries return `{"error": …}` in
177/// the data payload (partial degradation, per the wire contract).
178fn query_op(query: &str, params: &Map<String, Value>, opts: &Options) -> Response {
179    match query_data(query, params, opts) {
180        Some(data) => Response::ok_typed(&QueryResult {
181            query: query.to_string(),
182            data,
183        }),
184        None => Response::err(format!("unknown query: {query}")),
185    }
186}

188/// The query surface — `get`/`set`/`resolve`/`config`/`sources`.
189/// Returns `None` for unknown names; `{"error": …}` payloads for
190/// known names that failed (bad params, unwritable file).
191fn query_data(query: &str, params: &Map<String, Value>, opts: &Options) -> Option<Value> {
192    match query {
193        "resolve" => Some(load(opts).merged),
194        "config" => {
195            let cfg = load(opts);
196            Some(Value::Object(cfg.flat.into_iter().collect()))
197        }
198        "sources" => Some(sources_payload(opts)),
199        "get" => {
200            let key = params.get("key").and_then(Value::as_str)?;
201            let cfg = load(opts);
202            Some(match config::get(&cfg, key) {
203                Some(v) => json!({"key": key, "value": v, "found": true}),
204                None => json!({"key": key, "value": Value::Null, "found": false}),
205            })
206        }
207        "set" => {
208            let key = params.get("key").and_then(Value::as_str)?;
209            if !params.contains_key("value") {
210                return Some(json!({"error": "set requires params.value"}));
211            }
212            let value = params.get("value").cloned().unwrap_or(Value::Null);
213            let cfg = load(opts);
214            match config::set(&cfg, key, &value) {
215                Ok(outcome) => Some(json!({
216                    "key": outcome.key,
217                    "value": outcome.value,
218                    "path": outcome.path.display().to_string(),
219                    "written": true,
220                })),
221                Err(e) => Some(json!({"error": e})),
222            }
223        }
224        _ => None,
225    }
226}

228/// `submit` op — resolve the view's form node (by `form` id, else
229/// the first form), read its `submit_op`, and dispatch. Today the
230/// only action is `"set"` (the `config:edit` form): `values.key` +
231/// `values.value` go through the same write path as `query set`.
232fn submit_op(
233    view_id: &str,
234    form_id: Option<&str>,
235    values: &Map<String, Value>,
236    opts: &Options,
237) -> Response {
238    let Some(spec) = view::manifest().views.into_iter().find(|v| v.id == view_id) else {
239        return Response::err(format!("unknown view: {view_id}"));
240    };
241    let mut forms = Vec::new();
242    collect_forms(&spec.root, &mut forms);
243    let form = match form_id {
244        Some(id) => forms
245            .iter()
246            .find(|f| f.base.id.as_deref() == Some(id))
247            .copied(),
248        None => forms.first().copied(),
249    };
250    let Some(form) = form else {
251        return Response::err(format!("view '{view_id}' has no form node"));
252    };
253    let Some(op) = form.submit_op.as_deref() else {
254        return Response::err(format!("form in '{view_id}' has no submit_op"));
255    };
256    match op {
257        "set" => submit_set(values, opts),
258        other => Response::err(format!("unsupported submit_op: {other}")),
259    }
260}
261
262/// The `"set"` submit action — `values = {"key": <dotted>, "value":
263/// <any>}`. Missing/invalid fields come back as `SubmitResult.errors`
264/// (a rejected submit, not a wire error); write failures are
265/// `ok:false`.
266fn submit_set(values: &Map<String, Value>, opts: &Options) -> Response {
267    let mut errors = Map::new();
268    let key = values.get("key").and_then(Value::as_str);
269    if key.is_none() {
270        errors.insert("key".to_string(), json!("required — dotted config key"));
271    }
272    if !values.contains_key("value") {
273        errors.insert("value".to_string(), json!("required"));
274    }
275    let (Some(key), true) = (key, errors.is_empty()) else {
276        return Response::ok_typed(&SubmitResult {
277            errors,
278            ..SubmitResult::default()
279        });
280    };
281    let value = values.get("value").cloned().unwrap_or(Value::Null);
282    let cfg = load(opts);
283    match config::set(&cfg, key, &value) {
284        Ok(outcome) => Response::ok_typed(&SubmitResult {
285            message: Some(format!("set {}", outcome.key)),
286            data: Some(json!({
287                "key": outcome.key,
288                "value": outcome.value,
289                "path": outcome.path.display().to_string(),
290                "written": true,
291            })),
292            ..SubmitResult::default()
293        }),
294        Err(e) => Response::err(format!("set {key}: {e}")),
295    }
296}

298/// Collect every `form` node in a view tree (depth-first).
299fn collect_forms<'a>(node: &'a ViewNode, out: &mut Vec<&'a pavo_protocol::FormNode>) {
300    match node {
301        ViewNode::Form(f) => out.push(f),
302        ViewNode::Panel(n) => n.children.iter().for_each(|c| collect_forms(c, out)),
303        ViewNode::Row(n) => n.children.iter().for_each(|c| collect_forms(c, out)),
304        ViewNode::Column(n) => n.children.iter().for_each(|c| collect_forms(c, out)),
305        _ => {}
306    }
307}

309/// `sources` payload — every layer in precedence order, including
310/// pseudo-layers (defaults, env) and error/missing entries. This is
311/// the provenance view: which file gave which keys.
312fn sources_payload(opts: &Options) -> Value {
313    let cfg = load(opts);
314    let rows: Vec<Value> = cfg
315        .sources
316        .iter()
317        .map(|s| {
318            json!({
319                "source": s.path.as_ref().map(|p| p.display().to_string())
320                    .unwrap_or_else(|| s.label.clone()),
321                "path": s.path.as_ref().map(|p| p.display().to_string()),
322                "kind": s.kind,
323                "keys": s.keys,
324                "precedence": s.precedence,
325                "error": s.error,
326            })
327        })
328        .collect();
329    Value::Array(rows)
330}
