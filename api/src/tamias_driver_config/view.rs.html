1//! Manifest + view specs — the pavo-facing half of the driver.
2//!
3//! One namespace, `config`: deep links `config:show` and
4//! `config:sources`, queries `get`/`set`/`resolve`/`config`/`sources`.
5//! Other drivers and products call the `query` op over the serve
6//! wire — this is the "common config API" seed: namespaced dotted
7//! keys (`myco.theme`), never raw file paths.
8
9use pavo_protocol::{Contributes, Contribution, DriverManifest, QueryBinding, ViewNode, ViewSpec};
10use serde_json::{json, Map, Value};
11
12/// Parse a static ViewNode JSON literal — the trees below are
13/// compile-time constants, so a parse failure is a programming bug,
14/// not a runtime condition.
15fn node(v: Value) -> ViewNode {
16    serde_json::from_value(v).expect("static view tree is a valid ViewNode")
17}
18
19fn binding(name: &str, query: &str) -> QueryBinding {
20    let mut params = Map::new();
21    params.insert("query".to_string(), Value::from(query));
22    QueryBinding {
23        name: name.to_string(),
24        op: "query".to_string(),
25        params,
26    }
27}

29/// `config:show` — resolved config KV + provenance table + notes.
30fn show_view() -> ViewSpec {
31    ViewSpec {
32        id: "show".to_string(),
33        title: "Config".to_string(),
34        route: "/config".to_string(),
35        description: Some(
36            "Resolved pavo config (KV, dotted keys) plus every file/env \
37             layer that contributed — the common config API surface."
38                .to_string(),
39        ),
40        queries: vec![binding("config", "config"), binding("sources", "sources")],
41        root: node(json!({
42            "kind": "column",
43            "title": "Config",
44            "children": [
45                {"kind": "kv", "title": "Resolved", "query": "config"},
46                {"kind": "table", "title": "Sources", "query": "sources",
47                 "columns": [
48                    {"key": "source", "title": "Source"},
49                    {"key": "kind", "title": "Kind", "width": 8},
50                    {"key": "keys", "title": "Keys", "width": 6},
51                    {"key": "precedence", "title": "Prec", "width": 5}
52                 ]},
53                {"kind": "markdown", "text":
54                    "Layers low→high: built-in defaults < `~/.config/pavo/` \
55                     (global) < `pavo.toml`/`pavo.json` in cwd ancestors \
56                     (nearest wins) < `--config`/`PAVO_CONFIG` < `PAVO_CONFIG_*` \
57                     env. TOML + JSON; YAML unsupported. `set` writes the \
58                     project-level file."}
59            ]
60        })),
61    }
62}

64/// `config:edit` — the interactive proof: a `form` node whose

67/// key through the same `set` query path.

77        ),
78        queries: vec![],
79        root: node(json!({
80            "kind": "column",
81            "title": "Edit config",
82            "children": [
83                {"kind": "form", "id": "set-config", "title": "Set key",
84                 "submit_op": "set", "submit_label": "Save",
85                 "fields": [
86                    {"key": "key", "label": "Key", "kind": "text",

136/// The driver manifest — `manifest` verb payload.
137#[must_use]
138pub fn manifest() -> DriverManifest {
139    DriverManifest {
140        name: "config".to_string(),
141        title: "Config".to_string(),
142        icon: Some("⚙".to_string()),
143        version: env!("CARGO_PKG_VERSION").to_string(),
144        verbs: vec![
145            "manifest".to_string(),
146            "capabilities".to_string(),
147            "serve".to_string(),
148        ],
149        capabilities: vec!["pavo".to_string(), "config".to_string()],
150        contributes: Contributes {
151            statusbar: vec![Contribution {
152                id: "status".to_string(),
153                title: "config".to_string(),
154                route: "/config".to_string(),
155                priority: 90,
156                icon: None,
157            }],
158            ..Contributes::default()
159        },
160        views: vec![show_view(), sources_view(), edit_view()],
161        renderers: Vec::new(),
162    }
163}

165/// `views` op payload — summaries of [`manifest`]'s views.
166#[must_use]
167pub fn view_summaries() -> Vec<pavo_protocol::wire::ViewSummary> {
168    manifest()
169        .views
170        .iter()
171        .map(pavo_protocol::wire::ViewSummary::from)
172        .collect()
