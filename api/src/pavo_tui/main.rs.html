1//! pavo-tui — the ratatui host surface for pavo.
2//!
3//! `pavo-tui` discovers pavo-capable tamias drivers (`Host::discover`),
4//! merges their chrome contributions into a tab bar + status bar, and
5//! interprets each driver's ViewNode IR natively on a three-pane
6//! canvas: driver rail │ view canvas │ detail pane.
7//!
8//! Non-interactive modes:
9//!
10//! - `pavo-tui --list` — print discovered drivers + views, exit.
11//! - `pavo-tui --help` — usage.
12//!
13//! Deep links: `pavo-tui demo:board`, `--view demo:board`, or
14//! `--driver demo` (first view) land directly on a view.
15

58/// Parsed command line.
59struct Args {
60    /// Deep-link target to resolve after boot.
61    initial: Option<String>,
62    /// `--list` — print drivers + views and exit.
63    list: bool,
64    /// `-h`/`--help` — print usage and exit 0.
65    help: bool,
66}
67
68fn parse_args() -> Result<Args, String> {
69    parse_args_iter(std::env::args().skip(1))
70}
71
72/// [`parse_args`] over an explicit argv — split out so tests can
73/// exercise every flag without touching the process env.
74fn parse_args_iter<I: Iterator<Item = String>>(mut iter: I) -> Result<Args, String> {
75    let mut initial: Option<String> = None;
76    let mut list = false;
77    let mut help = false;
78    while let Some(arg) = iter.next() {
79        match arg.as_str() {
80            "-h" | "--help" => {
81                // Original semantics: help wins as soon as it's seen.
82                help = true;
83                break;
84            }
85            "--list" => list = true,
86            "--driver" => {
87                let kind = iter
88                    .next()
89                    .ok_or_else(|| "--driver requires a value".to_string())?;
90                initial = Some(kind);
91            }
92            "--view" => {
93                let target = iter
94                    .next()
95                    .ok_or_else(|| "--view requires a value".to_string())?;
96                initial = Some(target);
97            }
98            s if s.starts_with('-') => return Err(format!("unknown flag: {s}")),
99            s => {
100                if initial.is_some() {
101                    return Err(format!("unexpected extra argument: {s}"));
102                }
103                initial = Some(s.to_string());
104            }
105        }
106    }
107    Ok(Args {
108        initial,
109        list,
110        help,
111    })
112}
113
114fn main() -> ExitCode {
115    let args = match parse_args() {
116        Ok(a) => a,
117        Err(e) => {
118            eprintln!("pavo-tui: {e}\n\n{USAGE}");
119            return ExitCode::from(2);
120        }
121    };
122    if args.help {
123        print!("{USAGE}");
124        return ExitCode::SUCCESS;
125    }
126    if args.list {
127        return ExitCode::from(worker::list() as u8);
128    }
129    match run_tui(args.initial) {
130        Ok(()) => ExitCode::SUCCESS,
131        Err(e) => {
132            eprintln!("pavo-tui: {e}");
133            ExitCode::FAILURE
134        }
135    }
136}

138/// Enter the alternate screen, run the event loop, restore on exit.
139fn run_tui(initial: Option<String>) -> io::Result<()> {
140    let worker = worker::spawn();
141    let mut app = App::new(worker, initial);
142
143    let mut stdout = io::stdout();
144    enable_raw_mode()?;
145    execute!(stdout, EnterAlternateScreen, EnableMouseCapture)?;
146    let backend = CrosstermBackend::new(stdout);
147    let mut terminal = Terminal::new(backend)?;
148
149    let result = loop {
150        terminal.draw(|f| app.draw(f))?;
151        if !app.step() {
152            break Ok(());
153        }
154    };
155
156    // Restore the terminal no matter how we got here.
157    disable_raw_mode()?;
158    execute!(io::stdout(), LeaveAlternateScreen, DisableMouseCapture)?;
159    let mut out = io::stdout();
160    let _ = out.flush();
161    result
162}

164#[cfg(test)]
165#[cfg_attr(coverage, coverage(off))]
166mod tests {
167    #![allow(clippy::unwrap_used)]
168    use super::*;
169
170    fn args(v: &[&str]) -> Result<Args, String> {
171        parse_args_iter(v.iter().map(|s| s.to_string()))
172    }
173
174    #[test]
175    fn parse_args_covers_all_flags() {
176        let a = args(&[]).unwrap();
177        assert!(!a.list && !a.help && a.initial.is_none());
178
179        let a = args(&["--list"]).unwrap();
180        assert!(a.list);
181
182        for h in ["-h", "--help"] {
183            let a = args(&[h]).unwrap();
184            assert!(a.help);
185        }
186        // Help short-circuits later junk, like the old exit(0).
187        let a = args(&["--help", "--bogus"]).unwrap();
188        assert!(a.help);
189        // …but an error before --help still errors.
190        assert!(args(&["--bogus", "--help"]).is_err());
191
192        let a = args(&["--driver", "demo"]).unwrap();
193        assert_eq!(a.initial.as_deref(), Some("demo"));
194
195        let a = args(&["--view", "demo:board"]).unwrap();
196        assert_eq!(a.initial.as_deref(), Some("demo:board"));
197
198        let a = args(&["demo:item"]).unwrap();
199        assert_eq!(a.initial.as_deref(), Some("demo:item"));
200
201        // Flag-looking positionals are rejected; extras rejected.
202        assert!(args(&["--bogus"]).is_err());
203        assert!(args(&["a", "b"]).is_err());
204        assert!(args(&["--driver"]).is_err());
205        assert!(args(&["--view"]).is_err());
206        // --driver consumes the next arg even if it looks like a flag.
207        let a = args(&["--driver", "--list"]).unwrap();
208        assert_eq!(a.initial.as_deref(), Some("--list"));
209        assert!(!a.list);
210    }
