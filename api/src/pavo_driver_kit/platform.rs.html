1//! Platform feed constructors — [`JobItem`] and
2//! [`NotificationItem`], the payloads behind the `jobs` and
3//! `notifications` wire ops.
4//!
5//! A driver becomes a feed *provider* by declaring the matching
6//! `contributes` slot ([`crate::manifest::Manifest::jobs`] /
7//! [`crate::manifest::Manifest::notifications`]); the ops then carry
8//! item lists. Both are optional capabilities — drivers without a
9//! feed answer `ok:false` (or `[]`), which hosts read as "no feed".
10//!
11//! ```
12//! # use pavo_driver_kit as kit;
13//! let j = kit::job("build-1", "Build pavo", "running");
14//! let mut n = kit::notification("n-1", "Build finished");
15//! n.severity = Some("info".to_string());
16//! n.read = true;
17//! ```
18
19use serde_json::Map;
20
21use pavo_protocol::{JobItem, NotificationItem};
22
23/// One [`JobItem`] — `id`/`title`/`status` are the required trio
24/// (`status` is a free-form driver-owned string: `"queued"`,
25/// `"running"`, `"done"`, `"failed"`, …). `detail`, `route`,
26/// `updated_at` (RFC 3339), and `extra` are public fields.
27#[must_use]
28pub fn job(id: impl Into<String>, title: impl Into<String>, status: impl Into<String>) -> JobItem {
29    JobItem {
30        id: id.into(),
31        title: title.into(),
32        status: status.into(),
33        detail: None,
34        route: None,
35        updated_at: None,
36        extra: Map::new(),
37    }
38}
39
40/// One [`NotificationItem`] — `id`/`title` required; `body`,
41/// `severity` (free-form: `"info"`/`"warning"`/`"error"`/…),
42/// `route`, `created_at` (RFC 3339), `read`, and `extra` are public
43/// fields.
44#[must_use]
45pub fn notification(id: impl Into<String>, title: impl Into<String>) -> NotificationItem {
46    NotificationItem {
47        id: id.into(),
48        title: title.into(),
49        body: None,
50        severity: None,
51        route: None,
52        created_at: None,
53        read: false,
54        extra: Map::new(),
55    }
56}

58#[cfg(test)]
59#[cfg_attr(coverage, coverage(off))]
60mod tests {
61    #![allow(clippy::unwrap_used)]
62
63    use super::*;
64    use serde_json::json;
65
66    /// Constructors emit the required fields; optionals stay absent.
67    #[test]
68    fn feed_items() {
69        let j = job("j1", "Build", "running");
70        let v = serde_json::to_value(&j).unwrap();
71        assert_eq!(v["id"], json!("j1"));
72        assert_eq!(v["status"], json!("running"));
73        assert!(v.get("detail").is_none());
74
75        let mut n = notification("n1", "Hi");
76        n.severity = Some("warning".into());
77        n.read = true;
78        let v = serde_json::to_value(&n).unwrap();
79        assert_eq!(v["severity"], json!("warning"));
80        assert_eq!(v["read"], json!(true));
81    }
82}
