1//! The ndjson serve loop — one [`Request`] per line on stdin, one
2//! [`Response`] per line on stdout. Nothing else may touch stdout.
3//!
4//! This is the loop every pavo driver re-implements identically
5//! (`myco-driver`, `cuora-driver`, `sepia-driver-pavo`,
6//! `tamias-driver-tamias`, `sturn-driver`, `mimus-driver`,
7//! `bupha-driver`, `pavo-driver-config`); the kit absorbs it.
8
9use std::io::{self, BufRead, Write};
10
11use pavo_protocol::wire::{parse_request, response_line, Request, Response};
12
13/// Run the serve loop until `input` reaches EOF or the client goes
14/// away.
15///
16/// Each non-blank line is parsed as a wire [`Request`]; parse failures
17/// (malformed JSON, unknown `op`) become `ok:false` responses — a
18/// driver must never drop a line silently or crash the wire. Blank
19/// lines are skipped. `dispatch` maps a parsed request to its
20/// response; keep it cheap and side-effect-light for `manifest`.
21///
22/// Termination: stdin EOF ends the loop normally. A write or flush
23/// failure means the host went away — the loop stops and reports
24/// `Ok(())`, matching every existing driver's "exit quietly" rule.
25/// Stdin read errors likewise end the loop.
26///
27/// # Errors
28/// Currently infallible in practice — the signature returns
29/// [`io::Result`] so callers can `?` it inside larger `main`s.
30pub fn serve<R, W, F>(input: R, output: W, mut dispatch: F) -> io::Result<()>
31where
32    R: BufRead,
33    W: Write,
34    F: FnMut(&Request) -> Response,
35{
36    let mut out = output;
37    for line in input.lines() {
38        let Ok(line) = line else { break };
39        let line = line.trim();
40        if line.is_empty() {
41            continue;
42        }
43        let response = match parse_request(line) {
44            Ok(req) => dispatch(&req),
45            // Malformed JSON or an unknown op — the conformance
46            // contract wants `ok:false`, never a dropped line.
47            Err(e) => Response::err(e.to_string()),
48        };
49        // A response that can't serialize still owes the client a
50        // line — emit the fallback error shape by hand.
51        let text = response_line(&response).unwrap_or_else(|_| {
52            "{\"ok\":false,\"error\":\"response serialization failed\"}".to_string()
53        });
54        if writeln!(out, "{text}").is_err() || out.flush().is_err() {
55            return Ok(()); // client went away
56        }
57    }
58    Ok(())
59}

61/// [`serve`] over the process's real stdin/stdout — the `serve` verb
62/// body for a `tamias-driver-<type>` binary.
63///
64/// ```no_run
65/// # use pavo_driver_kit as kit;
66/// # use pavo_protocol::wire::{Request, Response};
67/// fn dispatch(req: &Request) -> Response {
68///     Response::err("not implemented")
69/// }
70/// fn main() {
71///     let _ = kit::serve_stdio(dispatch);
72/// }
73/// ```
74///
75/// # Errors
76/// See [`serve`] — propagates only unexpected I/O failures.
77pub fn serve_stdio(dispatch: impl FnMut(&Request) -> Response) -> io::Result<()> {
78    let stdin = io::stdin();
79    let stdout = io::stdout();
80    serve(stdin.lock(), stdout.lock(), dispatch)
81}

83#[cfg(test)]
84#[cfg_attr(coverage, coverage(off))]
85mod tests {
86    #![allow(clippy::unwrap_used)]

88    use super::*;
89    use pavo_protocol::wire::WireError;
90    use serde_json::{json, Value};
91
92    fn lines(out: &[u8]) -> Vec<Value> {
93        String::from_utf8_lossy(out)
94            .lines()
95            .map(|l| serde_json::from_str(l).unwrap())
96            .collect()
97    }
98
99    /// Requests round-trip; blank lines skip; bad ops and bad JSON
100    /// both answer `ok:false`.
101    #[test]
102    fn serve_dispatches_and_reports_errors() {
103        let input = concat!(
104            "{\"op\":\"manifest\"}\n",
105            "\n",
106            "   \n",
107            "{\"op\":\"jobs\"}\n",
108            "{\"op\":\"bogus\"}\n",
109            "not json\n",
110        );
111        let mut out = Vec::new();
112        serve(input.as_bytes(), &mut out, |req| match req {
113            Request::Manifest => Response::ok(json!({"name": "t"})),
114            Request::Jobs => Response::ok_typed(&Vec::<u8>::new()),
115            _ => Response::err("unexpected"),
116        })
117        .unwrap();
118        let got = lines(&out);
119        assert_eq!(got.len(), 4);
120        assert_eq!(got[0]["ok"], json!(true));
121        assert_eq!(got[0]["result"]["name"], json!("t"));
122        assert_eq!(got[1]["ok"], json!(true)); // jobs → []
123        assert_eq!(got[1]["result"], json!([]));
124        assert_eq!(got[2]["ok"], json!(false)); // unknown op
125        assert!(got[2]["error"].as_str().unwrap().contains("bogus"));
126        assert_eq!(got[3]["ok"], json!(false)); // invalid JSON
127    }
128
129    /// A dead client (write failure) ends the loop quietly — Ok, not
130    /// an error, matching every existing driver's exit rule.
131    #[test]
132    fn serve_tolerates_dead_client() {
133        struct Dead;
134        impl Write for Dead {
135            fn write(&mut self, _buf: &[u8]) -> io::Result<usize> {
136                Err(io::Error::new(io::ErrorKind::BrokenPipe, "gone"))
137            }
138            fn flush(&mut self) -> io::Result<()> {
139                Ok(())
140            }
141        }
142        let input = "{\"op\":\"manifest\"}\n{\"op\":\"manifest\"}\n";
143        serve(input.as_bytes(), Dead, |_| Response::ok(json!({}))).unwrap();
144    }

147    /// Vec<u8> suffice for tests and embedders.
