1//! Ops manifest → MCP tools over stdio (feature `mcp`).
2//!
3//! [`serve_stdio`] speaks MCP's stdio transport — newline-delimited
4//! JSON-RPC 2.0, the same "one line in, one line out" discipline as
5//! the pavo wire. Supported methods: `initialize`, `ping`,
6//! `tools/list`, `tools/call`; `notifications/*` are absorbed without
7//! reply. Each op becomes a tool named by its bare op name with an
8//! `inputSchema` generated from typed params (`arg`/`opt` → `string`,
9//! `flag` → `boolean`, `arg` params are `required`). `tools/call`
10//! arguments are the wire params map verbatim — the driver's `cli.*`
11//! dispatch owns semantics.
12//!
13//! `surfaces` (unbounded verbs) never become tools — a `tools/call`
14//! must terminate. Drivers wanting `watch`/`mcp`/`acp` expose them on
15//! a CLI surface instead.
16

23/// One op → one MCP tool declaration.
24fn tool(op: &OpSpec) -> Value {
25    let mut properties = Map::new();
26    let mut required = Vec::new();
27    for param in &op.params {
28        let ty = match param.kind {
29            ParamKind::Flag => "boolean",
30            ParamKind::Arg | ParamKind::Opt => "string",
31        };
32        let mut schema = json!({"type": ty});
33        if let Some(doc) = &param.doc {
34            schema["description"] = json!(doc);
35        }
36        if param.kind == ParamKind::Arg {
37            required.push(json!(param.name));
38        }
39        properties.insert(param.name.clone(), schema);
40    }
41    let mut input = json!({"type": "object", "properties": properties});
42    if !required.is_empty() {
43        input["required"] = json!(required);
44    }
45    json!({
46        "name": op.name,
47        "description": if op.doc.is_empty() { op.name.clone() } else { op.doc.clone() },
48        "inputSchema": input,
49    })
50}

52/// `tools/call` → dispatch → MCP result: stdout is the text payload,
53/// stderr joins as a second content block when non-empty, and a
54/// nonzero op `code` sets `isError` (the engine's exit status, honest
55/// data — not a protocol error).
56fn call(manifest: &OpsManifest, surface: &mut dyn Surface, params: &Map<String, Value>) -> Value {
57    let name = params
58        .get("name")
59        .and_then(Value::as_str)
60        .unwrap_or_default();
61    let Some(op) = manifest.op(name) else {
62        return json!({
63            "content": [{"type": "text", "text": format!("unknown tool: {name}")}],
64            "isError": true,
65        });
66    };
67    let arguments = params
68        .get("arguments")
69        .and_then(Value::as_object)
70        .cloned()
71        .unwrap_or_default();
72    let OpOutcome {
73        code,
74        stdout,
75        stderr,
76    } = surface.op(op, &arguments);
77    let mut content = vec![json!({"type": "text", "text": stdout})];
78    if !stderr.is_empty() {
79        content.push(json!({"type": "text", "text": stderr}));
80    }
81    json!({"content": content, "isError": code != 0})
82}

84/// One JSON-RPC request → its response value (`None` for
85/// notifications, which must not be answered).
86fn handle(manifest: &OpsManifest, surface: &mut dyn Surface, req: &Value) -> Option<Value> {
87    let id = req.get("id").cloned();
88    let method = req.get("method").and_then(Value::as_str).unwrap_or("");
89    let result = |result: Value| {
90        id.clone()
91            .map(|id| json!({"jsonrpc": "2.0", "id": id, "result": result}))
92    };
93    let error = |code: i32, message: String| {
94        id.clone().map(
95            |id| json!({"jsonrpc": "2.0", "id": id, "error": {"code": code, "message": message}}),
96        )
97    };
98    match method {
99        "initialize" => result(json!({
100            "protocolVersion": "2024-11-05",
101            "capabilities": {"tools": {}},
102            "serverInfo": {"name": manifest.engine, "version": env!("CARGO_PKG_VERSION")},
103        })),
104        "ping" => result(json!({})),
105        "tools/list" => result(json!({
106            "tools": manifest.ops.iter().map(tool).collect::<Vec<_>>(),
107        })),
108        "tools/call" => {
109            let params = req
110                .get("params")
111                .and_then(Value::as_object)
112                .cloned()
113                .unwrap_or_default();
114            result(call(manifest, surface, &params))
115        }
116        m if m.starts_with("notifications/") => None,
117        "" => None,
118        other => error(-32601, format!("method not found: {other}")),
119    }
120}

122/// The MCP serve loop over any `BufRead`/`Write` pair — notifications
123/// produce no line, parse failures answer `id:null` `-32700`.
124///
125/// # Errors
126/// Propagates unexpected I/O failures; a dead client ends the loop
127/// `Ok(())`, matching the pavo serve rule.
128pub fn serve<R, W>(
129    manifest: &OpsManifest,
130    surface: &mut dyn Surface,
131    input: R,
132    mut output: W,
133) -> io::Result<()>
134where
135    R: BufRead,
136    W: Write,
137{
138    for line in input.lines() {
139        let Ok(line) = line else { break };
140        let line = line.trim();
141        if line.is_empty() {
142            continue;
143        }
144        let reply = match serde_json::from_str::<Value>(line) {
145            Ok(req) => handle(manifest, surface, &req),
146            Err(e) => Some(json!({
147                "jsonrpc": "2.0", "id": Value::Null,
148                "error": {"code": -32700, "message": format!("parse error: {e}")},
149            })),
150        };
151        if let Some(reply) = reply {
152            let text = serde_json::to_string(&reply).unwrap_or_else(|_| {
153                "{\"jsonrpc\":\"2.0\",\"id\":null,\"error\":{\"code\":-32603,\"message\":\"serialization failed\"}}".to_string()
154            });
155            if writeln!(output, "{text}").is_err() || output.flush().is_err() {
156                return Ok(()); // client went away
157            }
158        }
159    }
160    Ok(())
161}

163/// [`serve`] over real stdin/stdout — the `mcp` verb body.
164///
165/// # Errors
166/// See [`serve`].
167pub fn serve_stdio(manifest: &OpsManifest, surface: &mut dyn Surface) -> io::Result<()> {
168    let stdin = io::stdin();
169    let stdout = io::stdout();
170    serve(manifest, surface, stdin.lock(), stdout.lock())
171}

173#[cfg(test)]
174#[cfg_attr(coverage, coverage(off))]
175mod tests {
176    #![allow(clippy::unwrap_used)]
177
178    use super::*;
179    use serde_json::json;
180
181    fn manifest() -> OpsManifest {
182        OpsManifest::from_value(&json!({
183            "engine": "fx",
184            "ops": [{"query": "cli.find", "doc": "sym -> loc",
185                     "params": [{"name": "symbol", "kind": "arg"},
186                                {"name": "docs", "kind": "flag"}]}],
187            "surfaces": ["watch"],
188        }))
189        .unwrap()
190    }
191
192    struct Echo;
193    impl Surface for Echo {
194        fn op(&mut self, op: &OpSpec, params: &Map<String, Value>) -> OpOutcome {
195            OpOutcome {
196                code: 0,
197                stdout: format!("{}:{}", op.name, params["symbol"].as_str().unwrap()),
198                stderr: String::new(),
199            }
200        }
201    }
202
203    fn lines(out: &[u8]) -> Vec<Value> {
204        String::from_utf8_lossy(out)
205            .lines()
206            .map(|l| serde_json::from_str(l).unwrap())
207            .collect()
208    }
209
210    #[test]
211    fn tools_list_derives_input_schemas() {
212        let m = manifest();
213        let mut out = Vec::new();
214        serve(
215            &m,
216            &mut Echo,
217            b"{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"tools/list\"}\n".as_slice(),
218            &mut out,
219        )
220        .unwrap();
221        let tools = lines(&out)[0]["result"]["tools"]
222            .as_array()
223            .unwrap()
224            .clone();
225        assert_eq!(tools.len(), 1); // `watch` is a surface, not a tool
226        assert_eq!(tools[0]["name"], json!("find"));
227        let schema = &tools[0]["inputSchema"];
228        assert_eq!(schema["required"], json!(["symbol"]));
229        assert_eq!(schema["properties"]["docs"]["type"], json!("boolean"));
230    }
231
232    #[test]
233    fn initialize_ping_and_call_round_trip() {
234        let m = manifest();
235        let input = concat!(
236            "{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"initialize\",\"params\":{}}\n",
237            "{\"jsonrpc\":\"2.0\",\"method\":\"notifications/initialized\"}\n",
238            "{\"jsonrpc\":\"2.0\",\"id\":2,\"method\":\"ping\"}\n",
239            "{\"jsonrpc\":\"2.0\",\"id\":3,\"method\":\"tools/call\",\"params\":{\"name\":\"find\",\"arguments\":{\"symbol\":\"main\",\"docs\":true}}}\n",
240            "{\"jsonrpc\":\"2.0\",\"id\":4,\"method\":\"bogus\"}\n",
241            "not json\n",
242        );
243        let mut out = Vec::new();
244        serve(&m, &mut Echo, input.as_bytes(), &mut out).unwrap();
245        let got = lines(&out);
246        assert_eq!(got.len(), 5); // the notification answers nothing
247        assert_eq!(got[0]["result"]["serverInfo"]["name"], json!("fx"));
248        assert_eq!(got[1]["result"], json!({}));
249        let call = &got[2]["result"];
250        assert_eq!(call["content"][0]["text"], json!("find:main"));
251        assert_eq!(call["isError"], json!(false));
252        assert_eq!(got[3]["error"]["code"], json!(-32601));
253        assert_eq!(got[4]["error"]["code"], json!(-32700));
254        assert_eq!(got[4]["id"], json!(Value::Null));
255    }
