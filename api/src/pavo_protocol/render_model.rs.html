1//! RenderModel — the canonical intermediate render output a
2//! view-kind renderer plugin emits for the `render` wire op.
3//!
4//! Renderer plugins (drivers declaring `renderers` in their manifest)
5//! don't ship surface code; they answer `{"op":"render", …}` with one
6//! of these tagged models and each surface maps it to native widgets:
7//!
8//! | variant      | CLI            | TUI             | web               |
9//! |--------------|----------------|-----------------|-------------------|
10//! | `rows`       | aligned table  | table rows      | `<table>`         |
11//! | `svg`        | raw markup     | `[svg]` hint    | `v-html` island   |
12//! | `text`       | lines          | lines           | `<pre>`           |
13//! | `placeholder`| `[component]`  | box + detail    | named component   |
14//!
15//! Wire shape (internally tagged on `"type"`):
16//!
17//! ```json
18//! {"type":"rows","headers":["id","label"],"aligns":["left","left"],
19//!  "rows":[["a","Alpha"]]}
20//! ```
21//!
22//! The set is deliberately small — plugins needing richer output use
23//! `placeholder` (web components) or `svg` (self-contained markup).
24//! Unknown `type` values must degrade gracefully on every surface
25//! (hosts treat them like `placeholder`).
26
27use serde::{Deserialize, Serialize};
28
29use crate::view_node::Align;
30
31/// One rendered node's canonical output — see the module docs for the
32/// per-surface mapping contract.
33#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
34#[serde(tag = "type", rename_all = "lowercase")]
35pub enum RenderModel {
36    /// Tabular output: headers, per-column alignment hints, and
37    /// stringified cell rows. The most portable model — every surface
38    /// has a table story.
39    Rows {
40        /// Column headings, in display order.
41        headers: Vec<String>,
42        /// Per-column alignment hints (parallel to `headers`); renderers
43        /// default missing/extra entries to left.
44        #[serde(default, skip_serializing_if = "Vec::is_empty")]
45        aligns: Vec<Align>,
46        /// Body rows — each a list of stringified cells aligned to
47        /// `headers` by index.
48        #[serde(default)]
49        rows: Vec<Vec<String>>,
50    },
51    /// Self-contained SVG markup — the plugin owns layout; surfaces
52    /// embed it verbatim (web) or degrade to a hint (TUI).
53    Svg {
54        /// The `<svg>…</svg>` document fragment.
55        markup: String,
56    },
57    /// Plain-text lines — the universal degradation target.
58    Text {
59        /// Lines to render top-to-bottom (no trailing newline needed).
60        lines: Vec<String>,
61    },
62    /// A named rich component the surface may resolve itself — the
63    /// plugin's escape hatch for visuals no built-in model covers.
64    /// Surfaces that can't resolve `component` show `props` as KV.
65    Placeholder {
66        /// Component/renderer name (web component tag, island id, …).
67        component: String,
68        /// Props passed verbatim to the component.
69        #[serde(default, skip_serializing_if = "serde_json::Map::is_empty")]
70        props: serde_json::Map<String, serde_json::Value>,
71    },
72}
73
74impl RenderModel {
75    /// Convenience constructor for [`RenderModel::Text`].
76    #[must_use]
77    pub fn text(lines: impl IntoIterator<Item = impl Into<String>>) -> Self {
78        Self::Text {
79            lines: lines.into_iter().map(Into::into).collect(),
80        }
81    }
82
83    /// Convenience constructor for [`RenderModel::Rows`].
84    #[must_use]
85    pub fn rows(headers: Vec<String>, rows: Vec<Vec<String>>) -> Self {
86        Self::Rows {
87            headers,
88            aligns: Vec::new(),
89            rows,
90        }
91    }
92}

94#[cfg(test)]
95#[cfg_attr(coverage, coverage(off))]
96mod tests {
97    #![allow(clippy::unwrap_used)]
98
99    use super::*;
100    use serde_json::json;
101
102    #[test]
103    fn rows_roundtrip() {
104        let m = RenderModel::Rows {
105            headers: vec!["id".into(), "label".into()],
106            aligns: vec![Align::Left, Align::Right],
107            rows: vec![vec!["a".into(), "Alpha".into()]],
108        };
109        let v = serde_json::to_value(&m).unwrap();
110        assert_eq!(v["type"], "rows");
111        assert_eq!(v["headers"], json!(["id", "label"]));
112        assert_eq!(v["aligns"], json!(["left", "right"]));
113        let back: RenderModel = serde_json::from_value(v).unwrap();
114        assert_eq!(back, m);
115    }
116
117    #[test]
118    fn all_variants_tag_and_roundtrip() {
119        for m in [
120            RenderModel::Svg {
121                markup: "<svg/>".into(),
122            },
123            RenderModel::text(["a", "b"]),
124            RenderModel::Placeholder {
125                component: "viz-gantt".into(),
126                props: Default::default(),
127            },
128        ] {
129            let v = serde_json::to_value(&m).unwrap();
130            let ty = v["type"].as_str().unwrap().to_string();
131            let back: RenderModel = serde_json::from_value(v).unwrap();
132            assert_eq!(back, m, "{ty}");
133        }
134        assert_eq!(
135            serde_json::to_value(RenderModel::text(["x"])).unwrap()["type"],
136            "text"
137        );
138        assert_eq!(
139            serde_json::to_value(RenderModel::Placeholder {
140                component: "c".into(),
141                props: Default::default()
142            })
143            .unwrap()["type"],
144            "placeholder"
145        );
146    }
147
148    #[test]
149    fn rows_defaults_for_absent_fields() {
150        // Forward-compat: a plugin omitting aligns/rows still parses.
151        let m: RenderModel =
152            serde_json::from_value(json!({"type":"rows","headers":["h"]})).unwrap();
153        match m {
154            RenderModel::Rows { headers, rows, .. } => {
155                assert_eq!(headers, vec!["h"]);
156                assert!(rows.is_empty());
157            }
158            other => panic!("expected rows, got {other:?}"),
159        }
160    }
161}
