13/// The driver manifest — the discoverability contract.
14///
15/// Wire shape (JSON):
16///
17/// ```json
18/// {
19///   "name": "myco",
20///   "title": "Myco",
21///   "version": "0.4.0",
22///   "verbs": ["manifest", "serve"],
23///   "capabilities": ["pavo"],
24///   "contributes": { "navbar": [...], "routes": [...] },
25///   "views": [ { "id": "board", "route": "/myco/board", "root": {…} } ]
26/// }
27/// ```
28#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
29pub struct DriverManifest {
30    /// Driver name — the `<type>` in `tamias-driver-<type>` and the
31    /// namespace for its routes (`/<name>/<view>`).
32    pub name: String,
33    /// Human-facing title for chrome (navbar entries, window titles).
34    pub title: String,
35    /// Optional icon — an emoji, an icon name, or a path the renderer
36    /// knows how to resolve.
37    #[serde(default, skip_serializing_if = "Option::is_none")]
38    pub icon: Option<String>,
39    /// Driver version (semver string).
40    pub version: String,
41    /// Wire verbs this executable answers beyond the tamias install
42    /// verbs — e.g. `["manifest", "serve"]` for pavo drivers.
43    /// Capability-negotiated: pavo treats a driver as a view driver
44    /// iff `serve` is listed (or `capabilities` contains `"pavo"`).
45    #[serde(default)]
46    pub verbs: Vec<String>,
47    /// Capability tags — free-form strings products use to declare
48    /// roles (`"pavo"`, `"config"`, `"chrome"`).
49    #[serde(default)]
50    pub capabilities: Vec<String>,
51    /// Contribution slots — where this driver's entries land in the
52    /// composed chrome. See [`Contributes`].
53    #[serde(default)]
54    pub contributes: Contributes,
55    /// Views this driver can render — each binds a route to a
56    /// declarative [`ViewNode`] tree plus the queries that feed it.
57    #[serde(default)]
58    pub views: Vec<ViewSpec>,
59    /// View-kind renderer plugins this driver ships — each declares
60    /// the node `kinds` it renders and which `surfaces` it supports.
61    /// See [`RendererDecl`]. A driver may declare renderers with zero
62    /// views (pure renderer plugin) or alongside views.
63    #[serde(default)]
64    pub renderers: Vec<RendererDecl>,
65}
66
67/// One view-kind renderer a driver ships — the manifest half of the
68/// renderer-plugin contract (the wire half is the `render` op).
69///
70/// A renderer claims node `kinds` (ViewNode `kind` strings, or `"*"`
71/// for any kind the host doesn't build in). When a surface meets a
72/// node of a claimed kind, the host routes it to this driver's
73/// `render` op instead of its built-in interpretation; the driver
74/// answers with a [`crate::render_model::RenderModel`].
75///
76/// Wire shape:
77///
78/// ```json
79/// { "id": "graph", "kinds": ["graph"],
80///   "surfaces": { "cli": {"formats": ["svg", "ascii"]},
81///                 "tui": true,
82///                 "web": {"island": true} },
83///   "priority": 100 }
84/// ```
85#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
86pub struct RendererDecl {
87    /// Renderer id — namespaced `<driver>:<id>` by convention; stable
88    /// for diagnostics and `/api/renderers` listings.
89    pub id: String,
90    /// Node kinds this renderer handles — ViewNode `kind` strings
91    /// (`"graph"`, `"timeline"`, …) or `"*"` for a catch-all. Explicit
92    /// kinds always beat `"*"` claims; among competing claims the
93    /// lowest [`RendererDecl::priority`] wins (ties: driver kind asc,
94    /// then renderer id asc — the chrome-merge ordering policy).
95    #[serde(default)]
96    pub kinds: Vec<String>,
97    /// Per-surface support. Absent surface = unsupported there.
98    #[serde(default)]
99    pub surfaces: RendererSurfaces,
100    /// Ordering hint within a kind — lower wins. Defaults to `100`
101    /// (same convention as [`Contribution::priority`]).
102    #[serde(default = "default_priority")]
103    pub priority: i32,
104    /// Optional entry-point hint — a component/module name the host
105    /// may prefer over `serve`-wire rendering (reserved; hosts
106    /// currently always use the wire).
107    #[serde(default, skip_serializing_if = "Option::is_none")]
108    pub entry: Option<String>,
109}
110
111/// Which surfaces a [`RendererDecl`] supports and how.
112#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
113pub struct RendererSurfaces {
114    /// CLI support — `formats` lists the `pavo view --format` values
115    /// the renderer claims beyond the built-ins (`text`/`json`/`dot`/
116    /// `mermaid`). An empty list is a wildcard: the renderer handles
117    /// every format it is asked for, including `text`.
118    #[serde(default, skip_serializing_if = "Option::is_none")]
119    pub cli: Option<CliSurface>,
120    /// TUI support — `true` renders plugin nodes on the ratatui host.
121    #[serde(default)]
122    pub tui: bool,
123    /// Web support — `component` names a web component/island the
124    /// host may mount; `island` marks the renderer as needing
125    /// client-side execution (SSR-incompatible output).
126    #[serde(default, skip_serializing_if = "Option::is_none")]
127    pub web: Option<WebSurface>,
128}
129
130/// CLI surface declaration for a [`RendererDecl`].
131#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
132pub struct CliSurface {
133    /// `--format` values this renderer claims. Empty = wildcard.
134    #[serde(default)]
135    pub formats: Vec<String>,
136}
137
138/// Web surface declaration for a [`RendererDecl`].
139#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
140pub struct WebSurface {
141    /// Optional component/island name — hint for hosts that resolve
142    /// `placeholder` models or `embed` nodes; the `render` op remains
143    /// the canonical path.
144    #[serde(default, skip_serializing_if = "Option::is_none")]
145    pub component: Option<String>,
146    /// `true` when the renderer's output requires client-side
147    /// execution (canvas, wasm, interactive svg) — hosts should treat
148    /// the node as an island.
149    #[serde(default)]
150    pub island: bool,
151}

153/// Contribution slots — the named places in pavo's chrome where
154/// drivers may register entries. A *chrome driver* owns the frame;
155/// view drivers fill slots. pavo merges contributions across drivers,
156/// resolving conflicts by driver priority.
157///
158/// The four built-in slots are `navbar`, `dashboard`, `statusbar`,
159/// and `routes`. Unknown slots are preserved via [`Contributes::extra`]
160/// so newer drivers degrade gracefully on older hosts.
161#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
162pub struct Contributes {
163    /// Entries for the top navigation bar (web/TUI chrome).
164    #[serde(default)]
165    pub navbar: Vec<Contribution>,
166    /// Entries for the shared `dashboard` route — composable panels
167    /// from many drivers on one screen.
168    #[serde(default)]
169    pub dashboard: Vec<Contribution>,
170    /// Entries for the bottom status bar — typically one-line
171    /// summaries (staleness, counts, claims).
172    #[serde(default)]
173    pub statusbar: Vec<Contribution>,
174    /// Standalone routes the driver owns outside its views — e.g. a
175    /// settings page. Each becomes a navigable URL on the web host.
176    #[serde(default)]
177    pub routes: Vec<Contribution>,
178    /// Forward-compat escape hatch: any slot name not listed above
179    /// round-trips untouched. Renderers ignore slots they don't know.
180    #[serde(flatten)]
181    pub extra: BTreeMap<String, Vec<Contribution>>,
182}
183
184/// One contributed entry in a slot.
185///
186/// `route` is a host-relative path (`/myco/board`, `/dashboard`);
187/// renderers resolve it against the driver's views or its own
188/// routes. `priority` orders entries within a slot — lower wins;
189/// ties break by driver discovery order (stable, deterministic).
190#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
191pub struct Contribution {
192    /// Stable id — namespaced as `<driver>:<id>` by convention so
193    /// contributions never collide across drivers.
194    pub id: String,
195    /// Label shown in chrome.
196    pub title: String,
197    /// Route the entry navigates to.
198    pub route: String,
199    /// Ordering hint within the slot; lower sorts first. Defaults to
200    /// `100` when absent so drivers can leave it unset.
201    #[serde(default = "default_priority")]
202    pub priority: i32,
203    /// Optional icon hint (same resolution rules as
204    /// [`DriverManifest::icon`]).
205    #[serde(default, skip_serializing_if = "Option::is_none")]
206    pub icon: Option<String>,
207}
208
209fn default_priority() -> i32 {
210    100
211}

213/// One view a driver offers — a route plus the declarative tree and
214/// the queries that feed it.
215#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
216pub struct ViewSpec {
217    /// View id, unique within the driver (`board`, `item`).
218    pub id: String,
219    /// Human-facing title.
220    pub title: String,
221    /// Route the view answers — `<driver>:<view>` deep links resolve
222    /// here (e.g. `myco:board` → `/myco/board`).
223    pub route: String,
224    /// Optional longer description for picker UIs.
225    #[serde(default, skip_serializing_if = "Option::is_none")]
226    pub description: Option<String>,
227    /// Query bindings — named data sources the [`ViewNode`] tree
228    /// references via `NodeBase::query`. Each `name` maps to a wire
229    /// `op` (usually `query` on the driver's `serve`) plus static
230    /// params merged with caller-supplied view params.
231    #[serde(default)]
232    pub queries: Vec<QueryBinding>,
233    /// The declarative UI tree — the renderers' input.
234    pub root: ViewNode,
235}
236
237/// A named query binding inside a [`ViewSpec`]: `name` is what
238/// ViewNodes reference; `op` is the wire op the host calls; `params`
239/// are static arguments merged under caller-supplied view params.
240#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
241pub struct QueryBinding {
242    /// Binding name — referenced by `query` fields in the tree.
243    pub name: String,
244    /// Wire op to invoke on the driver's `serve` (typically `"query"`,
245    /// with the real product op carried in `params`).
246    pub op: String,
247    /// Static params merged into every invocation of this binding.
248    #[serde(default, skip_serializing_if = "Map::is_empty")]
249    pub params: Map<String, Value>,
250}
