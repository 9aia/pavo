1//! ViewNode IR — the declarative UI tree drivers emit and every pavo
2//! renderer (CLI, TUI, web) interprets.
3//!
4//! Wire shape: each node is a JSON object with a `"kind"` discriminator
5//! (internally tagged — no wrapper envelope):
6//!
7//! ```json
8//! { "kind": "table", "title": "Board", "query": "items",
9//!   "columns": [{ "key": "id", "title": "ID" }],
10//!   "rows": [["T-001", "pavo scaffold"]] }
11//! ```
12//!
13//! Nodes bind to data through [`NodeBase::query`], which names a
14//! [`crate::manifest::QueryBinding`] in the enclosing [`ViewSpec`];
15//! the host resolves bindings to view-model data before rendering.
16//! Renderers that don't know a `kind` must degrade to a generic
17//! key/value rendering rather than failing — forward compatibility
18//! is a host obligation.
19
20use serde::{Deserialize, Serialize};
21use serde_json::{Map, Value};

23/// The ViewNode IR — one node of a declarative view tree.
24///
25/// Internally tagged on `"kind"` with lowercase variant names:
26/// `panel`, `table`, `tree`, `graph`, `form`, `kv`, `markdown`,
27/// `embed`, `row`, `column`. Every variant wraps a node struct that
28/// carries the shared [`NodeBase`] fields flattened to the top level.
29#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
30#[serde(tag = "kind", rename_all = "lowercase")]
31pub enum ViewNode {
32    /// A titled container grouping children — the universal wrapper.
33    Panel(PanelNode),
34    /// Tabular data: declared columns plus rows (inline or query-fed).
35    Table(TableNode),
36    /// Hierarchical data (nested items, breadcrumbs, file trees).
37    Tree(TreeNode),
38    /// Nodes + edges — the escape hatch for graph-shaped state;
39    /// typically rendered by a view-kind driver.
40    Graph(GraphNode),
41    /// An input form: fields, defaults, and a submit op.
42    Form(FormNode),
43    /// Key/value pairs — detail panes, metadata, degraded fallback
44    /// for unknown kinds.
45    Kv(KvNode),
46    /// Free-form markdown text.
47    Markdown(MarkdownNode),
48    /// Escape hatch: host renders `component` with `props` — used for
49    /// rich renderers (WASM/web components) on capable surfaces.
50    Embed(EmbedNode),
51    /// Horizontal layout: children flow left-to-right.
52    Row(RowNode),
53    /// Vertical layout: children stack top-to-bottom.
54    Column(ColumnNode),
55}
56
57/// Fields every ViewNode kind shares, flattened onto each node's JSON
58/// object so authors write one flat shape regardless of kind.
59#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
60pub struct NodeBase {
61    /// Optional stable id — targets for deep links, diffing, and
62    /// renderer-side expansion state.
63    #[serde(default, skip_serializing_if = "Option::is_none")]
64    pub id: Option<String>,
65    /// Optional heading the renderer may display above the node.
66    #[serde(default, skip_serializing_if = "Option::is_none")]
67    pub title: Option<String>,
68    /// Optional binding name — references a
69    /// [`crate::manifest::QueryBinding`] whose result supplies this
70    /// node's data. Meaning is kind-specific: `table` reads `rows`,
71    /// `kv` reads `pairs`, `tree` reads `items`, `graph` reads
72    /// `nodes`/`edges`.
73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    pub query: Option<String>,
75    /// Optional longer description (tooltips, help panes).
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub description: Option<String>,
78    /// Renderer hints — kind-specific knobs (`density`, `direction`,
79    /// `collapsed`, …). Renderers ignore keys they don't know.
80    #[serde(default, skip_serializing_if = "Map::is_empty")]
81    pub props: Map<String, Value>,
82}

136/// `table` — tabular data with declared columns.
137///
138/// Rows may be declared inline (`rows`) or supplied by the bound
139/// query (`base.query`); inline rows win when both are present so a
140/// driver can ship static previews.
141#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
142pub struct TableNode {
143    /// Shared node fields.
144    #[serde(flatten)]
145    pub base: NodeBase,
146    /// Column declarations, in display order.
147    #[serde(default)]
148    pub columns: Vec<TableColumn>,
149    /// Inline rows — each row is a list of cell values aligned to
150    /// `columns`. Cell values are arbitrary JSON (`string`, `number`,
151    /// `bool`, `null`); renderers stringify.
152    #[serde(default, skip_serializing_if = "Vec::is_empty")]
153    pub rows: Vec<Vec<Value>>,
154    /// When true the renderer may offer sorting by column.
155    #[serde(default)]
156    pub sortable: bool,
157}

264/// `form` — input fields plus a submit op.
265///
266/// A form is *interactive* when it carries a [`FormNode::submit_op`]
267/// (or sits in a view whose driver answers the wire `submit` op —
268/// see [`crate::wire::Request::Submit`]). Fields with
269/// [`FormField::editable`] `false` render read-only even on
270/// interactive surfaces; the rest submit their `key` → value pairs.
271#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
272pub struct FormNode {
273    /// Shared node fields — `base.id` is the form's submit target
274    /// (the wire `submit` op's optional `form` param).
275    #[serde(flatten)]
276    pub base: NodeBase,
277    /// Field declarations, in display order.
278    #[serde(default)]
279    pub fields: Vec<FormField>,
280    /// Driver-side action invoked on submit (e.g. `"set"`, `"update"`,
281    /// `"add"`) — the driver maps it to its own behavior; the wire op
282    /// is always `submit`. When absent the renderer may only preview
283    /// the form.
284    #[serde(default, skip_serializing_if = "Option::is_none")]
285    pub submit_op: Option<String>,
286    /// Submit button label; renderers default to "Submit".
287    #[serde(default, skip_serializing_if = "Option::is_none")]
288    pub submit_label: Option<String>,
289}

291/// Field kinds a [`FormField`] may declare.
292#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
293#[serde(rename_all = "lowercase")]
294pub enum FormFieldKind {
295    /// Single-line text.
296    #[default]
297    Text,
298    /// Multi-line text.
299    Textarea,
300    /// Integer or float.
301    Number,
302    /// Boolean toggle/checkbox.
303    Bool,
304    /// One-of selection from `options`.
305    Select,
306}

308/// One field in a [`FormNode`].
309#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
310pub struct FormField {
311    /// Field key — the name submitted with the form data.
312    pub key: String,
313    /// Human-facing label.
314    pub label: String,
315    /// Input kind; renderers may fall back to text for kinds they
316    /// don't support.
317    #[serde(default)]
318    pub kind: FormFieldKind,
319    /// Whether the field must be non-empty on submit.
320    #[serde(default)]
321    pub required: bool,
322    /// Options for `select` fields.
323    #[serde(default, skip_serializing_if = "Vec::is_empty")]
324    pub options: Vec<String>,
325    /// Default value (JSON — string/number/bool per `kind`).
326    #[serde(default, skip_serializing_if = "Option::is_none")]
327    pub default: Option<Value>,
328    /// Current value when the form renders existing state.
329    #[serde(default, skip_serializing_if = "Option::is_none")]
330    pub value: Option<Value>,
331    /// Help text rendered under the field.
332    #[serde(default, skip_serializing_if = "Option::is_none")]
333    pub hint: Option<String>,
334    /// Whether the field accepts user input on interactive surfaces.
335    /// `false` renders the current `value` read-only (a `value`-only
336    /// display field inside a form). Defaults to `true` — fields in
337    /// forms predating this flag were always meant to be inputs.
338    #[serde(default = "default_true")]
339    pub editable: bool,
340    /// Placeholder text shown while the field is empty.
341    #[serde(default, skip_serializing_if = "Option::is_none")]
342    pub placeholder: Option<String>,
343}
344
345/// Serde default for [`FormField::editable`] — fields are editable
346/// unless a driver says otherwise.
347fn default_true() -> bool {
348    true
349}

387/// `embed` — escape hatch for rich renderers: the host looks up
388/// `component` (a registered renderer or web component name) and
389/// hands it `props`. Drivers needing custom visuals (live graphs,
390/// timelines, WASM) use this rather than extending the IR.
391///
392/// Component props live in [`NodeBase::props`] — the flattened shared
393/// field *is* the `"props"` wire key. (An earlier draft gave this
394/// node a second `props` field; serde `flatten` made the two collide
395/// on one JSON key and silently swapped values on round-trip.)
396#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
397pub struct EmbedNode {
398    /// Shared node fields — `base.props` carries the component props.
399    #[serde(flatten)]
400    pub base: NodeBase,
401    /// Component/renderer name — resolved against the host's
402    /// renderer registry (and, on web, registered web components).
403    pub component: String,
404}

406/// Convenience: look up a node's shared fields without matching.
407impl ViewNode {
408    /// The node's [`NodeBase`] — id/title/query/description/props.
409    #[must_use]
410    pub fn base(&self) -> &NodeBase {
411        match self {
412            Self::Panel(n) => &n.base,
413            Self::Table(n) => &n.base,
414            Self::Tree(n) => &n.base,
415            Self::Graph(n) => &n.base,
416            Self::Form(n) => &n.base,
417            Self::Kv(n) => &n.base,
418            Self::Markdown(n) => &n.base,
419            Self::Embed(n) => &n.base,
420            Self::Row(n) => &n.base,
421            Self::Column(n) => &n.base,
422        }
423    }
424
425    /// Mutable access to the node's [`NodeBase`].
426    pub fn base_mut(&mut self) -> &mut NodeBase {
427        match self {
428            Self::Panel(n) => &mut n.base,
429            Self::Table(n) => &mut n.base,
430            Self::Tree(n) => &mut n.base,
431            Self::Graph(n) => &mut n.base,
432            Self::Form(n) => &mut n.base,
433            Self::Kv(n) => &mut n.base,
434            Self::Markdown(n) => &mut n.base,
435            Self::Embed(n) => &mut n.base,
436            Self::Row(n) => &mut n.base,
437            Self::Column(n) => &mut n.base,
438        }
439    }
440}

442#[cfg(test)]
443mod tests {
444    #![allow(clippy::unwrap_used)]
445    use super::*;
446    use serde_json::json;
447
448    /// Every kind parses and `base()`/`base_mut()` reach its NodeBase.
449    #[test]
450    fn base_accessors_cover_all_kinds() {
451        let kinds = [
452            json!({"kind":"panel","title":"P"}),
453            json!({"kind":"table","title":"T","columns":[],"rows":[]}),
454            json!({"kind":"tree","title":"Tr","items":[]}),
455            json!({"kind":"graph","title":"G","nodes":[],"edges":[]}),
456            json!({"kind":"form","title":"F","fields":[]}),
457            json!({"kind":"kv","title":"K","pairs":[]}),
458            json!({"kind":"markdown","title":"M","text":"hi"}),
459            json!({"kind":"embed","title":"E","component":"c"}),
460            json!({"kind":"row","title":"R","children":[]}),
461            json!({"kind":"column","title":"C","children":[]}),
462        ];
463        for (i, v) in kinds.into_iter().enumerate() {
464            let mut node: ViewNode = serde_json::from_value(v).unwrap();
465            assert!(node.base().title.is_some(), "kind {i}");
466            node.base_mut().id = Some(format!("n{i}"));
467            assert_eq!(node.base().id.as_deref(), Some(format!("n{i}").as_str()));
468        }
469    }
470
471    /// Pre-`editable`/`placeholder` form JSON still parses — fields
472    /// default to editable, placeholders to absent.
473    #[test]
474    fn legacy_form_field_json_still_parses() {
475        let node: ViewNode = serde_json::from_value(json!({
476            "kind": "form",
477            "id": "prefs",
478            "submit_op": "set",
479            "fields": [
480                {"key": "theme", "label": "Theme", "kind": "select",
481                 "options": ["dark", "light"], "value": "dark"},
482                {"key": "note", "label": "Note", "editable": false,
483                 "value": "read-only", "placeholder": "type here"}
484            ]
485        }))
486        .unwrap();
487        let ViewNode::Form(form) = node else {
488            panic!("expected form");
489        };
490        assert_eq!(form.base.id.as_deref(), Some("prefs"));
491        assert!(form.fields[0].editable);
492        assert!(form.fields[0].placeholder.is_none());
493        assert!(!form.fields[1].editable);
494        assert_eq!(form.fields[1].placeholder.as_deref(), Some("type here"));
495    }
