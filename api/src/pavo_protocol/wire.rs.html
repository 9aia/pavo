1//! Wire ops — the ndjson JSON-RPC convention every sibling `serve`
2//! speaks, applied to pavo drivers.
3//!
4//! Transport: one JSON object per line on stdin/stdout (CLI/TUI) or
5//! bridged to ws/http by `pavo serve` (web). Requests are
6//! `{"op":"…", …}`; responses are `{"ok":true,"result":…}` or
7//! `{"ok":false,"error":"…"}`. This is the same shape as
8//! `tamias-serve`, `myco serve`, `cuora serve`, `sepia serve`.
9//!
10//! Ops (v1):
11//!
12//! | op        | params                              | result            |
13//! |-----------|-------------------------------------|-------------------|
14//! | `manifest`| —                                   | [`DriverManifest`]|
15//! | `views`   | —                                   | `Vec<ViewSummary>`|
16//! | `view`    | `view`, `params?`                   | [`ViewResult`]    |
17//! | `query`   | `query`, `params?`                  | [`QueryResult`]   |
18//! | `render`  | `kind`, `node`, `data`, `surface`,  | [`RenderModel`]   |
19//! |           | `format?`                           |                   |
20//! | `submit`  | `view`, `form?`, `values`           | [`SubmitResult`]  |
21//! | `jobs`    | —                                   | `Vec<JobItem>`    |
22//! | `notifications` | —                          | `Vec<NotificationItem>` |
23//!
24//! `jobs` and `notifications` are the **platform-feed ops**: drivers
25//! whose manifest declares the matching `contributes` slot
26//! (`jobs`/`notifications`) answer them with item lists — see
27//! [`crate::platform`]. Both are optional capabilities: drivers
28//! without the feed answer `ok:false` (or `[]`) per the usual
29//! convention, and hosts treat either as "no feed".
30//!
31//! `submit` is the form-interaction verb: a renderer collects the
32//! editable [`crate::view_node::FormField`] values of a `form` node
33//! and posts them here. `view` names the view the form lives in;
34//! `form` optionally names the form node's `id` when a view carries
35//! several forms (absent = the driver's default/first form);
36//! `values` maps field `key` → submitted JSON value. The driver
37//! dispatches on the form's `submit_op`; the result may carry a
38//! message, per-field validation `errors`, driver `data`, and an
39//! updated `view` for re-render.
40//!
41//! `render` is the view-kind renderer-plugin verb: a driver whose
42//! manifest declares `renderers` answers it with a canonical
43//! [`crate::render_model::RenderModel`] for one node on one surface.
44//! Drivers that declare no renderers may ignore it (hosts only call
45//! it for kinds the manifest claims).
46//!
47//! Thin drivers may proxy a product's existing `serve` for `query`
48//! payloads; `manifest`/`views`/`view` are always answered by the
49//! driver itself.
50

56/// A wire request — one JSON object per line, discriminated by `op`.
57///
58/// Unknown ops fail deserialization; hosts should treat that as a
59/// protocol error and surface the driver's raw error text.
60#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
61#[serde(tag = "op", rename_all = "lowercase")]
62pub enum Request {
63    /// `{"op":"manifest"}` → [`DriverManifest`]. Discovery verb —
64    /// cheap, must not require product state.
65    Manifest,
66    /// `{"op":"views"}` → `Vec<ViewSummary>` — view ids/titles/routes
67    /// without the full trees (lightweight listing).
68    Views,
69    /// `{"op":"view","view":"board","params":{…}}` → [`ViewResult`] —
70    /// the resolved tree plus query-fed data for one view.
71    View {
72        /// View id from the driver's manifest (`board`, `item`).
73        view: String,
74        /// Caller-supplied params merged into each query binding.
75        #[serde(default, skip_serializing_if = "Map::is_empty")]
76        params: Map<String, Value>,
77    },
78    /// `{"op":"query","query":"items","params":{…}}` → [`QueryResult`] —
79    /// execute a single named query binding (or raw product op, for
80    /// thin drivers) and return its data payload.
81    Query {
82        /// Query name from the view's `queries` bindings, or a raw
83        /// product op when the driver proxies `serve`.
84        query: String,
85        /// Params merged into the binding's static params.
86        #[serde(default, skip_serializing_if = "Map::is_empty")]
87        params: Map<String, Value>,
88    },
89    /// `{"op":"render","kind":"graph","node":{…},"data":{…},
90    /// "surface":"cli","format":"svg"}` → [`RenderModel`] — the
91    /// renderer-plugin verb. `node` is the raw ViewNode JSON (the
92    /// host passes it untyped so unknown/future fields survive);
93    /// `data` is the view's query-binding map verbatim; `surface` is
94    /// `"cli"`, `"tui"`, or `"web"` (open string — hosts may add
95    /// surfaces); `format` is the surface's format hint (`"svg"`,
96    /// `"ascii"`, …) when one applies.
97    Render {
98        /// The node's `kind` — also the renderer claim key.
99        kind: String,
100        /// The node as raw JSON (a ViewNode object, `{"kind":…}`).
101        #[serde(default)]
102        node: Value,
103        /// The view's bound query data — the same map `view` results
104        /// carry, so the plugin can resolve `node.query` itself.
105        #[serde(default)]
106        data: Map<String, Value>,
107        /// Target surface — `"cli"`, `"tui"`, `"web"`.
108        surface: String,
109        /// Optional format hint (CLI `--format` value, etc.).
110        #[serde(default, skip_serializing_if = "Option::is_none")]
111        format: Option<String>,
112    },
113    /// `{"op":"submit","view":"settings","form":"prefs",
114    /// "values":{"theme":"dark"}}` → [`SubmitResult`] — the
115    /// form-submission verb. `view` is the view id the form node
116    /// belongs to; `form` optionally names the form node's `id`
117    /// (its `NodeBase::id`) when the view has several forms —
118    /// drivers resolve the node to read its `submit_op` and field
119    /// declarations. `values` maps each submitted field's `key` to
120    /// its JSON value (`string`/`number`/`bool` per field `kind`).
121    /// Drivers that don't implement forms answer `ok:false`.
122    Submit {
123        /// View id from the driver's manifest — same namespace as
124        /// the `view` op.
125        view: String,
126        /// Optional form-node id (`NodeBase::id` of the `form`
127        /// node); absent submits the view's default form.
128        #[serde(default, skip_serializing_if = "Option::is_none")]
129        form: Option<String>,
130        /// Submitted field values keyed by `FormField::key`. Keys
131        /// the driver doesn't know are ignored; missing `required`
132        /// fields come back in [`SubmitResult::errors`].
133        #[serde(default, skip_serializing_if = "Map::is_empty")]
134        values: Map<String, Value>,
135    },
136    /// `{"op":"jobs"}` → `Vec<JobItem>` — the job/queue feed op.
137    /// Drivers that declare `contributes.jobs` answer with their
138    /// current job list; drivers without a job feed answer `ok:false`
139    /// (or an empty list) — both mean "no feed" to hosts.
140    Jobs,
141    /// `{"op":"notifications"}` → `Vec<NotificationItem>` — the
142    /// notification feed op. Same optional-capability convention as
143    /// `jobs`: `contributes.notifications` marks the provider, the
144    /// op carries the items.
145    Notifications,
146}

148/// A wire response — one JSON object per line.
149///
150/// `ok:true` carries `result`; `ok:false` carries `error`. Fields
151/// for the other arm are absent (not null) so consumers can match on
152/// `ok` alone.
153#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
154pub struct Response {
155    /// Success flag — the only field guaranteed present.
156    pub ok: bool,
157    /// Result payload when `ok` is true.
158    #[serde(default, skip_serializing_if = "Option::is_none")]
159    pub result: Option<Value>,
160    /// Human/machine-readable error string when `ok` is false.
161    #[serde(default, skip_serializing_if = "Option::is_none")]
162    pub error: Option<String>,
163}
164
165impl Response {
166    /// Build a success response around an arbitrary JSON result.
167    #[must_use]
168    pub fn ok(result: Value) -> Self {
169        Self {
170            ok: true,
171            result: Some(result),
172            error: None,
173        }
174    }
175
176    /// Build a success response from any serializable payload.
177    ///
178    /// Serialization failure is reported as an error response rather
179    /// than panicking — a driver must never crash the wire.
180    #[must_use]
181    pub fn ok_typed<T: Serialize>(value: &T) -> Self {
182        match serde_json::to_value(value) {
183            Ok(v) => Self::ok(v),
184            Err(e) => Self::err(format!("result serialization failed: {e}")),
185        }
186    }
187
188    /// Build an error response.
189    #[must_use]
190    pub fn err(message: impl Into<String>) -> Self {
191        Self {
192            ok: false,
193            result: None,
194            error: Some(message.into()),
195        }
196    }
197
198    /// Typed accessor — the result payload deserialized as `T`.
199    ///
200    /// # Errors
201    /// Returns the serde error when `result` is absent or doesn't
202    /// match `T`.
203    pub fn result_as<T: for<'de> Deserialize<'de>>(&self) -> Result<T, serde_json::Error> {
204        match &self.result {
205            Some(v) => serde_json::from_value(v.clone()),
206            None => serde_json::from_value(Value::Null),
207        }
208    }
209}

211/// Result of the `views` op — a [`ViewSpec`] minus the `root` tree,
212/// for cheap listing. Renderers fetch full trees via `view`.
213#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
214pub struct ViewSummary {
215    /// View id.
216    pub id: String,
217    /// Human-facing title.
218    pub title: String,
219    /// Route the view answers.
220    pub route: String,
221    /// Optional description.
222    #[serde(default, skip_serializing_if = "Option::is_none")]
223    pub description: Option<String>,
224    /// Names of the query bindings this view uses.
225    #[serde(default)]
226    pub queries: Vec<String>,
227}
228
229impl From<&ViewSpec> for ViewSummary {
230    fn from(spec: &ViewSpec) -> Self {
231        Self {
232            id: spec.id.clone(),
233            title: spec.title.clone(),
234            route: spec.route.clone(),
235            description: spec.description.clone(),
236            queries: spec.queries.iter().map(|q| q.name.clone()).collect(),
237        }
238    }
239}

241/// Result of the `view` op — the resolved view: the declarative tree
242/// plus every query binding's data, keyed by binding name.
243#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
244pub struct ViewResult {
245    /// The view's manifest spec (tree included).
246    pub view: ViewSpec,
247    /// Query results keyed by binding name — `data["items"]` feeds
248    /// every node whose `base.query == "items"`. Bindings that failed
249    /// appear with an `{"error": "…"}` payload rather than failing the
250    /// whole view (partial degradation).
251    #[serde(default)]
252    pub data: Map<String, Value>,
253}
254
255/// Result of the `query` op — one binding's data payload.
256#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
257pub struct QueryResult {
258    /// The query/binding name that produced this payload.
259    pub query: String,
260    /// The data — shape is defined by the driver/product; hosts pass
261    /// it through to renderers verbatim.
262    pub data: Value,
263}

265/// Result of the `submit` op — what a form submission produced.
266///
267/// Every field is optional so drivers answer with only what they
268/// have: a bare `{"message":"saved"}` is a complete result. Hosts
269/// treat `errors` non-empty as a validation failure (the form should
270/// re-render with the messages, not navigate away); `view`, when
271/// present, is a fresh [`ViewResult`] to swap in.
272#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
273pub struct SubmitResult {
274    /// Human/machine-facing outcome (`"saved myco.theme"`,
275    /// `"applied 2 keys"`). Renderers show it as a status line.
276    #[serde(default, skip_serializing_if = "Option::is_none")]
277    pub message: Option<String>,
278    /// Per-field validation errors keyed by `FormField::key` —
279    /// non-empty means the submit was rejected and the form should
280    /// re-render with these messages. Values are driver-defined
281    /// (usually strings).
282    #[serde(default, skip_serializing_if = "Map::is_empty")]
283    pub errors: Map<String, Value>,
284    /// Driver-defined result payload (what changed, new ids, …) —
285    /// shape is the driver's, hosts pass it through verbatim.
286    #[serde(default, skip_serializing_if = "Option::is_none")]
287    pub data: Option<Value>,
288    /// An updated [`ViewResult`] to re-render in place of the
289    /// submitted view (e.g. the form now shows the saved values).
290    #[serde(default, skip_serializing_if = "Option::is_none")]
291    pub view: Option<Box<ViewResult>>,
292    /// Forward-compat catch-all — unknown result fields a future
293    /// driver emits survive a typed round-trip.
294    #[serde(flatten)]
295    pub extra: Map<String, Value>,
296}

301/// Errors pavo-protocol itself can produce (parsing helpers, not the
302/// wire — wire errors travel as `Response::err` strings).
303#[derive(Debug, thiserror::Error)]
304pub enum WireError {
305    /// The line wasn't valid JSON for the expected type.
306    #[error("invalid wire JSON: {0}")]
307    Json(#[from] serde_json::Error),
308    /// The request `op` isn't in the v1 op set.
309    #[error("unknown op: {op}")]
310    UnknownOp {
311        /// The unrecognized op string.
312        op: String,
313    },
314}

316/// Parse one ndjson line into a [`Request`].
317///
318/// # Errors
319/// [`WireError::Json`] on malformed JSON; [`WireError::UnknownOp`]
320/// when `op` is missing or not a v1 op.
321pub fn parse_request(line: &str) -> Result<Request, WireError> {
322    let value: Value = serde_json::from_str(line)?;
323    let op = value
324        .get("op")
325        .and_then(Value::as_str)
326        .ok_or_else(|| WireError::UnknownOp {
327            op: "<missing>".to_string(),
328        })?;
329    match op {
330        "manifest" | "views" | "view" | "query" | "render" | "submit" | "jobs"
331        | "notifications" => Ok(serde_json::from_value(value)?),
332        other => Err(WireError::UnknownOp {
333            op: other.to_string(),
334        }),
335    }
336}

338/// Serialize a [`Response`] to its single-line ndjson form.
339///
340/// # Errors
341/// [`WireError::Json`] if serialization fails (should not happen for
342/// well-formed responses).
343pub fn response_line(response: &Response) -> Result<String, WireError> {
344    Ok(serde_json::to_string(response)?)
345}

347#[cfg(test)]
348#[cfg_attr(coverage, coverage(off))]
349mod tests {
350    #![allow(clippy::unwrap_used)]

352    use super::*;
353    use serde_json::json;

355    /// `submit` parses from its wire shape and round-trips; absent
356    /// `form`/`values` default cleanly (old hosts may omit them).
357    #[test]
358    fn submit_request_roundtrip() {
359        let req = parse_request(
360            r#"{"op":"submit","view":"settings","form":"prefs","values":{"theme":"dark","zoom":2}}"#,
361        )
362        .unwrap();
363        match req {
364            Request::Submit { view, form, values } => {
365                assert_eq!(view, "settings");
366                assert_eq!(form.as_deref(), Some("prefs"));
367                assert_eq!(values["theme"], json!("dark"));
368                assert_eq!(values["zoom"], json!(2));
369            }
370            other => panic!("expected Submit, got {other:?}"),
371        }
372        // Minimal form: only `view` required.
373        let req = parse_request(r#"{"op":"submit","view":"v"}"#).unwrap();
374        match req {
375            Request::Submit { view, form, values } => {
376                assert_eq!(view, "v");
377                assert!(form.is_none());
378                assert!(values.is_empty());
379            }
380            other => panic!("expected Submit, got {other:?}"),
381        }
382        // Serialized form keeps the same shape.
383        let line = serde_json::to_string(&Request::Submit {
384            view: "v".into(),
385            form: None,
386            values: Map::new(),
387        })
388        .unwrap();
389        assert_eq!(line, r#"{"op":"submit","view":"v"}"#);
390    }
391
392    /// `SubmitResult` parses sparse driver answers, carries
393    /// per-field errors, and preserves unknown fields via `extra`.
394    #[test]
395    fn submit_result_parses_and_preserves_extras() {
396        let r: SubmitResult =
397            serde_json::from_value(json!({"message":"saved","future_field":{"x":1}})).unwrap();
398        assert_eq!(r.message.as_deref(), Some("saved"));
399        assert!(r.errors.is_empty());
400        assert_eq!(r.extra["future_field"], json!({"x": 1}));
401
402        let r: SubmitResult = serde_json::from_value(json!({
403            "errors": {"theme": "unknown value"},
404            "data": {"applied": 2},
405            "view": {"view": {"id":"v","title":"V","route":"/v","root":{"kind":"kv","pairs":[]}},"data":{}}
406        }))
407        .unwrap();
408        assert_eq!(r.errors["theme"], json!("unknown value"));
409        assert!(r.view.is_some());
410    }

412    /// `jobs`/`notifications` joined the v1 op set.
413    #[test]
414    fn platform_feed_ops_parse() {
415        assert!(matches!(
416            parse_request(r#"{"op":"jobs"}"#).unwrap(),
417            Request::Jobs
418        ));
419        assert!(matches!(
420            parse_request(r#"{"op":"notifications"}"#).unwrap(),
421            Request::Notifications
422        ));
423        assert_eq!(
424            serde_json::to_string(&Request::Jobs).unwrap(),
425            r#"{"op":"jobs"}"#
426        );
427        assert_eq!(
428            serde_json::to_string(&Request::Notifications).unwrap(),
429            r#"{"op":"notifications"}"#
430        );
431    }

433    /// Unknown ops still fail; `submit` joined the v1 op set.
434    #[test]
435    fn parse_request_rejects_unknown_ops() {

438            Err(WireError::UnknownOp { .. })
