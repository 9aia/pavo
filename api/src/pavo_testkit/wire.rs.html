1//! Raw ndjson wire session — spawn `<driver> serve`, send one JSON
2//! line, read one JSON line back, with a per-request timeout.
3//!
4//! This is the transport [`crate::conformance`] and
5//! [`crate::record`] are built on; it's `pub` so tests can speak
6//! malformed lines or probe transport failures the typed helpers
7//! don't cover. Sync only — a stdout reader thread feeds an `mpsc`
8//! channel so `recv_timeout` gives the timeout without an async
9//! runtime (same shape as `pavo-core`'s `DriverClient`).
10
11use std::io::{BufRead, BufReader, Write};
12use std::path::{Path, PathBuf};
13use std::process::{Child, ChildStdin, Command, Stdio};
14use std::sync::mpsc::{channel, Receiver};
15use std::thread;
16use std::time::Duration;
17
18use serde_json::Value;
19
20use pavo_protocol::Response;

22/// Errors a [`Session`] can produce. All are transport/protocol-level;
23/// a driver answering `ok:false` is *not* an error here — inspect the
24/// [`Response`] instead.
25#[derive(Debug)]
26pub enum WireError {
27    /// Spawn/write/read failure.
28    Io(std::io::Error),
29    /// No response line within the request timeout.
30    Timeout {
31        /// The timeout that elapsed.
32        timeout: Duration,
33    },
34    /// The driver's stdout closed (process exited or crashed).
35    Exited {
36        /// What was observed — `"stdout closed"`, an exit status, …
37        detail: String,
38    },
39    /// A response line arrived but isn't a [`Response`] JSON object.
40    BadResponse {
41        /// The offending line (truncated).
42        line: String,
43    },
44}
45
46impl std::fmt::Display for WireError {
47    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
48        match self {
49            Self::Io(e) => write!(f, "io error: {e}"),
50            Self::Timeout { timeout } => write!(f, "timed out after {timeout:?}"),
51            Self::Exited { detail } => write!(f, "driver exited: {detail}"),
52            Self::BadResponse { line } => {
53                let short: String = line.chars().take(120).collect();
54                write!(f, "response line isn't a Response: {short}")
55            }
56        }
57    }
58}
59
60impl std::error::Error for WireError {}
61
62impl From<std::io::Error> for WireError {
63    fn from(e: std::io::Error) -> Self {
64        Self::Io(e)
65    }
66}

68/// A live `<driver> serve` session: child + stdin + stdout line
69/// channel. Dropping kills the child.
70pub struct Session {
71    child: Child,
72    stdin: ChildStdin,
73    rx: Receiver<String>,
74    driver: PathBuf,
75}
76
77impl Session {
78    /// Spawn `<driver> serve` with piped stdio.
79    ///
80    /// Freshly-written fixture drivers can hit `ETXTBSY`
81    /// (`ExecutableFileBusy`) when exec'd in the same instant the
82    /// writer's `close` lands — retry briefly rather than fail.
83    ///
84    /// # Errors
85    /// [`WireError::Io`] if the process can't be spawned or its pipes
86    /// can't be captured.
87    pub fn spawn(driver: &Path) -> Result<Self, WireError> {
88        let mut cmd = Command::new(driver);
89        cmd.arg("serve")
90            .stdin(Stdio::piped())
91            .stdout(Stdio::piped())
92            .stderr(Stdio::null());
93        let mut child = spawn_retry(&mut cmd)?;
94        let (Some(stdin), Some(stdout)) = (child.stdin.take(), child.stdout.take()) else {
95            let _ = child.kill();
96            let _ = child.wait();
97            return Err(WireError::Io(std::io::Error::other(
98                "stdio pipes not captured",
99            )));
100        };
101        let (tx, rx) = channel::<String>();
102        thread::spawn(move || {
103            for line in BufReader::new(stdout).lines() {
104                match line {
105                    Ok(line) => {
106                        if tx.send(line).is_err() {
107                            break;
108                        }
109                    }
110                    Err(_) => break,
111                }
112            }
113        });
114        Ok(Self {
115            child,
116            stdin,
117            rx,
118            driver: driver.to_path_buf(),
119        })
120    }
121
122    /// The driver executable path.
123    #[must_use]
124    pub fn driver(&self) -> &Path {
125        &self.driver
126    }
127
128    /// Whether the child has exited (best-effort `try_wait`).
129    #[must_use]
130    pub fn has_exited(&mut self) -> bool {
131        self.child.try_wait().ok().flatten().is_some()
132    }
133
134    /// Send one raw line and return the raw response line.
135    ///
136    /// # Errors
137    /// [`WireError::Io`] on write failure, [`WireError::Timeout`] when
138    /// no line arrives in `timeout`, [`WireError::Exited`] when stdout
139    /// closes first.
140    pub fn request_raw(&mut self, line: &str, timeout: Duration) -> Result<String, WireError> {
141        writeln!(self.stdin, "{line}")?;
142        self.stdin.flush()?;
143        match self.rx.recv_timeout(timeout) {
144            Ok(line) => Ok(line),
145            Err(std::sync::mpsc::RecvTimeoutError::Timeout) => Err(WireError::Timeout { timeout }),
146            Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => Err(WireError::Exited {
147                detail: "stdout closed".to_string(),
148            }),
149        }
150    }
151
152    /// Send a JSON request value and parse the response line as a
153    /// [`Response`]. `ok:false` is returned as `Ok(Response)` — it's a
154    /// wire-level answer, not a transport failure.
155    ///
156    /// # Errors
157    /// Transport failures per [`Session::request_raw`];
158    /// [`WireError::BadResponse`] when the line isn't a `Response`.
159    pub fn call(&mut self, request: &Value, timeout: Duration) -> Result<Response, WireError> {
160        let line = serde_json::to_string(request).map_err(|e| {
161            WireError::Io(std::io::Error::other(format!("request serialization: {e}")))
162        })?;
163        let raw = self.request_raw(&line, timeout)?;
164        serde_json::from_str(&raw).map_err(|_| WireError::BadResponse { line: raw })
165    }
166}
167
168/// `cmd.spawn()` with a short retry loop for `ExecutableFileBusy`
169/// (ETXTBSY) — see [`Session::spawn`].
170fn spawn_retry(cmd: &mut Command) -> Result<Child, WireError> {
171    let mut attempt = 0u32;
172    loop {
173        match cmd.spawn() {
174            Ok(child) => return Ok(child),
175            Err(e) if e.kind() == std::io::ErrorKind::ExecutableFileBusy && attempt < 20 => {
176                attempt += 1;
177                thread::sleep(Duration::from_millis(10));
178            }
179            Err(e) => return Err(WireError::Io(e)),
180        }
181    }
182}
183
184impl Drop for Session {
185    fn drop(&mut self) {
186        let _ = self.child.kill();
187        let _ = self.child.wait();
188    }
189}

191#[cfg(test)]
192mod tests {
193    #![allow(clippy::unwrap_used)]
194    use super::*;
195
196    #[test]
197    fn wire_error_display_is_informative() {
198        let cases = [
199            WireError::Io(std::io::Error::other("boom")),
200            WireError::Timeout {
201                timeout: Duration::from_millis(5),
202            },
203            WireError::Exited {
204                detail: "stdout closed".to_string(),
205            },
206            WireError::BadResponse {
207                line: "not json".to_string(),
208            },
209        ];
210        for e in &cases {
211            let s = e.to_string();
212            assert!(!s.is_empty());
213        }
214        // BadResponse truncates long lines.
215        let long = "x".repeat(500);
216        let s = WireError::BadResponse { line: long }.to_string();
217        assert!(s.len() < 200, "{s}");
218    }
