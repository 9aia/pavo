1//! Driver conformance — [`check_driver`] runs the pavo wire contract
2//! suite against a `tamias-driver-*` executable and returns structured
3//! [`Violation`]s rather than panicking.
4//!
5//! The suite (all over `<driver> serve`, one persistent session):
6//!
7//! 1. `{"op":"manifest"}` → parses as [`DriverManifest`] and
8//!    advertises pavo (`"serve"` in `verbs` or `"pavo"` in
9//!    `capabilities`).
10//! 2. `{"op":"views"}` → parses as `Vec<ViewSummary>` whose id set
11//!    matches the manifest's declared views.
12//! 3. `{"op":"view","view":<id>"}` for every declared view → parses
13//!    as [`ViewResult`], echoes the requested view id, and its `data`
14//!    map carries every declared query binding.
15//! 4. `{"op":"query","query":<name>"}` for every declared binding →
16//!    parses as [`QueryResult`].
17//! 5. Malformed requests — an unknown op and a non-JSON line — must
18//!    both get an `ok:false` [`Response`].
19//! 6. The process must still answer after all of the above (session
20//!    liveness across multiple requests).
21//!
22//! Transport failures (timeout, EOF, unparseable response lines)
23//! become violations on the check that hit them; the suite keeps
24//! going so one broken op doesn't hide the rest.
25

30use pavo_protocol::wire::{QueryResult, Response, ViewResult, ViewSummary};

37/// Which contract check produced a [`Violation`].
38#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize)]
39#[serde(rename_all = "snake_case")]
40pub enum Check {
41    /// The `serve` process couldn't be spawned.
42    Spawn,
43    /// `manifest` op: transport failure, `ok:false`, or a result that
44    /// doesn't parse as [`DriverManifest`].
45    Manifest,
46    /// The manifest parses but doesn't advertise pavo service.
47    AdvertisesPavo,
48    /// `views` op: transport failure, `ok:false`, or a result that
49    /// doesn't parse as `Vec<ViewSummary>`.
50    Views,
51    /// `views` result ids don't match the manifest's view ids.
52    ViewsConsistent,
53    /// `view` op for a declared view: transport failure, `ok:false`,
54    /// or a result that doesn't parse as [`ViewResult`].
55    View,
56    /// `view` result shape: echoed id mismatch or `data` missing a
57    /// declared query binding.
58    ViewShape,
59    /// `query` op for a declared binding: transport failure,
60    /// `ok:false`, or a result that doesn't parse as [`QueryResult`].
61    Query,
62    /// Malformed requests (unknown op, non-JSON) didn't get `ok:false`.
63    MalformedRejected,
64    /// The session died or stopped answering partway through.
65    SessionAlive,
66}
67
68impl std::fmt::Display for Check {
69    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
70        let s = match self {
71            Self::Spawn => "spawn",
72            Self::Manifest => "manifest",
73            Self::AdvertisesPavo => "advertises_pavo",
74            Self::Views => "views",
75            Self::ViewsConsistent => "views_consistent",
76            Self::View => "view",
77            Self::ViewShape => "view_shape",
78            Self::Query => "query",
79            Self::MalformedRejected => "malformed_rejected",
80            Self::SessionAlive => "session_alive",
81        };
82        f.write_str(s)
83    }
84}

86/// One contract violation — which check failed and why.
87#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
88pub struct Violation {
89    /// The check that failed.
90    pub check: Check,
91    /// Human/machine-readable detail (driver errors, parse errors,
92    /// mismatched ids, …).
93    pub detail: String,
94}
95
96impl std::fmt::Display for Violation {
97    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
98        write!(f, "[{}] {}", self.check, self.detail)
99    }
100}
101
102impl std::error::Error for Violation {}
103
104fn violation(check: Check, detail: impl Into<String>) -> Violation {
105    Violation {
106        check,
107        detail: detail.into(),
108    }
109}

111/// Knobs for [`check_driver_with`].
112#[derive(Clone, Debug)]
113pub struct CheckOptions {
114    /// Per-request response timeout. Default 10s.
115    pub timeout: Duration,
116}
117
118impl Default for CheckOptions {
119    fn default() -> Self {
120        Self {
121            timeout: Duration::from_secs(10),
122        }
123    }
124}

126/// Run the conformance suite against the `tamias-driver-*` executable
127/// at `driver` with default options. Returns every violation found —
128/// an empty vec means the driver is conformant.
129#[must_use]
130pub fn check_driver(driver: &Path) -> Vec<Violation> {
131    check_driver_with(driver, CheckOptions::default())
132}
133
134/// [`check_driver`] with explicit [`CheckOptions`] (short timeouts for
135/// delay-injection tests, etc.).
136#[must_use]
137pub fn check_driver_with(driver: &Path, options: CheckOptions) -> Vec<Violation> {
138    let mut out = Vec::new();
139    let timeout = options.timeout;
140    let mut session = match Session::spawn(driver) {
141        Ok(s) => s,
142        Err(e) => {
143            out.push(violation(Check::Spawn, format!("{driver:?}: {e}")));
144            return out;
145        }
146    };
147
148    // -- 1. manifest -------------------------------------------------
149    let manifest = match typed_call(
150        &mut session,
151        json!({"op": "manifest"}),
152        timeout,
153        Check::Manifest,
154        &mut out,
155    ) {
156        Some(value) => match serde_json::from_value::<DriverManifest>(value) {
157            Ok(m) => {
158                let advertises = m.verbs.iter().any(|v| v == "serve")
159                    || m.capabilities.iter().any(|c| c == "pavo");
160                if !advertises {
161                    out.push(violation(
162                        Check::AdvertisesPavo,
163                        format!(
164                            "manifest doesn't advertise pavo service (verbs={:?}, capabilities={:?})",
165                            m.verbs, m.capabilities
166                        ),
167                    ));
168                }
169                Some(m)
170            }
171            Err(e) => {
172                out.push(violation(
173                    Check::Manifest,
174                    format!("manifest result isn't a DriverManifest: {e}"),
175                ));
176                None
177            }
178        },
179        None => None,
180    };
181
182    // -- 2. views ------------------------------------------------------
183    let summaries = typed_call(
184        &mut session,
185        json!({"op": "views"}),
186        timeout,
187        Check::Views,
188        &mut out,
189    )
190    .and_then(
191        |value| match serde_json::from_value::<Vec<ViewSummary>>(value) {
192            Ok(v) => Some(v),
193            Err(e) => {
194                out.push(violation(
195                    Check::Views,
196                    format!("views result isn't Vec<ViewSummary>: {e}"),
197                ));
198                None
199            }
200        },
201    );
202
203    // views ↔ manifest consistency
204    if let (Some(m), Some(s)) = (&manifest, &summaries) {
205        let declared: BTreeSet<&str> = m.views.iter().map(|v| v.id.as_str()).collect();
206        let served: BTreeSet<&str> = s.iter().map(|v| v.id.as_str()).collect();
207        let missing: Vec<&str> = declared.difference(&served).copied().collect();
208        let extra: Vec<&str> = served.difference(&declared).copied().collect();
209        if !missing.is_empty() || !extra.is_empty() {
210            out.push(violation(
211                Check::ViewsConsistent,
212                format!(
213                    "views/manifest mismatch — in manifest not views: {missing:?}; in views not manifest: {extra:?}"
214                ),
215            ));
216        }
217    }
218
219    // -- 3. view per declared view -------------------------------------
220    // View ids to exercise: the manifest's when we have it, else the
221    // views op's.
222    let view_ids: Vec<String> = manifest
223        .as_ref()
224        .map(|m| m.views.iter().map(|v| v.id.clone()).collect())
225        .or_else(|| {
226            summaries
227                .as_ref()
228                .map(|s| s.iter().map(|v| v.id.clone()).collect())
229        })
230        .unwrap_or_default();
231    let mut binding_names: BTreeSet<String> = BTreeSet::new();
232    if let Some(m) = &manifest {
233        for spec in &m.views {
234            for b in &spec.queries {
235                binding_names.insert(b.name.clone());
236            }
237        }
238    }
239
240    for id in &view_ids {
241        let request = json!({"op": "view", "view": id});
242        let Some(value) = typed_call(&mut session, request, timeout, Check::View, &mut out) else {
243            continue;
244        };
245        match serde_json::from_value::<ViewResult>(value) {
246            Ok(result) => {
247                if result.view.id != *id {
248                    out.push(violation(
249                        Check::ViewShape,
250                        format!("view '{id}' echoed id '{}'", result.view.id),
251                    ));
252                }
253                for b in &result.view.queries {
254                    if !result.data.contains_key(&b.name) {
255                        out.push(violation(
256                            Check::ViewShape,
257                            format!("view '{id}' data missing binding '{}'", b.name),
258                        ));
259                    }
260                }
261                for name in result.view.queries.iter().map(|b| b.name.clone()) {
262                    binding_names.insert(name);
263                }
264            }
265            Err(e) => out.push(violation(
266                Check::View,
267                format!("view '{id}' result isn't a ViewResult: {e}"),
268            )),
269        }
270    }
271
272    // -- 4. query per declared binding ---------------------------------
273    for name in &binding_names {
274        let request = json!({"op": "query", "query": name});
275        if let Some(value) = typed_call(&mut session, request, timeout, Check::Query, &mut out) {
276            if let Err(e) = serde_json::from_value::<QueryResult>(value) {
277                out.push(violation(
278                    Check::Query,
279                    format!("query '{name}' result isn't a QueryResult: {e}"),
280                ));
281            }
282        }
283    }
284
285    // -- 5. malformed requests must get ok:false ------------------------
286    for raw in ["{\"op\":\"bogus\"}", "this is not json"] {
287        match session.request_raw(raw, timeout) {
288            Ok(line) => match serde_json::from_str::<Response>(&line) {
289                Ok(resp) if !resp.ok => {}
290                Ok(resp) => out.push(violation(
291                    Check::MalformedRejected,
292                    format!("malformed request {raw:?} got ok:{}", resp.ok),
293                )),
294                Err(_) => out.push(violation(
295                    Check::MalformedRejected,
296                    format!("malformed request {raw:?} got non-Response line: {line}"),
297                )),
298            },
299            Err(e) => out.push(violation(
300                Check::MalformedRejected,
301                format!("malformed request {raw:?}: {e}"),
302            )),
303        }
304    }
305
306    // -- 6. session still alive -----------------------------------------
307    match session.call(&json!({"op": "views"}), timeout) {
308        Ok(resp) if resp.ok => {}
309        Ok(resp) => out.push(violation(
310            Check::SessionAlive,
311            format!(
312                "final views probe got ok:false: {}",
313                resp.error.unwrap_or_default()
314            ),
315        )),
316        Err(e) => out.push(violation(Check::SessionAlive, format!("{e}"))),
317    }
318
319    out
320}

322/// One wire round-trip → the `result` value on `ok:true`.
323///
324/// Pushes a [`Violation`] under `check` and returns `None` on
325/// transport failure, `ok:false`, or an unparseable response line.
326fn typed_call(
327    session: &mut Session,
328    request: Value,
329    timeout: Duration,
330    check: Check,
331    out: &mut Vec<Violation>,
332) -> Option<Value> {
333    match session.call(&request, timeout) {
334        Ok(resp) if resp.ok => Some(resp.result.unwrap_or(Value::Null)),
335        Ok(resp) => {
336            out.push(violation(
337                check,
338                resp.error
339                    .unwrap_or_else(|| "ok:false without error".to_string()),
340            ));
341            None
342        }
343        Err(e) => {
344            out.push(violation(check, wire_detail(&request, &e)));
345            None
346        }
347    }
348}
349
350/// Render a transport error with the request that triggered it.
351fn wire_detail(request: &Value, e: &WireError) -> String {
352    let op = request.get("op").and_then(Value::as_str).unwrap_or("?");
353    let extra = request
354        .get("view")
355        .or_else(|| request.get("query"))
356        .and_then(Value::as_str)
357        .map(|s| format!(" '{s}'"))
358        .unwrap_or_default();
359    format!("op '{op}'{extra}: {e}")
360}
