1//! Tamias home resolution — copied from `tamias-driver` (the helpers
2//! are tiny and pavo must not depend on tamias crates; the *layout*
3//! is the shared contract, not the code).
4//!
5//! `<tamias-home>` resolves as `$TAMIAS_HOME` →
6//! `$XDG_DATA_HOME/tamias` → `~/.local/share/tamias` → `./.tamias`.
7//! Pavo-capable drivers live alongside install drivers in
8//! `<tamias-home>/drivers/` — one dir, `manifest`/`capabilities`
9//! disambiguates (P-003 locked decision: no second discovery dir).
10
11use std::path::PathBuf;
12
13/// `<tamias-home>` — `$TAMIAS_HOME` → `$XDG_DATA_HOME/tamias` →
14/// `~/.local/share/tamias` → `./.tamias`.
15#[must_use]
16pub fn tamias_home() -> PathBuf {
17    std::env::var_os("TAMIAS_HOME")
18        .map(PathBuf::from)
19        .or_else(|| std::env::var_os("XDG_DATA_HOME").map(|d| PathBuf::from(d).join("tamias")))
20        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".local/share/tamias")))
21        .unwrap_or_else(|| PathBuf::from("./.tamias"))
22}
23
24/// The `<tamias-home>/drivers` dir — where installed products drop
25/// their `tamias-driver-*` executables.
26#[must_use]
27pub fn drivers_dir() -> PathBuf {
28    tamias_home().join("drivers")
29}

31#[cfg(test)]
32mod tests {
33    #![allow(clippy::unwrap_used)]
34    use super::*;
35
36    // One test, sequential env mutation — parallel tests mutating the
37    // same vars would race each other (and anything reading env).
38    #[test]
39    fn tamias_home_resolution_precedence() {
40        let tmp = tempfile::tempdir().unwrap();
41
42        // $TAMIAS_HOME wins outright.
43        let custom = tmp.path().join("custom-home");
44        std::env::set_var("TAMIAS_HOME", &custom);
45        std::env::set_var("XDG_DATA_HOME", tmp.path().join("xdg"));
46        assert_eq!(tamias_home(), custom);
47        assert_eq!(drivers_dir(), custom.join("drivers"));
48
49        // Without it, $XDG_DATA_HOME/tamias.
50        std::env::remove_var("TAMIAS_HOME");
51        assert_eq!(tamias_home(), tmp.path().join("xdg/tamias"));
52
53        std::env::remove_var("XDG_DATA_HOME");
54    }
55}
