1//! Contribution merge — fold every discovered driver's
2//! [`pavo_protocol::Contributes`] into one [`Chrome`] description the
3//! renderers lay out.
4//!
5//! ## Ordering policy (deterministic)
6//!
7//! Within each slot, entries sort by **`priority` ascending — lower
8//! wins** — per `pavo_protocol::Contribution`'s contract ("lower
9//! sorts first"). Ties break by **driver kind ascending**, then
10//! **contribution id ascending**. The protocol doc says ties break by
11//! "driver discovery order"; we substitute driver name because PATH
12//! order is environment-dependent and nondeterministic across
13//! machines — name order is stable and gives the same "one winner"
14//! guarantee. Same-slot duplicate ids from *different* drivers can't
15//! collide: every id is namespaced `<driver>:<id>` at merge time.
16//!
17//! `extra` slots (unknown to this host version) merge the same way
18//! and pass through untouched — forward compatibility per the
19//! protocol's degrade-gracefully rule.
20
21use std::collections::BTreeMap;
22
23use pavo_protocol::{Contribution, DriverManifest};
24
25/// One slot entry after merging — the contribution plus which driver
26/// it came from.
27#[derive(Clone, Debug, PartialEq)]
28pub struct SlotEntry {
29    /// Driver kind (`<type>` of `tamias-driver-<type>`).
30    pub driver: String,
31    /// Driver display title (manifest `title`).
32    pub driver_title: String,
33    /// The contribution, with `id` already namespaced as
34    /// `<driver>:<id>`.
35    pub contribution: Contribution,
36}
37
38/// The composed chrome — every slot merged across all drivers.
39#[derive(Clone, Debug, Default, PartialEq)]
40pub struct Chrome {
41    /// Top navigation bar entries.
42    pub navbar: Vec<SlotEntry>,
43    /// Shared dashboard panels.
44    pub dashboard: Vec<SlotEntry>,
45    /// Bottom status bar entries.
46    pub statusbar: Vec<SlotEntry>,
47    /// Standalone routes drivers own outside their views.
48    pub routes: Vec<SlotEntry>,
49    /// Unknown/forward-compat slots, merged the same way, keyed by
50    /// slot name. Renderers ignore slots they don't know.
51    pub extra: BTreeMap<String, Vec<SlotEntry>>,
52}
53
54/// Merge `drivers` — `(kind, manifest)` pairs — into a [`Chrome`].
55///
56/// Drivers with no `contributes` contribute nothing; empty slots stay
57/// empty. Ordering per the module docs.
58#[must_use]
59pub fn merge_chrome<'a, I>(drivers: I) -> Chrome
60where
61    I: IntoIterator<Item = (&'a str, &'a DriverManifest)>,
62{
63    let mut chrome = Chrome::default();
64    for (kind, manifest) in drivers {
65        let title = manifest.title.as_str();
66        collect(
67            &mut chrome.navbar,
68            kind,
69            title,
70            &manifest.contributes.navbar,
71        );
72        collect(
73            &mut chrome.dashboard,
74            kind,
75            title,
76            &manifest.contributes.dashboard,
77        );
78        collect(
79            &mut chrome.statusbar,
80            kind,
81            title,
82            &manifest.contributes.statusbar,
83        );
84        collect(
85            &mut chrome.routes,
86            kind,
87            title,
88            &manifest.contributes.routes,
89        );
90        for (slot, entries) in &manifest.contributes.extra {
91            let target = chrome.extra.entry(slot.clone()).or_default();
92            for c in entries {
93                target.push(slot_entry(kind, title, c));
94            }
95        }
96    }
97    sort_slot(&mut chrome.navbar);
98    sort_slot(&mut chrome.dashboard);
99    sort_slot(&mut chrome.statusbar);
100    sort_slot(&mut chrome.routes);
101    for entries in chrome.extra.values_mut() {
102        sort_slot(entries);
103    }
104    chrome
105}
106
107/// Append one driver's contributions to a slot, namespacing ids.
108fn collect(out: &mut Vec<SlotEntry>, kind: &str, title: &str, entries: &[Contribution]) {
109    for c in entries {
110        out.push(slot_entry(kind, title, c));
111    }
112}
113
114/// Build a [`SlotEntry`] with the id namespaced `<driver>:<id>`.
115fn slot_entry(kind: &str, title: &str, c: &Contribution) -> SlotEntry {
116    let mut contribution = c.clone();
117    contribution.id = format!("{kind}:{}", c.id);
118    SlotEntry {
119        driver: kind.to_string(),
120        driver_title: title.to_string(),
121        contribution,
122    }
123}
124
125/// The deterministic order: priority asc (lower wins), driver kind
126/// asc, id asc.
127fn sort_slot(entries: &mut [SlotEntry]) {
128    entries.sort_by(|a, b| {
129        a.contribution
130            .priority
131            .cmp(&b.contribution.priority)
132            .then_with(|| a.driver.cmp(&b.driver))
133            .then_with(|| a.contribution.id.cmp(&b.contribution.id))
134    });
135}

138#[cfg_attr(coverage, coverage(off))]

142    use super::*;

145    fn manifest(name: &str, title: &str, contributes: Contributes) -> DriverManifest {
146        DriverManifest {
147            name: name.to_string(),
148            title: title.to_string(),
149            icon: None,
150            version: "0.0.0".to_string(),
151            verbs: vec!["serve".to_string()],
152            capabilities: vec!["pavo".to_string()],
153            contributes,
154            views: vec![],
155            renderers: vec![],
156        }
157    }
158
159    fn contrib(id: &str, priority: i32) -> Contribution {
160        Contribution {
161            id: id.to_string(),
162            title: id.to_string(),
163            route: format!("/{id}"),
164            priority,
165            icon: None,
166        }
167    }
168
169    #[test]
170    fn merges_and_sorts_by_priority_then_driver() {
171        let a = manifest(
172            "aaa",
173            "A",
174            Contributes {
175                navbar: vec![contrib("late", 50), contrib("first", 10)],
176                ..Contributes::default()
177            },
178        );
179        let b = manifest(
180            "bbb",
181            "B",
182            Contributes {
183                navbar: vec![contrib("mid", 10)],
184                ..Contributes::default()
185            },
186        );
187        let chrome = merge_chrome([("aaa", &a), ("bbb", &b)]);
188        let ids: Vec<&str> = chrome
189            .navbar
190            .iter()
191            .map(|e| e.contribution.id.as_str())
192            .collect();
193        // priority 10 tie → driver name asc: aaa:mid? no — aaa has
194        // "first"@10 and "late"@50; bbb has "mid"@10. Order: the two
195        // priority-10 entries sort by driver: aaa:first, bbb:mid, then
196        // aaa:late.
197        assert_eq!(ids, ["aaa:first", "bbb:mid", "aaa:late"]);
198        assert_eq!(chrome.navbar[0].driver_title, "A");
199    }
200
201    #[test]
202    fn namespaces_ids_and_routes_statusbar_dashboard() {
203        let mut extra = std::collections::BTreeMap::new();
204        extra.insert("weird".to_string(), vec![contrib("w", 1)]);
205        let m = manifest(
206            "drv",
207            "Drv",
208            Contributes {
209                dashboard: vec![contrib("dash", 5)],
210                statusbar: vec![contrib("st", 1)],
211                routes: vec![contrib("r", 1)],
212                extra,
213                ..Contributes::default()
214            },
215        );
216        let chrome = merge_chrome([("drv", &m)]);
217        assert_eq!(chrome.dashboard[0].contribution.id, "drv:dash");
218        assert_eq!(chrome.statusbar[0].contribution.id, "drv:st");
219        assert_eq!(chrome.routes[0].contribution.id, "drv:r");
220        assert_eq!(
221            chrome.extra["weird"][0].contribution.id, "drv:w",
222            "extra slots merge + namespace too"
223        );
224    }
225
226    #[test]
227    fn empty_inputs_give_empty_chrome() {
228        let chrome = merge_chrome(std::iter::empty::<(&str, &DriverManifest)>());
229        assert!(chrome.navbar.is_empty() && chrome.extra.is_empty());
230    }
