1//! Contribution merge — fold every discovered driver's
2//! [`pavo_protocol::Contributes`] into one [`Chrome`] description the
3//! renderers lay out.
4//!
5//! ## Ordering policy (deterministic)
6//!
7//! Within each slot, entries sort by **`priority` ascending — lower
8//! wins** — per `pavo_protocol::Contribution`'s contract ("lower
9//! sorts first"). Ties break by **driver kind ascending**, then
10//! **contribution id ascending**. The protocol doc says ties break by
11//! "driver discovery order"; we substitute driver name because PATH
12//! order is environment-dependent and nondeterministic across
13//! machines — name order is stable and gives the same "one winner"
14//! guarantee. Same-slot duplicate ids from *different* drivers can't
15//! collide: every id is namespaced `<driver>:<id>` at merge time.
16//!
17//! `extra` slots (unknown to this host version) merge the same way
18//! and pass through untouched — forward compatibility per the
19//! protocol's degrade-gracefully rule. The platform slots
20//! (`sidebar`, `headerActions`, `notifications`, `jobs`) are typed
21//! [`Chrome`] fields like the original four; manifests from older
22//! hosts that carried them in `extra` still merge — they just land
23//! in `extra` instead.
24//!
25//! `notifications`/`jobs` slot *entries* are display metadata only —
26//! the feed data flows through the `notifications`/`jobs` wire ops
27//! ([`crate::host::Host::notifications`] / [`crate::host::Host::jobs`]).
28
29use std::collections::BTreeMap;
30
31use pavo_protocol::{Contribution, DriverManifest};
32
33/// One slot entry after merging — the contribution plus which driver
34/// it came from.
35#[derive(Clone, Debug, PartialEq)]
36pub struct SlotEntry {
37    /// Driver kind (`<type>` of `tamias-driver-<type>`).
38    pub driver: String,
39    /// Driver display title (manifest `title`).
40    pub driver_title: String,
41    /// The contribution, with `id` already namespaced as
42    /// `<driver>:<id>`.
43    pub contribution: Contribution,
44}
45
46/// The composed chrome — every slot merged across all drivers.
47#[derive(Clone, Debug, Default, PartialEq)]
48pub struct Chrome {
49    /// Top navigation bar entries.
50    pub navbar: Vec<SlotEntry>,
51    /// Shared dashboard panels.
52    pub dashboard: Vec<SlotEntry>,
53    /// Bottom status bar entries.
54    pub statusbar: Vec<SlotEntry>,
55    /// Standalone routes drivers own outside their views.
56    pub routes: Vec<SlotEntry>,
57    /// Left-sidebar shell entries (`contributes.sidebar`).
58    pub sidebar: Vec<SlotEntry>,
59    /// Top-bar actions (`contributes.headerActions` — wire spelling;
60    /// the field is snake_case here like everywhere else).
61    pub header_actions: Vec<SlotEntry>,
62    /// Notification-provider display entries
63    /// (`contributes.notifications`) — metadata only; the feed data
64    /// comes from the `notifications` wire op.
65    pub notifications: Vec<SlotEntry>,
66    /// Job-feed provider display entries (`contributes.jobs`) —
67    /// metadata only; data comes from the `jobs` wire op.
68    pub jobs: Vec<SlotEntry>,
69    /// Unknown/forward-compat slots, merged the same way, keyed by
70    /// slot name. Renderers ignore slots they don't know.
71    pub extra: BTreeMap<String, Vec<SlotEntry>>,
72}
73
74/// Merge `drivers` — `(kind, manifest)` pairs — into a [`Chrome`].
75///
76/// Drivers with no `contributes` contribute nothing; empty slots stay
77/// empty. Ordering per the module docs.
78#[must_use]
79pub fn merge_chrome<'a, I>(drivers: I) -> Chrome
80where
81    I: IntoIterator<Item = (&'a str, &'a DriverManifest)>,
82{
83    let mut chrome = Chrome::default();
84    for (kind, manifest) in drivers {
85        let title = manifest.title.as_str();
86        collect(
87            &mut chrome.navbar,
88            kind,
89            title,
90            &manifest.contributes.navbar,
91        );
92        collect(
93            &mut chrome.dashboard,
94            kind,
95            title,
96            &manifest.contributes.dashboard,
97        );
98        collect(
99            &mut chrome.statusbar,
100            kind,
101            title,
102            &manifest.contributes.statusbar,
103        );
104        collect(
105            &mut chrome.routes,
106            kind,
107            title,
108            &manifest.contributes.routes,
109        );
110        collect(
111            &mut chrome.sidebar,
112            kind,
113            title,
114            &manifest.contributes.sidebar,
115        );
116        collect(
117            &mut chrome.header_actions,
118            kind,
119            title,
120            &manifest.contributes.header_actions,
121        );
122        collect(
123            &mut chrome.notifications,
124            kind,
125            title,
126            &manifest.contributes.notifications,
127        );
128        collect(&mut chrome.jobs, kind, title, &manifest.contributes.jobs);
129        for (slot, entries) in &manifest.contributes.extra {
130            let target = chrome.extra.entry(slot.clone()).or_default();
131            for c in entries {
132                target.push(slot_entry(kind, title, c));
133            }
134        }
135    }
136    sort_slot(&mut chrome.navbar);
137    sort_slot(&mut chrome.dashboard);
138    sort_slot(&mut chrome.statusbar);
139    sort_slot(&mut chrome.routes);
140    sort_slot(&mut chrome.sidebar);
141    sort_slot(&mut chrome.header_actions);
142    sort_slot(&mut chrome.notifications);
143    sort_slot(&mut chrome.jobs);
144    for entries in chrome.extra.values_mut() {
145        sort_slot(entries);
146    }
147    chrome
148}
149
150/// Append one driver's contributions to a slot, namespacing ids.
151fn collect(out: &mut Vec<SlotEntry>, kind: &str, title: &str, entries: &[Contribution]) {
152    for c in entries {
153        out.push(slot_entry(kind, title, c));
154    }
155}
156
157/// Build a [`SlotEntry`] with the id namespaced `<driver>:<id>`.
158fn slot_entry(kind: &str, title: &str, c: &Contribution) -> SlotEntry {
159    let mut contribution = c.clone();
160    contribution.id = format!("{kind}:{}", c.id);
161    SlotEntry {
162        driver: kind.to_string(),
163        driver_title: title.to_string(),
164        contribution,
165    }
166}
167
168/// The deterministic order: priority asc (lower wins), driver kind
169/// asc, id asc.
170fn sort_slot(entries: &mut [SlotEntry]) {
171    entries.sort_by(|a, b| {
172        a.contribution
173            .priority
174            .cmp(&b.contribution.priority)
175            .then_with(|| a.driver.cmp(&b.driver))
176            .then_with(|| a.contribution.id.cmp(&b.contribution.id))
177    });
178}

188    fn manifest(name: &str, title: &str, contributes: Contributes) -> DriverManifest {
189        DriverManifest {
190            name: name.to_string(),
191            title: title.to_string(),
192            icon: None,
193            version: "0.0.0".to_string(),
194            verbs: vec!["serve".to_string()],
195            capabilities: vec!["pavo".to_string()],
196            deps: None,
197            contributes,
198            views: vec![],
199            renderers: vec![],
200        }
201    }
202
203    fn contrib(id: &str, priority: i32) -> Contribution {
204        Contribution {
205            id: id.to_string(),
206            title: id.to_string(),
207            route: format!("/{id}"),
208            priority,
209            icon: None,
210        }
211    }

213    #[test]
214    fn merges_and_sorts_by_priority_then_driver() {
215        let a = manifest(
216            "aaa",
217            "A",
218            Contributes {
219                navbar: vec![contrib("late", 50), contrib("first", 10)],
220                ..Contributes::default()
221            },
222        );
223        let b = manifest(
224            "bbb",
225            "B",
226            Contributes {
227                navbar: vec![contrib("mid", 10)],
228                ..Contributes::default()
229            },
230        );
231        let chrome = merge_chrome([("aaa", &a), ("bbb", &b)]);
232        let ids: Vec<&str> = chrome
233            .navbar
234            .iter()
235            .map(|e| e.contribution.id.as_str())
236            .collect();
237        // priority 10 tie → driver name asc: aaa:mid? no — aaa has
238        // "first"@10 and "late"@50; bbb has "mid"@10. Order: the two
239        // priority-10 entries sort by driver: aaa:first, bbb:mid, then
240        // aaa:late.
241        assert_eq!(ids, ["aaa:first", "bbb:mid", "aaa:late"]);
242        assert_eq!(chrome.navbar[0].driver_title, "A");
243    }

245    #[test]
246    fn namespaces_ids_and_routes_statusbar_dashboard() {
247        let mut extra = std::collections::BTreeMap::new();

249        let m = manifest(

264        assert_eq!(
265            chrome.extra["weird"][0].contribution.id, "drv:w",
266            "extra slots merge + namespace too"

290    #[test]
291    fn platform_slots_sort_across_drivers() {
292        let a = manifest(
293            "aaa",
294            "A",
295            Contributes {
296                sidebar: vec![contrib("s", 50)],
297                ..Contributes::default()
298            },
299        );
300        let b = manifest(
301            "bbb",
302            "B",
303            Contributes {
304                sidebar: vec![contrib("s", 10)],
305                header_actions: vec![contrib("h", 1)],
306                ..Contributes::default()
307            },
308        );
309        let chrome = merge_chrome([("bbb", &b), ("aaa", &a)]);
310        // priority asc wins regardless of input/driver order.
311        assert_eq!(
312            chrome
313                .sidebar
314                .iter()
315                .map(|e| e.contribution.id.as_str())
316                .collect::<Vec<_>>(),
317            ["bbb:s", "aaa:s"]
318        );
319        assert_eq!(chrome.header_actions.len(), 1);
320        assert_eq!(chrome.header_actions[0].driver, "bbb");
321        // Untouched slots stay empty.
322        assert!(chrome.notifications.is_empty() && chrome.jobs.is_empty());
323    }
