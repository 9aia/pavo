1//! ViewNode IR helpers — the walk/evaluate utilities renderers need.
2//!
3//! The IR is a tree of [`ViewNode`]s; containers (`panel`, `row`,
4//! `column`) nest `children`, everything else is a leaf. These
5//! helpers give renderers three things without re-implementing the
6//! traversal:
7//!
8//! - [`walk`] / [`find_nodes`] / [`children`] / [`kind_of`] — tree
9//!   traversal and kind inspection.
10//! - [`collect_queries`] — every `NodeBase::query` binding name in a
11//!   tree, so a renderer/host knows which view-model keys a tree
12//!   consumes.
13//! - KV degradation — [`to_kv_fallback`] turns any *known*
14//!   [`ViewNode`] into a [`KvNode`]; [`node_kind_json`] +
15//!   [`kv_fallback_json`] do the same for a raw `serde_json::Value`
16//!   whose `kind` this host doesn't know (the protocol's
17//!   forward-compat rule: unknown kinds degrade to generic KV, never
18//!   fail).
19
20use std::collections::BTreeSet;
21
22use pavo_protocol::{KvNode, ViewNode};
23use serde_json::{Map, Value};
24
25/// The `kind` discriminator of a node, as the wire spells it
26/// (`"panel"`, `"table"`, …).
27#[must_use]
28pub fn kind_of(node: &ViewNode) -> &'static str {
29    match node {
30        ViewNode::Panel(_) => "panel",
31        ViewNode::Table(_) => "table",
32        ViewNode::Tree(_) => "tree",
33        ViewNode::Graph(_) => "graph",
34        ViewNode::Form(_) => "form",
35        ViewNode::Kv(_) => "kv",
36        ViewNode::Markdown(_) => "markdown",
37        ViewNode::Embed(_) => "embed",
38        ViewNode::Row(_) => "row",
39        ViewNode::Column(_) => "column",
40    }
41}
42
43/// A node's children — `panel`/`row`/`column` nest `Vec<ViewNode>`;
44/// every other kind is a leaf and yields an empty slice.
45#[must_use]
46pub fn children(node: &ViewNode) -> &[ViewNode] {
47    match node {
48        ViewNode::Panel(n) => &n.children,
49        ViewNode::Row(n) => &n.children,
50        ViewNode::Column(n) => &n.children,
51        _ => &[],
52    }
53}
54
55/// Depth-first pre-order walk over the whole tree — `f` sees every
56/// node exactly once, root first.
57pub fn walk<'a>(root: &'a ViewNode, f: &mut impl FnMut(&'a ViewNode)) {
58    f(root);
59    for child in children(root) {
60        walk(child, f);
61    }
62}
63
64/// All nodes in the tree whose [`kind_of`] equals `kind`,
65/// pre-order.
66#[must_use]
67pub fn find_nodes<'a>(root: &'a ViewNode, kind: &str) -> Vec<&'a ViewNode> {
68    let mut out = Vec::new();
69    walk(root, &mut |n| {
70        if kind_of(n) == kind {
71            out.push(n);
72        }
73    });
74    out
75}
76
77/// Every query binding name referenced by `NodeBase::query` anywhere
78/// in the tree — the set of view-model keys the tree consumes.
79/// Sorted (`BTreeSet`) for deterministic output.
80#[must_use]
81pub fn collect_queries(root: &ViewNode) -> BTreeSet<String> {
82    let mut out = BTreeSet::new();
83    walk(root, &mut |n| {
84        if let Some(q) = &n.base().query {
85            out.insert(q.clone());
86        }
87    });
88    out
89}

91/// Degrade any known [`ViewNode`] to a [`KvNode`] — the universal
92/// fallback renderers use for kinds they don't implement (and the
93/// shape [`kv_fallback_json`] produces for unknown wire kinds).
94///
95/// The node's scalar fields become pairs (`kind`, `id`, `title`,
96/// plus kind-specific scalars like `text`/`component`); nested
97/// objects/arrays are skipped — a degraded node shows what it can.
98#[must_use]
99pub fn to_kv_fallback(node: &ViewNode) -> KvNode {
100    // Serializing a ViewNode can't fail in practice; on the impossible
101    // error we degrade to an empty kv rather than panic.
102    let value = serde_json::to_value(node).unwrap_or(Value::Null);
103    let json = kv_fallback_json(&value);
104    serde_json::from_value(json).unwrap_or_else(|_| KvNode {
105        base: node.base().clone(),
106        pairs: Vec::new(),
107    })
108}

110/// The `kind` string of a raw node JSON — `None` when absent or not
111/// a string. Renderers call this *before* deserializing so unknown
112/// kinds can route to [`kv_fallback_json`] instead of failing.
113#[must_use]
114pub fn node_kind_json(node: &Value) -> Option<&str> {
115    node.get("kind").and_then(Value::as_str)
116}
117
118/// Degrade a raw node JSON object to `{"kind":"kv", …}` — same rule
119/// as [`to_kv_fallback`] but for trees that haven't (or can't)
120/// deserialize into [`ViewNode`] because their `kind` is unknown to
121/// this host version.
122///
123/// Output shape: `kind` forced to `"kv"`; `id`/`title`/`query`/
124/// `description` are preserved as node fields when scalar; `pairs[]`
125/// gets a synthesized `{"key":"kind","value":<original kind>}` entry
126/// followed by every other scalar field. Objects/arrays don't become
127/// pairs. Non-object input yields an empty kv node.
128#[must_use]
129pub fn kv_fallback_json(node: &Value) -> Value {
130    let mut out = Map::new();
131    out.insert("kind".to_string(), Value::from("kv"));
132    let Some(obj) = node.as_object() else {
133        out.insert("pairs".to_string(), Value::Array(Vec::new()));
134        return Value::Object(out);
135    };
136    for key in ["id", "title", "query", "description"] {
137        if let Some(v) = obj.get(key).filter(|v| is_scalar(v)) {
138            out.insert(key.to_string(), v.clone());
139        }
140    }
141    let mut pairs: Vec<Value> = Vec::new();
142    // The original kind survives as the first pair — a degraded view
143    // can still show "this was a hologram".
144    if let Some(kind) = node_kind_json(node) {
145        let mut pair = Map::new();
146        pair.insert("key".to_string(), Value::from("kind"));
147        pair.insert("value".to_string(), Value::from(kind));
148        pairs.push(Value::Object(pair));
149    }
150    pairs.extend(obj.iter().filter_map(|(k, v)| {
151        if matches!(
152            k.as_str(),
153            "kind" | "id" | "title" | "query" | "description"
154        ) || !is_scalar(v)
155        {
156            return None;
157        }
158        let mut pair = Map::new();
159        pair.insert("key".to_string(), Value::from(k.as_str()));
160        pair.insert("value".to_string(), v.clone());
161        Some(Value::Object(pair))
162    }));
163    out.insert("pairs".to_string(), Value::Array(pairs));
164    Value::Object(out)
165}
166
167/// Scalars become kv pairs; objects/arrays don't (they'd nest
168/// structure a degraded renderer can't show anyway).
169fn is_scalar(v: &Value) -> bool {
170    !(v.is_object() || v.is_array())
171}

173#[cfg(test)]
174#[cfg_attr(coverage, coverage(off))]
175mod tests {
176    #![allow(clippy::unwrap_used)]
177
178    use super::*;
179    use pavo_protocol::KvNode;
180    use serde_json::json;
181
182    fn tree() -> ViewNode {
183        serde_json::from_value(json!({
184            "kind": "panel", "title": "Root",
185            "children": [
186                {"kind": "table", "id": "t1", "query": "items",
187                 "columns": [{"key": "id", "title": "ID"}],
188                 "rows": [["a"]]},
189                {"kind": "column", "children": [
190                    {"kind": "kv", "query": "meta",
191                     "pairs": [{"key": "k", "value": 1}]},
192                    {"kind": "markdown", "text": "hi"}
193                ]}
194            ]
195        }))
196        .unwrap()
197    }
198
199    #[test]
200    fn kind_children_and_walk() {
201        let root = tree();
202        assert_eq!(kind_of(&root), "panel");
203        assert_eq!(children(&root).len(), 2);
204        assert!(children(&children(&root)[0]).is_empty());
205        let mut kinds = Vec::new();
206        walk(&root, &mut |n| kinds.push(kind_of(n).to_string()));
207        assert_eq!(kinds, ["panel", "table", "column", "kv", "markdown"]);
208    }
209
210    #[test]
211    fn find_nodes_and_queries() {
212        let root = tree();
213        let tables = find_nodes(&root, "table");
214        assert_eq!(tables.len(), 1);
215        assert_eq!(tables[0].base().id.as_deref(), Some("t1"));
216        assert!(find_nodes(&root, "graph").is_empty());
217        let queries = collect_queries(&root);
218        assert!(queries.contains("items"));
219        assert!(queries.contains("meta"));
220        assert_eq!(queries.len(), 2);
221    }
222
223    #[test]
224    fn kv_fallback_for_known_node() {
225        let node: ViewNode = serde_json::from_value(json!({
226            "kind": "markdown", "id": "m", "title": "Note",
227            "text": "hello", "props": {"a": 1}
228        }))
229        .unwrap();
230        let kv = to_kv_fallback(&node);
231        assert_eq!(kv.base.id.as_deref(), Some("m"));
232        assert_eq!(kv.base.title.as_deref(), Some("Note"));
233        let keys: Vec<&str> = kv.pairs.iter().map(|p| p.key.as_str()).collect();
234        assert!(keys.contains(&"kind"), "original kind survives as a pair");
235        assert!(keys.contains(&"text"));
236        assert!(
237            !keys.contains(&"title"),
238            "title is a node field, not a pair"
239        );
240        assert!(!keys.contains(&"props"), "objects don't degrade to pairs");
241    }
242
243    #[test]
244    fn kv_fallback_json_for_unknown_kind() {
245        let raw = json!({
246            "kind": "hologram", "id": "h1", "title": "H",
247            "intensity": 0.8, "nested": {"x": 1}
248        });
249        assert_eq!(node_kind_json(&raw), Some("hologram"));
250        let kv = kv_fallback_json(&raw);
251        assert_eq!(kv["kind"], "kv");
252        assert_eq!(kv["id"], "h1");
253        assert_eq!(kv["title"], "H");
254        let pairs = kv["pairs"].as_array().unwrap();
255        // synthesized kind pair + scalar fields; nested object skipped
256        assert_eq!(pairs.len(), 2);
257        assert_eq!(pairs[0]["key"], "kind");
258        assert_eq!(pairs[0]["value"], "hologram");
259        assert_eq!(pairs[1]["key"], "intensity");
260        // non-object input → empty kv
261        let empty = kv_fallback_json(&json!(42));
262        assert_eq!(empty["kind"], "kv");
263        assert!(empty["pairs"].as_array().unwrap().is_empty());
264    }
265
266    #[test]
267    fn node_kind_json_absent() {
268        assert_eq!(node_kind_json(&json!({})), None);
269        assert_eq!(node_kind_json(&json!({"kind": 3})), None);
270    }
271
272    #[test]
273    fn typed_kv_node_still_parses() {
274        // sanity: the fallback output deserializes as KvNode
275        let kv: KvNode = serde_json::from_value(kv_fallback_json(&json!({
276            "kind": "x", "a": 1
277        })))
278        .unwrap();
279        assert_eq!(kv.pairs.len(), 2); // synthesized kind pair + "a"
280    }
