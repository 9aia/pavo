1//! Renderer registry — collect `renderers` declarations from every
2//! discovered driver's manifest into a kind → driver lookup, and find
3//! the plugin-rendered nodes inside a resolved view tree.
4//!
5//! This is the manifest half of the view-kind renderer contract (the
6//! wire half is the `render` op, see [`crate::client::DriverClient::render`]):
7//! a driver declares [`pavo_protocol::RendererDecl`]s; the host routes
8//! matching nodes to that driver's `serve` process and gets back a
9//! [`pavo_protocol::RenderModel`].
10//!
11//! ## Claim resolution (deterministic)
12//!
13//! - **Explicit kinds beat `"*"`.** A renderer listing `"graph"`
14//!   always wins over a catch-all for the same node.
15//! - **Among competing claims** for one kind (or the wildcard pool),
16//!   lowest [`RendererDecl::priority`] wins; ties break by driver
17//!   kind ascending, then renderer id ascending — the same policy
18//!   [`crate::merge`] applies to chrome contributions (driver-name
19//!   order substitutes for discovery order for determinism).
20//! - **Built-in kinds are not reserved.** A plugin may claim `graph`
21//!   (or even `table`) and override the host's built-in rendering on
22//!   the surfaces it declares — that is the migration path. Kinds no
23//!   plugin claims keep their built-in interpretation, and unknown
24//!   kinds still degrade to KV.
25//!
26//! Surface matching: `"cli"` requires `surfaces.cli`; a `format`
27//! argument further narrows to `cli.formats` (empty = wildcard,
28//! claims every format including `text`). `"tui"` requires
29//! `surfaces.tui == true`; `"web"` requires `surfaces.web` present.
30
31use std::collections::BTreeMap;
32
33use pavo_protocol::{DriverManifest, RendererDecl, ViewNode};
34use serde_json::Value;

36/// A renderer claim resolved to its driver — what the host needs to
37/// route a `render` op.
38#[derive(Clone, Debug, PartialEq)]
39pub struct RendererBinding {
40    /// Driver kind that owns the renderer.
41    pub driver: String,
42    /// Driver display title (manifest `title`).
43    pub driver_title: String,
44    /// The renderer's namespaced id — `<driver>:<decl.id>`.
45    pub id: String,
46    /// The declaration (kinds/surfaces/priority/entry).
47    pub decl: RendererDecl,
48}
49
50/// The merged renderer registry across all discovered drivers.
51#[derive(Clone, Debug, Default)]
52pub struct RendererRegistry {
53    /// kind → winning binding for explicitly claimed kinds.
54    explicit: BTreeMap<String, RendererBinding>,
55    /// Winning `"*"` claim (at most one — same conflict policy).
56    wildcard: Option<RendererBinding>,
57    /// Every binding, for `/api/renderers`-style listings (sorted by
58    /// namespaced id).
59    all: Vec<RendererBinding>,
60}
61
62/// One plugin-rendered node found in a view tree.
63#[derive(Clone, Debug)]
64pub struct PluginNode {
65    /// Tree path — child indices joined by `.`; the root is `""`.
66    /// Stable key renderers use to match results back to nodes.
67    pub path: String,
68    /// The node's `kind`.
69    pub kind: String,
70    /// The winning binding for this node+surface.
71    pub binding: RendererBinding,
72    /// The node serialized to raw JSON — passed verbatim to the
73    /// driver's `render` op.
74    pub node: Value,
75}
76
77impl RendererRegistry {
78    /// Build the registry from `(driver kind, manifest)` pairs —
79    /// same input shape as [`crate::merge::merge_chrome`]. Drivers
80    /// with no `renderers` contribute nothing.
81    #[must_use]
82    pub fn from_manifests<'a, I>(drivers: I) -> Self
83    where
84        I: IntoIterator<Item = (&'a str, &'a DriverManifest)>,
85    {
86        let mut candidates: Vec<RendererBinding> = Vec::new();
87        for (kind, manifest) in drivers {
88            for decl in &manifest.renderers {
89                candidates.push(RendererBinding {
90                    driver: kind.to_string(),
91                    driver_title: manifest.title.clone(),
92                    id: format!("{kind}:{}", decl.id),
93                    decl: decl.clone(),
94                });
95            }
96        }
97        // Deterministic candidate order: priority asc, driver kind
98        // asc, renderer id asc — first wins per kind.
99        candidates.sort_by(|a, b| {
100            a.decl
101                .priority
102                .cmp(&b.decl.priority)
103                .then_with(|| a.driver.cmp(&b.driver))
104                .then_with(|| a.id.cmp(&b.id))
105        });
106        let mut registry = Self::default();
107        for binding in candidates {
108            if binding.decl.kinds.iter().any(|k| k == "*") {
109                if registry.wildcard.is_none() {
110                    registry.wildcard = Some(binding.clone());
111                }
112                registry.all.push(binding);
113                continue;
114            }
115            for kind in &binding.decl.kinds {
116                registry
117                    .explicit
118                    .entry(kind.clone())
119                    .or_insert_with(|| binding.clone());
120            }
121            registry.all.push(binding);
122        }
123        registry.all.sort_by(|a, b| a.id.cmp(&b.id));
124        registry
125    }
126
127    /// The winning binding for node `kind` on `surface` — explicit
128    /// claims first, then the wildcard. `format` narrows CLI claims
129    /// (`cli.formats`; empty list = wildcard). Returns `None` when no
130    /// renderer claims the kind for that surface.
131    #[must_use]
132    pub fn for_kind(
133        &self,
134        kind: &str,
135        surface: &str,
136        format: Option<&str>,
137    ) -> Option<&RendererBinding> {
138        let binding = self.explicit.get(kind).or(self.wildcard.as_ref())?;
139        supports_surface(binding, surface, format).then_some(binding)
140    }
141
142    /// Whether ANY registered renderer claims `kind` on `surface` —
143    /// cheaper than [`RendererRegistry::for_kind`] when only presence
144    /// matters (e.g. "does this view need plugin renders?").
145    #[must_use]
146    pub fn handles(&self, kind: &str, surface: &str, format: Option<&str>) -> bool {
147        self.for_kind(kind, surface, format).is_some()
148    }
149
150    /// Every binding, sorted by namespaced id — the `/api/renderers`
151    /// listing shape.
152    #[must_use]
153    pub fn entries(&self) -> &[RendererBinding] {
154        &self.all
155    }
156
157    /// Number of renderer declarations registered.
158    #[must_use]
159    pub fn len(&self) -> usize {
160        self.all.len()
161    }
162
163    /// Whether the registry is empty.
164    #[must_use]
165    pub fn is_empty(&self) -> bool {
166        self.all.is_empty()
167    }
168}

170/// Does `binding` support `surface` (and `format` on cli)?
171fn supports_surface(binding: &RendererBinding, surface: &str, format: Option<&str>) -> bool {
172    match surface {
173        "cli" => match &binding.decl.surfaces.cli {
174            None => false,
175            Some(cli) => match format {
176                // No format asked (e.g. "does it handle cli at all")
177                // → presence of the cli surface is enough.
178                None => true,
179                Some(f) => cli.formats.is_empty() || cli.formats.iter().any(|x| x == f),
180            },
181        },
182        "tui" => binding.decl.surfaces.tui,
183        "web" => binding.decl.surfaces.web.is_some(),
184        // Unknown surfaces: only an explicit `"*"`-style surface can't
185        // exist — RendererSurfaces is closed — so unsupported.
186        _ => false,
187    }
188}

190/// Find every node in `root`'s tree claimed by a renderer plugin for
191/// `surface`/`format`, pre-order, with stable [`PluginNode::path`]
192/// keys. Nodes whose kind has no plugin claim are skipped — callers
193/// fall back to built-in rendering for them.
194///
195/// `data` is unused here (kept out of the signature — plugins resolve
196/// bindings themselves from the `data` map they're handed); it lives
197/// on [`PluginNode::node`] only via the node's own `query` field.
198#[must_use]
199pub fn plugin_nodes(
200    root: &ViewNode,
201    registry: &RendererRegistry,
202    surface: &str,
203    format: Option<&str>,
204) -> Vec<PluginNode> {
205    let mut out = Vec::new();
206    walk_plugin(root, registry, surface, format, String::new(), &mut out);
207    out
208}
209
210fn walk_plugin(
211    node: &ViewNode,
212    registry: &RendererRegistry,
213    surface: &str,
214    format: Option<&str>,
215    path: String,
216    out: &mut Vec<PluginNode>,
217) {
218    let kind = crate::viewnode::kind_of(node);
219    if let Some(binding) = registry.for_kind(kind, surface, format) {
220        // Serialize the node verbatim — unknown/future fields survive
221        // to the plugin (forward compat is the plugin's problem too).
222        let json = serde_json::to_value(node).unwrap_or(Value::Null);
223        out.push(PluginNode {
224            path: path.clone(),
225            kind: kind.to_string(),
226            binding: binding.clone(),
227            node: json,
228        });
229        // A plugin-rendered container still counts once — plugins own
230        // the whole subtree semantics for claimed kinds. But children
231        // of UNCLAIMED containers must still be visited: only descend
232        // when this node itself wasn't claimed.
233        return;
234    }
235    for (i, child) in pavo_core_children(node).iter().enumerate() {
236        let child_path = if path.is_empty() {
237            i.to_string()
238        } else {
239            format!("{path}.{i}")
240        };
241        walk_plugin(child, registry, surface, format, child_path, out);
242    }
243}
244
245/// Children accessor — delegates to [`crate::viewnode::children`].
246fn pavo_core_children(node: &ViewNode) -> &[ViewNode] {
247    crate::viewnode::children(node)
248}

258    fn manifest(renderers: Vec<RendererDecl>) -> DriverManifest {
259        DriverManifest {
260            name: "drv".to_string(),
261            title: "Drv".to_string(),
262            icon: None,
263            version: "0".to_string(),
264            verbs: vec!["serve".to_string()],
265            capabilities: vec!["pavo".to_string()],
266            contributes: Default::default(),
267            views: vec![],
268            renderers,
269        }
270    }

272    fn decl(id: &str, kinds: &[&str], priority: i32) -> RendererDecl {
273        RendererDecl {
274            id: id.to_string(),
275            kinds: kinds.iter().map(|s| s.to_string()).collect(),
276            surfaces: RendererSurfaces {
277                cli: Some(CliSurface {
278                    formats: vec!["svg".to_string(), "text".to_string()],
279                }),
280                tui: true,
281                web: Some(WebSurface {
282                    component: None,
283                    island: true,
284                }),
285            },
286            priority,
287            entry: None,
288        }
289    }

291    #[test]
292    fn registry_resolves_explicit_and_wildcard() {
293        let graph = decl("graph", &["graph"], 100);
294        let any = decl("any", &["*"], 100);
295        let m1 = manifest(vec![graph]);
296        let m2 = manifest(vec![any]);
297        let reg = RendererRegistry::from_manifests([("aaa", &m1), ("zzz", &m2)]);
298        // explicit beats wildcard regardless of driver order
299        let b = reg.for_kind("graph", "cli", Some("svg")).unwrap();
300        assert_eq!(b.driver, "aaa");
301        // wildcard covers unclaimed kinds on its declared surfaces
302        let b = reg.for_kind("timeline", "tui", None).unwrap();
303        assert_eq!(b.driver, "zzz");
304        let b = reg.for_kind("table", "cli", Some("text")).unwrap();
305        assert_eq!(b.driver, "zzz");
306        // …but not on surfaces the wildcard doesn't declare
307        assert!(reg.for_kind("table", "watch", None).is_none());
308        // no claim at all for an empty-kind case is covered above;
309        // a registry with no renderers handles nothing.
310        assert!(RendererRegistry::default()
311            .for_kind("graph", "cli", Some("svg"))
312            .is_none());
313    }

315    #[test]
316    fn priority_and_driver_order_resolve_conflicts() {
317        // Two drivers claim "graph"; lower priority wins.
318        let hi = manifest(vec![decl("g", &["graph"], 50)]);
319        let lo = manifest(vec![decl("g", &["graph"], 10)]);
320        let reg = RendererRegistry::from_manifests([("aaa", &hi), ("zzz", &lo)]);
321        assert_eq!(
322            reg.for_kind("graph", "cli", Some("svg")).unwrap().driver,
323            "zzz"
324        );
325        // Same priority → driver kind asc.
326        let a = manifest(vec![decl("g", &["graph"], 10)]);
327        let b = manifest(vec![decl("g", &["graph"], 10)]);
328        let reg = RendererRegistry::from_manifests([("bbb", &b), ("aaa", &a)]);
329        assert_eq!(
330            reg.for_kind("graph", "cli", Some("svg")).unwrap().driver,
331            "aaa"
332        );
333    }

337        let m = manifest(vec![decl("g", &["graph"], 100)]);
338        let reg = RendererRegistry::from_manifests([("d", &m)]);
339        assert!(reg.for_kind("graph", "cli", Some("svg")).is_some());

357    #[test]
358    fn plugin_nodes_walks_tree_with_paths() {
359        let m = manifest(vec![decl("g", &["graph"], 100)]);
360        let reg = RendererRegistry::from_manifests([("d", &m)]);
361        let root: ViewNode = serde_json::from_value(json!({
362            "kind": "column", "children": [
363                {"kind": "kv", "pairs": []},
364                {"kind": "panel", "children": [
365                    {"kind": "graph", "nodes": [], "edges": []}
366                ]}
367            ]
368        }))
369        .unwrap();
370        let nodes = plugin_nodes(&root, &reg, "tui", None);
371        assert_eq!(nodes.len(), 1);
372        assert_eq!(nodes[0].path, "1.0");
373        assert_eq!(nodes[0].kind, "graph");
374        assert_eq!(nodes[0].node["kind"], "graph");
375        // cli+svg also finds it; cli+unlisted-format doesn't
376        assert_eq!(plugin_nodes(&root, &reg, "cli", Some("svg")).len(), 1);
377        assert!(plugin_nodes(&root, &reg, "cli", Some("dot")).is_empty());
378    }

382        // A plugin claiming "column" owns the whole subtree — its
383        // children aren't separately routed.
384        let m = manifest(vec![decl("c", &["column"], 100)]);
385        let reg = RendererRegistry::from_manifests([("d", &m)]);
386        let root: ViewNode = serde_json::from_value(json!({
387            "kind": "column", "children": [
388                {"kind": "graph", "nodes": [], "edges": []}

391        .unwrap();
392        let nodes = plugin_nodes(&root, &reg, "tui", None);
393        assert_eq!(nodes.len(), 1);
394        assert_eq!(nodes[0].kind, "column");
395        assert_eq!(nodes[0].path, "");
