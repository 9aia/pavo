1//! [`Host`] — the composition root renderers talk to.
2//!
3//! Lifecycle: [`Host::discover`] scans for pavo-capable drivers and
4//! keeps their manifests (eager where the `manifest` verb answered,
5//! lazy via the `serve` wire otherwise). From there:
6//!
7//! - [`Host::chrome`] merges every driver's `contributes` into one
8//!   [`Chrome`] (see [`crate::merge`] for the ordering policy).
9//! - [`Host::list_views`] / [`Host::list_routes`] enumerate what the
10//!   host can show — namespaced `<driver>:<view>` ids included.
11//! - [`Host::resolve_view`] turns a deep link (`"myco:board"`, or a
12//!   bare `"myco"` for the driver's first view) into a
13//!   [`ViewResult`] by spawning/using that driver's `serve` client.
14//! - [`Host::query`] / [`Host::client`] are the escape hatches for
15//!   view-model ops and form submits.
16//!
17//! View results are cached in a [`TtlCache`] (default 10s,
18//! [`Host::with_view_cache_ttl`]) keyed by `driver:view` + canonical
19//! params — repeat renders of the same deep link don't respawn
20//! round-trips. `serve` clients are spawned lazily on first use and
21//! live until the `Host` drops (each `DriverClient`'s `Drop` kills its
22//! child).
23//!
24//! [`Host`] is `Send + Sync` and every method takes `&self` — share
25//! one behind an `Arc` (e.g. `pavo serve`'s per-connection threads).
26//! Concurrency is serialized **per driver**: each `DriverClient` sits
27//! behind its own [`Mutex`], so requests to different drivers proceed
28//! in parallel while requests to the same driver stay one-in-flight
29//! on its single `serve` process (the ndjson wire is
30//! request/response ordered — interleaved writes would misattribute
31//! responses).
32//!
33//! Errors are typed: [`PavoError::UnknownDriver`] /
34//! [`PavoError::UnknownView`] / [`PavoError::BadViewRef`] for bad
35//! targets, driver/transport errors propagated from the client.
36

56/// The pavo host — discovered drivers, lazy serve clients, merged
57/// chrome, view cache. `Send + Sync`: all mutable state is behind
58/// mutexes, and the only lock held across a driver round-trip is that
59/// driver's own client lock.
60pub struct Host {
61    drivers: Vec<DriverInfo>,
62    /// Live `serve` clients keyed by driver kind. The outer mutex
63    /// guards only the map (lookup + lazy spawn); each value carries
64    /// its own [`Mutex`] so requests to different drivers run in
65    /// parallel while same-driver requests serialize on the wire.
66    clients: Mutex<HashMap<String, Arc<Mutex<DriverClient>>>>,
67    /// Manifests fetched lazily over the `serve` wire for drivers
68    /// whose `manifest` probe didn't answer eagerly (those stay on
69    /// [`DriverInfo::manifest`]).
70    manifest_cache: Mutex<HashMap<String, DriverManifest>>,
71    client_config: ClientConfig,
72    view_cache: TtlCache<ViewResult>,
73}
74
75/// A mutex-poisoned failure — a thread panicked while holding a host
76/// lock. Surfaced as `io` so callers see an ordinary error.
77fn poisoned(what: &str) -> PavoError {
78    PavoError::Io(std::io::Error::other(format!("{what} lock poisoned")))
79}

196    /// The driver's manifest — from discovery when eager, otherwise
197    /// fetched once over the `serve` wire's `manifest` op and cached
198    /// in [`Host::manifest_cache`].
199    ///
200    /// # Errors
201    /// [`PavoError::UnknownDriver`] for undiscovered kinds; client
202    /// errors propagate.
203    pub fn manifest(&self, kind: &str) -> Result<DriverManifest, PavoError> {
204        let info = self
205            .drivers
206            .iter()
207            .find(|d| d.kind == kind)
208            .ok_or_else(|| PavoError::UnknownDriver(kind.to_string()))?;
209        if let Some(manifest) = &info.manifest {
210            return Ok(manifest.clone());
211        }
212        if let Some(manifest) = self
213            .manifest_cache
214            .lock()
215            .map_err(|_| poisoned("manifest cache"))?
216            .get(kind)
217            .cloned()
218        {
219            return Ok(manifest);
220        }
221        // Two threads may race the fetch — both answer correctly and
222        // the last insert wins; the wire op is idempotent.
223        let manifest = self.with_client(kind, DriverClient::manifest)?;
224        self.manifest_cache
225            .lock()
226            .map_err(|_| poisoned("manifest cache"))?
227            .insert(kind.to_string(), manifest.clone());
228        Ok(manifest)
229    }

231    /// The live `serve` client for `kind`, spawning on first use.
232    ///
233    /// The returned [`Arc<Mutex<DriverClient>>`] is the per-driver
234    /// serialization point — [`Host::with_client`] locks it for you.
235    /// The spawn happens under the clients-map lock so two threads
236    /// can't double-spawn; the lock is released before any request.
237    ///
238    /// # Errors
239    /// [`PavoError::UnknownDriver`] / [`PavoError::MissingDriver`] /
240    /// spawn errors.
241    pub fn client(&self, kind: &str) -> Result<Arc<Mutex<DriverClient>>, PavoError> {
242        let mut map = self.clients.lock().map_err(|_| poisoned("clients"))?;
243        if let Some(client) = map.get(kind) {
244            return Ok(Arc::clone(client));
245        }
246        let info = self
247            .drivers
248            .iter()
249            .find(|d| d.kind == kind)
250            .ok_or_else(|| PavoError::UnknownDriver(kind.to_string()))?;
251        let client = DriverClient::spawn(info.path.clone(), self.client_config.clone())?;
252        let client = Arc::new(Mutex::new(client));
253        map.insert(kind.to_string(), Arc::clone(&client));
254        Ok(client)
255    }

257    /// Run `f` against `kind`'s client under its per-driver lock —
258    /// the only place a [`DriverClient`] is borrowed mutably.
259    fn with_client<T>(
260        &self,
261        kind: &str,
262        f: impl FnOnce(&mut DriverClient) -> Result<T, PavoError>,
263    ) -> Result<T, PavoError> {
264        let client = self.client(kind)?;
265        let mut guard = client.lock().map_err(|_| poisoned("driver client"))?;
266        f(&mut guard)
267    }

269    /// Merge every driver's `contributes` into one [`Chrome`].
270    /// Drivers whose lazy manifest can't be fetched are skipped —
271    /// their slots contribute nothing rather than failing the whole
272    /// chrome.
273    #[must_use]
274    pub fn chrome(&self) -> Chrome {
275        let kinds: Vec<String> = self.drivers.iter().map(|d| d.kind.clone()).collect();
276        let mut pairs: Vec<(String, DriverManifest)> = Vec::new();
277        for kind in kinds {
278            // A driver whose manifest can't be produced contributes
279            // nothing — skip it rather than fail the whole chrome.
280            if let Ok(m) = self.manifest(&kind) {
281                pairs.push((kind, m));
282            }
283        }
284        merge_chrome(pairs.iter().map(|(k, m)| (k.as_str(), m)))
285    }

287    /// Every view every driver offers, namespaced. Fetches lazy
288    /// manifests as needed; drivers that can't produce one are
289    /// skipped.
290    #[must_use]
291    pub fn list_views(&self) -> Vec<ViewEntry> {
292        let mut out = Vec::new();
293        let kinds: Vec<String> = self.drivers.iter().map(|d| d.kind.clone()).collect();
294        for kind in kinds {
295            let Ok(manifest) = self.manifest(&kind) else {
296                continue;
297            };
298            for spec in &manifest.views {
299                out.push(ViewEntry {
300                    driver: kind.clone(),
301                    driver_title: manifest.title.clone(),
302                    id: spec.id.clone(),
303                    namespaced: format!("{kind}:{}", spec.id),
304                    route: spec.route.clone(),
305                    title: spec.title.clone(),
306                    description: spec.description.clone(),
307                    queries: spec.queries.iter().map(|q| q.name.clone()).collect(),
308                });
309            }
310        }
311        out.sort_by(|a, b| a.namespaced.cmp(&b.namespaced));
312        out
313    }

315    /// Every navigable route: view routes (`RouteKind::View`) plus
316    /// `contributes.routes` slot entries (`RouteKind::Slot`).
317    #[must_use]
318    pub fn list_routes(&self) -> Vec<RouteEntry> {
319        let mut out = Vec::new();
320        for view in self.list_views() {
321            out.push(RouteEntry {
322                route: view.route,
323                driver: view.driver,
324                title: view.title,
325                target: view.namespaced,
326                kind: RouteKind::View,
327            });
328        }
329        for entry in self.chrome().routes {
330            out.push(RouteEntry {
331                route: entry.contribution.route.clone(),
332                driver: entry.driver.clone(),
333                title: entry.contribution.title.clone(),
334                target: entry.contribution.id.clone(),
335                kind: RouteKind::Slot("routes".to_string()),
336            });
337        }
338        out.sort_by(|a, b| a.route.cmp(&b.route).then(a.target.cmp(&b.target)));
339        out
340    }

342    /// Resolve a deep link to a rendered view.
343    ///
344    /// `target` is `"<driver>:<view>"` (e.g. `"myco:board"`); a bare
345    /// `"<driver>"` resolves to the driver's first declared view.
346    /// `params` are caller-supplied view params passed to the driver's
347    /// `view` op. Results are cached per `driver:view` + params.
348    ///
349    /// # Errors
350    /// [`PavoError::BadViewRef`] on unparseable targets,
351    /// [`PavoError::UnknownDriver`]/[`PavoError::UnknownView`] for
352    /// missing pieces, client errors propagate.
353    pub fn resolve_view(&self, target: &str, params: Params) -> Result<ResolvedView, PavoError> {
354        let (driver, view) = self.parse_target(target)?;
355        let manifest = self.manifest(&driver)?;
356        let view_id = match view {
357            Some(v) => v,
358            None => manifest
359                .views
360                .first()
361                .map(|v| v.id.clone())
362                .ok_or_else(|| PavoError::UnknownView {
363                    driver: driver.clone(),
364                    view: "<default>".to_string(),
365                })?,
366        };
367        if !manifest.views.iter().any(|v| v.id == view_id) {
368            return Err(PavoError::UnknownView {
369                driver,
370                view: view_id,
371            });
372        }
373
374        // Cache key: namespaced id + canonical params JSON.
375        let namespaced = format!("{driver}:{view_id}");
376        let key = format!(
377            "{namespaced}\u{1f}{}",
378            serde_json::to_string(&Value::Object(params.clone())).unwrap_or_default()
379        );
380        if let Some(result) = self.view_cache.get(&key) {
381            return Ok(ResolvedView {
382                driver,
383                view: view_id,
384                namespaced,
385                result,
386            });
387        }
388
389        let result = self.with_client(&driver, |client| client.view(&view_id, params))?;
390        self.view_cache.insert(key, result.clone());
391        Ok(ResolvedView {
392            driver,
393            view: view_id,
394            namespaced,
395            result,
396        })
397    }

413    /// `{"op":"submit"}` — submit a form's field values to the driver
414    /// that owns `target` (`"<driver>:<view>"`, bare `"<driver>"` =
415    /// its first view, same as [`Host::resolve_view`]). `form`
416    /// optionally names the form node's `id` when the view carries
417    /// several forms.
418    ///
419    /// On success every cached result for that `driver:view` is
420    /// invalidated — submits mutate driver state, so the next
421    /// `resolve_view` must re-fetch (a returned
422    /// [`SubmitResult::view`] is *not* inserted into the cache; it
423    /// belongs to the submitting renderer).
424    ///
425    /// # Errors
426    /// [`PavoError::BadViewRef`] / [`PavoError::UnknownDriver`] /
427    /// [`PavoError::UnknownView`] for bad targets; `ok:false` driver
428    /// answers (validation failures included — check
429    /// [`SubmitResult::errors`] on `Ok`) propagate as
430    /// [`PavoError::Driver`].
431    pub fn submit(
432        &self,
433        target: &str,
434        form: Option<&str>,
435        values: Params,
436    ) -> Result<SubmitResult, PavoError> {
437        let (driver, view) = self.parse_target(target)?;
438        let manifest = self.manifest(&driver)?;
439        let view_id = match view {
440            Some(v) => v,
441            None => manifest
442                .views
443                .first()
444                .map(|v| v.id.clone())
445                .ok_or_else(|| PavoError::UnknownView {
446                    driver: driver.clone(),
447                    view: "<default>".to_string(),
448                })?,
449        };
450        if !manifest.views.iter().any(|v| v.id == view_id) {
451            return Err(PavoError::UnknownView {
452                driver,
453                view: view_id,
454            });
455        }
456        let result = self.with_client(&driver, |client| client.submit(&view_id, form, values))?;
457        // Submits mutate state — drop every cached params variant of
458        // this view so the next resolve re-fetches.
459        self.view_cache
460            .invalidate_prefix(&format!("{driver}:{view_id}\u{1f}"));
461        Ok(result)
462    }

474    /// The merged renderer-plugin registry — every `renderers`
475    /// declaration across all discovered drivers' manifests, resolved
476    /// to kind → driver bindings (see [`crate::renderers`] for the
477    /// conflict policy). Fetches lazy manifests as needed; drivers

479    #[must_use]
480    pub fn renderers(&self) -> RendererRegistry {
481        let kinds: Vec<String> = self.drivers.iter().map(|d| d.kind.clone()).collect();
482        let mut pairs: Vec<(String, DriverManifest)> = Vec::new();
483        for kind in kinds {
484            if let Ok(m) = self.manifest(&kind) {

486            }
487        }
488        RendererRegistry::from_manifests(pairs.iter().map(|(k, m)| (k.as_str(), m)))
489    }

511    /// Route one node to whichever driver claims `kind` on `surface`
512    /// — the registry lookup + `render` op in one call. Used by
513    /// `pavo serve`'s `/api/render`.
514    ///
515    /// # Errors
516    /// [`PavoError::UnknownDriver`] when no renderer claims the kind
517    /// for that surface; client errors propagate.
518    pub fn render_by_kind(
519        &self,
520        kind: &str,
521        node: &Value,
522        data: &Params,
523        surface: &str,
524        format: Option<&str>,
525    ) -> Result<RenderModel, PavoError> {
526        let registry = self.renderers();
527        let binding =
528            registry
529                .for_kind(kind, surface, format)
530                .ok_or_else(|| PavoError::UnknownView {
531                    driver: "<renderers>".to_string(),
532                    view: format!("{kind} on {surface}"),
533                })?;
534        let driver = binding.driver.clone();
535        self.render_node(&driver, kind, node, data, surface, format)
536    }

538    /// Render every plugin-claimed node in a resolved view for
539    /// `surface` — one `render` op per node, keyed by the node's
540    /// [`crate::renderers::PluginNode::path`]. Nodes whose plugin
541    /// call fails are skipped (the caller falls back to built-in
542    /// rendering for that node — a broken plugin degrades one node,
543    /// never the whole view).
544    ///
545    /// `format` is the surface's format hint (CLI `--format`); pass
546    /// `None` for the surface default.
547    #[must_use]
548    pub fn render_view_plugins(
549        &self,
550        resolved: &ResolvedView,
551        surface: &str,
552        format: Option<&str>,
553    ) -> std::collections::HashMap<String, RenderModel> {
554        let registry = self.renderers();
555        let nodes = plugin_nodes(&resolved.result.view.root, &registry, surface, format);
556        let mut out = std::collections::HashMap::new();
557        // Group by driver so each client's requests stay sequential —
558        // one driver lock is held for the whole group (the wire is
559        // one-in-flight anyway), never two drivers' locks at once.
560        let mut by_driver: std::collections::BTreeMap<String, Vec<_>> =
561            std::collections::BTreeMap::new();
562        for n in nodes {
563            by_driver
564                .entry(n.binding.driver.clone())
565                .or_default()
566                .push(n);
567        }
568        for (driver, group) in by_driver {
569            let _ = self.with_client(&driver, |client| {
570                for n in group {
571                    if let Ok(model) =
572                        client.render(&n.kind, &n.node, &resolved.result.data, surface, format)
573                    {
574                        out.insert(n.path, model);
575                    }
576                }
577                Ok(())
578            });
579        }
580        out
581    }

583    /// Drop a cached view result — call after mutations that should
584    /// refresh the next `resolve_view`.
585    pub fn invalidate_view(&self, driver: &str, view: &str, params: &Params) {
586        let namespaced = format!("{driver}:{view}");
587        let key = format!(
588            "{namespaced}\u{1f}{}",
589            serde_json::to_string(&Value::Object(params.clone())).unwrap_or_default()
590        );
591        self.view_cache.invalidate(&key);
592    }

594    /// Clear the whole view cache.

599    /// Parse `"driver:view"` / `"driver"` into `(driver, Option<view>)`.
600    fn parse_target(&self, target: &str) -> Result<(String, Option<String>), PavoError> {
601        let target = target.trim();
602        if target.is_empty() {
603            return Err(PavoError::BadViewRef(target.to_string()));
604        }
605        match target.split_once(':') {
606            None => Ok((target.to_string(), None)),
607            Some((driver, view)) => {
608                if driver.is_empty() || view.is_empty() || view.contains(':') {
609                    return Err(PavoError::BadViewRef(target.to_string()));
610                }
611                Ok((driver.to_string(), Some(view.to_string())))
612            }
613        }
614    }

683    fn end_to_end_discover_chrome_views_resolve() {
684        let tmp = tempfile::tempdir().unwrap();
685        let bin = tmp.path().join("bin");
686        write_driver(&bin, "tamias-driver-fx", FX);
687        write_driver(&bin, "tamias-driver-lz", LZ);
688        let home = tmp.path().join("home");
689
690        // Short timeout: lz's serve never answers `submit`, and the
691        // test shouldn't wait out the default 10s.
692        let host = Host::discover_with(opts(&bin, &home)).with_client_config(ClientConfig {
693            timeout: Duration::from_secs(2),
694            max_restarts: 0,
695            ..ClientConfig::default()
696        });
697        assert_eq!(host.drivers().len(), 2);
698
699        // chrome merges eager + lazy manifests
700        let chrome = host.chrome();
701        assert_eq!(chrome.navbar.len(), 1);
702        assert_eq!(chrome.navbar[0].contribution.id, "fx:nav");
703        assert_eq!(chrome.statusbar.len(), 1);
704        assert_eq!(chrome.statusbar[0].driver, "lz");
705        // chrome.routes = contributes.routes only (view routes are
706        // NOT chrome entries) → just fx:set.
707        assert_eq!(chrome.routes.len(), 1);
708        assert_eq!(chrome.routes[0].contribution.id, "fx:set");
709
710        // views: fx:board + lz:home
711        let views = host.list_views();
712        assert_eq!(views.len(), 2);
713        let fx = views.iter().find(|v| v.driver == "fx").unwrap();
714        assert_eq!(fx.namespaced, "fx:board");
715
716        // routes include view routes + slot routes
717        let routes = host.list_routes();
718        assert!(routes
719            .iter()
720            .any(|r| r.kind == RouteKind::View && r.route == "/fx/board"));
721        assert!(routes.iter().any(|r| matches!(r.kind, RouteKind::Slot(_))));
722
723        // resolve: namespaced, bare-driver default, params pass-through
724        let v = host.resolve_view("fx:board", Params::new()).unwrap();
725        assert_eq!(v.namespaced, "fx:board");
726        assert_eq!(v.result.data["items"], serde_json::json!([1, 2]));
727        let v2 = host.resolve_view("lz", Params::new()).unwrap();
728        assert_eq!(v2.view, "home");
729        // cached: second resolve hits the cache (same result, cheap)
730        let v3 = host.resolve_view("fx:board", Params::new()).unwrap();
731        assert_eq!(v3.result.data["items"], serde_json::json!([1, 2]));
732
733        // query passthrough
734        let q = host.query("fx", "items", Params::new()).unwrap();
735        assert_eq!(q.data, serde_json::json!([1, 2]));
736
737        // submit: namespaced + bare-driver targets, then cache drop
738        let s = host.submit("fx:board", None, Params::new()).unwrap();
739        assert_eq!(s.message.as_deref(), Some("saved"));
740        assert!(host.submit("lz", None, Params::new()).is_err()); // lz has no submit op
741        assert!(matches!(
742            host.submit("fx:nope", None, Params::new()),
743            Err(PavoError::UnknownView { .. })
744        ));
745        // resolve again post-submit — the view cache was invalidated
746        // for fx:board (refetch succeeds against the same fixture).
747        let v = host.resolve_view("fx:board", Params::new()).unwrap();
748        assert_eq!(v.result.data["items"], serde_json::json!([1, 2]));
749    }

826        let tmp = tempfile::tempdir().unwrap();
827        let bin = tmp.path().join("bin");
828        write_driver(&bin, "tamias-driver-dv", DV);
829        write_driver(&bin, "tamias-driver-vz", VZ);

831
832        // Registry sees the plugin claim (renderer-only driver).
833        let reg = host.renderers();
834        assert_eq!(reg.len(), 1);
835        assert!(reg.handles("graph", "cli", Some("svg")));
836        assert!(!reg.handles("graph", "cli", Some("ascii")));
837        assert!(reg.handles("graph", "tui", None));
838
839        // render_by_kind routes to vz.
840        let node = serde_json::json!({"kind":"graph","nodes":[]});
841        let model = host
842            .render_by_kind("graph", &node, &Params::new(), "cli", Some("svg"))
843            .unwrap();
844        assert_eq!(
845            model,
846            RenderModel::Svg {
847                markup: "<svg>VZ</svg>".into()
848            }
849        );
850        // Unclaimed kind+surface → error.
851        assert!(host
852            .render_by_kind("table", &node, &Params::new(), "cli", Some("svg"))
853            .is_err());
854
855        // render_view_plugins finds the root graph node (path "").
856        let resolved = host.resolve_view("dv:g", Params::new()).unwrap();
857        let renders = host.render_view_plugins(&resolved, "cli", Some("svg"));
858        assert_eq!(renders.len(), 1);
859        assert!(matches!(renders[""], RenderModel::Svg { .. }));
860        // cli+ascii isn't claimed → empty map → built-in fallback.
861        assert!(host
862            .render_view_plugins(&resolved, "cli", Some("ascii"))
863            .is_empty());
864        // tui IS claimed but vz answers svg anyway (fixture ignores
865        // surface) — still routed.
866        assert_eq!(host.render_view_plugins(&resolved, "tui", None).len(), 1);
867    }
