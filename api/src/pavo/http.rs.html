1//! Minimal HTTP/1.1 primitives for `pavo serve` — a hand-rolled
2//! localhost server (std-only, no async runtime, no deps).
3//!
4//! Scope is deliberately narrow: GET/HEAD-style JSON APIs plus static
5//! file serving for the Nuxt build output. [`read_request`] parses one
6//! request off a [`TcpStream`] with hard caps on head and body size;
7//! [`Response`] serializes status line + headers + body with optional
8//! CORS. Keep-alive is supported — the caller loops until the client
9//! closes or sends `Connection: close`.
10
11use std::collections::BTreeMap;
12use std::io::{Read, Write};
13use std::net::TcpStream;
14
15/// Max bytes for the request head (request line + headers).
16pub const MAX_HEAD: usize = 64 * 1024;
17/// Max request body — the API is GET-only; this just bounds abuse.
18pub const MAX_BODY: usize = 1024 * 1024;
19
20/// A parsed HTTP request.
21#[derive(Debug)]
22pub struct Request {
23    /// Uppercased method (`GET`, `HEAD`, …).
24    pub method: String,
25    /// Raw request target (path + query, still percent-encoded).
26    pub target: String,
27    /// Headers with lowercased names, in wire order.
28    pub headers: Vec<(String, String)>,
29    /// Request body bytes (bounded by [`MAX_BODY`]). Currently unused
30    /// — the API is GET-only — but consumed so keep-alive framing
31    /// stays correct if a client sends one.
32    #[allow(dead_code)]
33    pub body: Vec<u8>,
34}
35
36impl Request {
37    /// First value of `name` (case-insensitive — names are stored
38    /// lowercased).
39    #[must_use]
40    pub fn header(&self, name: &str) -> Option<&str> {
41        self.headers
42            .iter()
43            .find(|(k, _)| k == name)
44            .map(|(_, v)| v.as_str())
45    }
46
47    /// Whether the client asked to close after this response.
48    #[must_use]
49    pub fn wants_close(&self) -> bool {
50        self.header("connection")
51            .is_some_and(|v| v.eq_ignore_ascii_case("close"))
52    }
53}

55/// Why [`read_request`] couldn't produce a request.
56#[derive(Debug)]
57pub enum ReadError {
58    /// Malformed/oversized input — respond with this status.
59    Http { status: u16, message: String },
60    /// Underlying io failure.
61    Io(std::io::Error),
62}
63
64impl From<std::io::Error> for ReadError {
65    fn from(e: std::io::Error) -> Self {
66        Self::Io(e)
67    }
68}

70/// Read one request from the stream. Returns `Ok(None)` on clean EOF
71/// (client closed an idle keep-alive connection).
72///
73/// # Errors
74/// [`ReadError::Http`] on malformed input (caller should send the
75/// status and close); [`ReadError::Io`] on transport failure.
76pub fn read_request(stream: &mut TcpStream) -> Result<Option<Request>, ReadError> {
77    // Read until the head terminator, bounded by MAX_HEAD.
78    let mut head = Vec::new();
79    let mut chunk = [0u8; 8192];
80    let head_end = loop {
81        if let Some(pos) = find_subslice(&head, b"\r\n\r\n") {
82            break pos + 4;
83        }
84        if let Some(pos) = find_subslice(&head, b"\n\n") {
85            break pos + 2;
86        }
87        if head.len() >= MAX_HEAD {
88            return Err(ReadError::Http {
89                status: 431,
90                message: "request headers too large".to_string(),
91            });
92        }
93        let n = stream.read(&mut chunk)?;
94        if n == 0 {
95            if head.is_empty() {
96                return Ok(None);
97            }
98            return Err(ReadError::Http {
99                status: 400,
100                message: "connection closed mid-request".to_string(),
101            });
102        }
103        head.extend_from_slice(&chunk[..n]);
104    };
105
106    // Split head into text lines (head is attacker-controlled but
107    // bounded; lossy UTF-8 keeps parsing total).
108    let head_text = String::from_utf8_lossy(&head[..head_end]);
109    let mut lines = head_text.lines();
110    let request_line = lines.next().unwrap_or("").trim_end_matches('\r');
111    let mut parts = request_line.split_whitespace();
112    let (Some(method), Some(target)) = (parts.next(), parts.next()) else {
113        return Err(ReadError::Http {
114            status: 400,
115            message: "malformed request line".to_string(),
116        });
117    };
118    // HTTP version token is ignored — we only speak 1.1 semantics.
119
120    let mut headers = Vec::new();
121    let mut content_length: Option<usize> = None;
122    for line in lines {
123        let line = line.trim_end_matches('\r');
124        if line.is_empty() {
125            continue;
126        }
127        let Some((name, value)) = line.split_once(':') else {
128            return Err(ReadError::Http {
129                status: 400,
130                message: format!("malformed header: {line:?}"),
131            });
132        };
133        let name = name.trim().to_ascii_lowercase();
134        let value = value.trim().to_string();
135        if name == "content-length" {
136            match value.parse::<usize>() {
137                Ok(n) => content_length = Some(n),
138                Err(_) => {
139                    return Err(ReadError::Http {
140                        status: 400,
141                        message: "bad Content-Length".to_string(),
142                    });
143                }
144            }
145        }
146        headers.push((name, value));
147    }
148
149    // Body: whatever already arrived past the head, then read_exact.
150    let mut body: Vec<u8> = head[head_end..].to_vec();
151    let declared = content_length.unwrap_or(0);
152    if declared > MAX_BODY {
153        return Err(ReadError::Http {
154            status: 413,
155            message: "request body too large".to_string(),
156        });
157    }
158    while body.len() < declared {
159        let mut buf = vec![0u8; declared - body.len()];
160        match stream.read(&mut buf) {
161            Ok(0) => {
162                return Err(ReadError::Http {
163                    status: 400,
164                    message: "connection closed mid-body".to_string(),
165                });
166            }
167            Ok(n) => body.extend_from_slice(&buf[..n]),
168            Err(e) if e.kind() == std::io::ErrorKind::Interrupted => continue,
169            Err(e) => return Err(ReadError::Io(e)),
170        }
171    }
172    body.truncate(declared);
173
174    Ok(Some(Request {
175        method: method.to_string(),
176        target: target.to_string(),
177        headers,
178        body,
179    }))
180}

182fn find_subslice(hay: &[u8], needle: &[u8]) -> Option<usize> {
183    hay.windows(needle.len()).position(|w| w == needle)
184}

186/// An HTTP response to write back.
187pub struct Response {
188    /// Status code.
189    pub status: u16,
190    /// Content-Type header value.
191    pub content_type: String,
192    /// Body bytes.
193    pub body: Vec<u8>,
194    /// Add `Access-Control-Allow-Origin: *` (API endpoints).
195    pub cors: bool,
196    /// Extra headers (e.g. `Allow`).
197    pub extra: Vec<(String, String)>,
198}
199
200impl Response {
201    /// JSON body response.
202    #[must_use]
203    pub fn json(status: u16, value: &serde_json::Value, cors: bool) -> Self {
204        Self {
205            status,
206            content_type: "application/json; charset=utf-8".to_string(),
207            body: serde_json::to_vec(value).unwrap_or_else(|_| b"{}".to_vec()),
208            cors,
209            extra: Vec::new(),
210        }
211    }
212
213    /// Raw body response.
214    #[must_use]
215    pub fn raw(status: u16, content_type: impl Into<String>, body: Vec<u8>, cors: bool) -> Self {
216        Self {
217            status,
218            content_type: content_type.into(),
219            body,
220            cors,
221            extra: Vec::new(),
222        }
223    }
224
225    /// Serialize + write the response. `close` forces
226    /// `Connection: close`.
227    ///
228    /// # Errors
229    /// Propagates io failures.
230    pub fn write(&self, stream: &mut TcpStream, close: bool) -> std::io::Result<()> {
231        let reason = reason_phrase(self.status);
232        let mut head = format!(
233            "HTTP/1.1 {} {}\r\ncontent-type: {}\r\ncontent-length: {}\r\nconnection: {}\r\n",
234            self.status,
235            reason,
236            self.content_type,
237            self.body.len(),
238            if close { "close" } else { "keep-alive" },
239        );
240        if self.cors {
241            head.push_str("access-control-allow-origin: *\r\n");
242        }
243        for (name, value) in &self.extra {
244            head.push_str(&format!("{name}: {value}\r\n"));
245        }
246        head.push_str("\r\n");
247        stream.write_all(head.as_bytes())?;
248        stream.write_all(&self.body)?;
249        stream.flush()
250    }
251}
252
253fn reason_phrase(status: u16) -> &'static str {
254    match status {
255        200 => "OK",
256        204 => "No Content",
257        301 => "Moved Permanently",
258        302 => "Found",
259        304 => "Not Modified",
260        400 => "Bad Request",
261        404 => "Not Found",
262        405 => "Method Not Allowed",
263        413 => "Payload Too Large",
264        431 => "Request Header Fields Too Large",
265        500 => "Internal Server Error",
266        _ => "Status",
267    }
268}

270/// Split a request target into (path, raw query string).
271#[must_use]
272pub fn split_target(target: &str) -> (&str, &str) {
273    match target.split_once('?') {
274        Some((p, q)) => (p, q),
275        None => (target, ""),
276    }
277}

279/// Parse a `a=1&b=2` query string into decoded pairs. Malformed pairs
280/// (no `=`) are kept with an empty value; decoding is lossy-safe.
281#[must_use]
282pub fn parse_query(query: &str) -> Vec<(String, String)> {
283    if query.is_empty() {
284        return Vec::new();
285    }
286    query
287        .split('&')
288        .filter(|p| !p.is_empty())
289        .map(|pair| match pair.split_once('=') {
290            Some((k, v)) => (percent_decode(k), percent_decode(v)),
291            None => (percent_decode(pair), String::new()),
292        })
293        .collect()
294}
295
296/// Percent-decode a URL component (`%XX` + `+` → space). Invalid
297/// sequences pass through literally.
298#[must_use]
299pub fn percent_decode(input: &str) -> String {
300    let bytes = input.as_bytes();
301    let mut out = Vec::with_capacity(bytes.len());
302    let mut i = 0;
303    while i < bytes.len() {
304        match bytes[i] {
305            b'+' => {
306                out.push(b' ');
307                i += 1;
308            }
309            b'%' if i + 2 < bytes.len() => {
310                // Decode hex from raw bytes — slicing the &str here
311                // would panic when the two bytes after '%' fall inside
312                // a multi-byte UTF-8 char (e.g. "%€x").
313                match (hex_val(bytes[i + 1]), hex_val(bytes[i + 2])) {
314                    (Some(hi), Some(lo)) => {
315                        out.push(hi * 16 + lo);
316                        i += 3;
317                    }
318                    _ => {
319                        out.push(b'%');
320                        i += 1;
321                    }
322                }
323            }
324            b => {
325                out.push(b);
326                i += 1;
327            }
328        }
329    }
330    // Path/query targets are UTF-8 in practice; lossy keeps us total.
331    String::from_utf8_lossy(&out).into_owned()
332}
333
334/// Single hex digit → nibble value, `None` for non-hex bytes.
335fn hex_val(b: u8) -> Option<u8> {
336    match b {
337        b'0'..=b'9' => Some(b - b'0'),
338        b'a'..=b'f' => Some(b - b'a' + 10),
339        b'A'..=b'F' => Some(b - b'A' + 10),
340        _ => None,
341    }
342}

344/// Look up a decoded query param (first occurrence wins).
345#[must_use]
346pub fn query_get<'a>(params: &'a [(String, String)], key: &str) -> Option<&'a str> {
347    params
348        .iter()
349        .find(|(k, _)| k == key)
350        .map(|(_, v)| v.as_str())
351}

353/// Parse one param value: JSON when possible (`42`, `true`, `{…}`),
354/// else a string — same rule as the CLI's `--param`.
355#[must_use]
356pub fn param_json(v: &str) -> serde_json::Value {
357    serde_json::from_str(v).unwrap_or_else(|_| serde_json::Value::String(v.to_string()))
358}
359
360/// Collect `param.<key>=<value>` pairs into a map.
361#[must_use]
362pub fn collect_params(pairs: &[(String, String)]) -> BTreeMap<String, serde_json::Value> {
363    let mut out = BTreeMap::new();
364    for (k, v) in pairs {
365        if let Some(key) = k.strip_prefix("param.") {
366            if !key.is_empty() {
367                out.insert(key.to_string(), param_json(v));
368            }
369        }
370    }
371    out
372}

431    /// A connected loopback pair: (client, server). The server end is
432    /// what `read_request`/`Response::write` operate on.
433    fn pair() -> (std::net::TcpStream, std::net::TcpStream) {
434        let listener = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
435        let addr = listener.local_addr().unwrap();
436        let client = std::net::TcpStream::connect(addr).unwrap();
437        let (server, _) = listener.accept().unwrap();
438        (client, server)
439    }

611    }

645                ],
646                0..4096,
647            ),
648        ) {
649            let (mut client, mut server) = pair();
650            let _ = client.write_all(&bytes);
651            let _ = client.shutdown(std::net::Shutdown::Write);
652            let _ = read_request(&mut server);
653        }

655        /// Well-formed-ish requests built from arbitrary parts,
656        /// sometimes truncated: still total.
657        #[test]
658        fn read_request_never_panics_on_template_requests(
659            method in prop_oneof![Just("GET".to_string()), Just("POST".to_string()), tricky_text()],
660            target in tricky_text(),
661            headers in prop::collection::vec((tricky_text(), tricky_text()), 0..8),
662            body in prop::collection::vec(any::<u8>(), 0..256),
663            truncate in any::<bool>(),
664        ) {
665            let mut req = format!("{method} {target} HTTP/1.1\r\n");
666            for (k, v) in &headers {
667                req.push_str(&format!("{k}: {v}\r\n"));
668            }
669            req.push_str(&format!("Content-Length: {}\r\n\r\n", body.len()));
670            let mut bytes = req.into_bytes();
671            bytes.extend_from_slice(&body);
672            if truncate {
673                let cut = bytes.len().saturating_sub(1);
674                bytes.truncate(cut);
675            }
676            let (mut client, mut server) = pair();
677            let _ = client.write_all(&bytes);
678            let _ = client.shutdown(std::net::Shutdown::Write);
679            let res = read_request(&mut server);
680            // Invariant: a parsed request always has non-empty
681            // method/target tokens and body length == declared CL.
682            if let Ok(Some(r)) = res {
683                prop_assert!(!r.method.is_empty());
684                prop_assert!(!r.target.is_empty());
685            }
686        }

688        /// Response serialization is total on arbitrary status /
689        /// header values and bodies — and never panics writing.
690        #[test]
691        fn response_write_never_panics(
692            status in any::<u16>(),
693            body in prop::collection::vec(any::<u8>(), 0..512),
694            extra in prop::collection::vec((tricky_text(), tricky_text()), 0..4),
695            cors in any::<bool>(),
696            close in any::<bool>(),
697        ) {
698            let (client, mut server) = pair();

702            }
703            resp.write(&mut server, close).unwrap();
704            let _ = client.shutdown(std::net::Shutdown::Both);
705        }
