2//! pavo — the CLI surface of the viz host.
3//!
4//! Verbs:
5//!
6//! - `drivers` — list discovered `tamias-driver-*` executables that
7//!   speak pavo (manifest version, view count, contributes summary).
8//! - `views [--driver <name>]` — every view as a `<driver>:<view>`
9//!   ref with title/route.
10//! - `chrome` — the merged contribution slots (navbar/dashboard/
11//!   statusbar/routes + forward-compat extras).
12//! - `view <driver>:<view> [--param k=v …] [--format …]` — resolve a
13//!   deep link and render it: `text` (default) summarizes the
14//!   ViewNode tree + bound data, `json` dumps the raw
15//!   [`ViewResult`], `dot`/`mermaid` emit the view's graph nodes.
16//! - `query <driver> <query> [--param k=v …]` — raw `query` op
17//!   passthrough.
18//! - `submit <driver>:<view> [--form <id>] --param k=v …` — the
19//!   `submit` wire op: send a form's field values to its driver and
20//!   print the `SubmitResult` (message/errors/data; `--json` dumps
21//!   the full payload including any updated view).
22//! - `serve [--port N]` — the localhost HTTP bridge for the web host:
23//!   `/api/*` JSON endpoints over the same [`Host`] plus static
24//!   serving of the Nuxt build (`web/.output/public`). First stdout
25//!   line is `listening http://127.0.0.1:<port>`.
26//! - `open [<driver>:<view>]` — hand off to `pavo-tui` (the ratatui
27//!   surface) with the deep link, or the bare dashboard when no
28//!   target is given. On a non-TTY it falls back to `pavo view`
29//!   text output for the given target.
30//! - `manifest` / `schema` — the CLI's own capability manifest and
31//!   verb schema (sibling convention: tamias/kuda).
32//!
33//! `--json` is a per-subcommand flag (sibling convention) on the
34//! listing/query verbs; `view --format json` is its machine-readable
35//! mode. Errors go to stderr; exit codes are CI-meaningful: `0` ok,
36//! `1` view/driver error, `2` usage (bad `--param`, malformed view
37//! ref, bad format — clap reports the last itself).
38//!
39//! Discovery is environment-driven like pavo-core's
40//! [`Host::discover`]: drivers are found on `PATH` and under
41//! `$TAMIAS_HOME/drivers` — set `PATH`/`TAMIAS_HOME` to point the CLI
42//! at fixture drivers (integration tests do exactly this).

153/// The built-in `view --format` values — all but `json` can be
154/// claimed by renderer plugins (`renderers[].surfaces.cli.formats`,
155/// empty = wildcard); a plugin claim substitutes its RenderModel for
156/// the claimed nodes and the built-in covers the rest. `json` is the

158const BUILTIN_FORMATS: [&str; 4] = ["text", "json", "dot", "mermaid"];

160/// A CLI failure: exit code + human message (stderr).
161struct Fail {
162    code: u8,
163    msg: String,
164}
165
166impl Fail {
167    /// View/driver/runtime error → exit 1.
168    fn err(msg: impl Into<String>) -> Self {
169        Self {
170            code: 1,
171            msg: msg.into(),
172        }
173    }
174
175    /// Usage error → exit 2.
176    fn usage(msg: impl Into<String>) -> Self {
177        Self {
178            code: 2,
179            msg: msg.into(),
180        }
181    }
182}
183
184impl From<PavoError> for Fail {
185    fn from(e: PavoError) -> Self {
186        // A malformed `<driver>:<view>` ref is a usage error; driver
187        // and resolution failures are runtime errors.
188        let code = match &e {
189            PavoError::BadViewRef(_) => 2,
190            _ => 1,
191        };
192        Self {
193            code,
194            msg: e.to_string(),
195        }
196    }
197}

199fn main() -> ExitCode {
200    let cli = Cli::parse();
201    match run(cli) {
202        Ok(code) => ExitCode::from(code),
203        Err(f) => {
204            eprintln!("pavo: {}", f.msg);
205            ExitCode::from(f.code)
206        }
207    }
208}

210/// `Ok` carries the process exit code (only `open` can produce a
211/// non-zero one — it propagates `pavo-tui`'s status).
212fn run(cli: Cli) -> Result<u8, Fail> {
213    match cli.command {
214        None => {
215            println!("pavo {}", env!("CARGO_PKG_VERSION"));
216            Ok(0)
217        }
218        Some(Command::Drivers { json }) => cmd_drivers(json).map(|_| 0),
219        Some(Command::Views { driver, json }) => cmd_views(driver.as_deref(), json).map(|_| 0),
220        Some(Command::Chrome { json }) => cmd_chrome(json).map(|_| 0),
221        Some(Command::View {
222            target,
223            params,
224            format,
225        }) => cmd_view(&target, &params, &format).map(|_| 0),
226        Some(Command::Query {
227            driver,
228            query,
229            params,
230            json,
231        }) => cmd_query(&driver, &query, &params, json).map(|_| 0),
232        Some(Command::Submit {
233            target,
234            form,
235            params,
236            json,
237        }) => cmd_submit(&target, form.as_deref(), &params, json).map(|_| 0),
238        Some(Command::Serve { port }) => serve::run(port)
239            .map(|_| 0)
240            .map_err(|e| Fail::err(format!("serve: {e}"))),
241        Some(Command::Open { target }) => cmd_open(target.as_deref()),
242        Some(Command::Manifest) => {
243            println!(
244                "{}",
245                json!({
246                    "name": "pavo",
247                    "version": env!("CARGO_PKG_VERSION"),
248                    "verbs": [
249                        "drivers", "views", "chrome", "view", "query",
250                        "submit", "serve", "open", "manifest", "schema"
251                    ],
252                })
253            );
254            Ok(0)
255        }
256        Some(Command::Schema) => {
257            println!(
258                "{}",
259                serde_json::to_string_pretty(&schema_doc())
260                    .map_err(|e| Fail::err(format!("schema serialization: {e}")))?
261            );
262            Ok(0)
263        }
264    }
265}

322/// `pavo open` — hand the deep link to `pavo-tui` on a TTY, else
323/// degrade to `view` text output. Returns `pavo-tui`'s exit status.
324fn cmd_open(target: Option<&str>) -> Result<u8, Fail> {
325    if std::io::stdout().is_terminal() {
326        return spawn_tui(target);
327    }
328    let Some(target) = target else {
329        return Err(Fail::usage(
330            "pavo open needs a terminal for the TUI — pass <driver>:<view> \
331             (or use `pavo view`) for text output",
332        ));
333    };
334    eprintln!("pavo: not a TTY — rendering text output instead of the TUI");
335    cmd_view(target, &[], "text").map(|_| 0)
336}

338/// Candidate `pavo-tui` binaries: the sibling of this executable
339/// (covers `cargo run`/target-dir installs) then `PATH`.
340fn tui_candidates() -> Vec<std::path::PathBuf> {
341    let mut out = Vec::new();
342    if let Ok(exe) = std::env::current_exe() {
343        if let Some(dir) = exe.parent() {
344            out.push(dir.join("pavo-tui"));
345        }
346    }
347    out.push(std::path::PathBuf::from("pavo-tui"));
348    out
349}
350
351/// Spawn `pavo-tui` (inheriting stdio) with the optional deep link
352/// and propagate its exit status.
353fn spawn_tui(target: Option<&str>) -> Result<u8, Fail> {
354    let mut last_err = String::new();
355    for cand in tui_candidates() {
356        let mut cmd = std::process::Command::new(&cand);
357        if let Some(t) = target {
358            cmd.arg(t);
359        }
360        match cmd.status() {
361            Ok(status) => {
362                // Signals yield no code; clamp exotic values to 1.
363                let code = status
364                    .code()
365                    .and_then(|c| u8::try_from(c).ok())
366                    .unwrap_or(1);
367                return Ok(code);
368            }
369            Err(e) => last_err = format!("{}: {e}", cand.display()),
370        }
371    }
372    Err(Fail::err(format!(
373        "pavo-tui not found on PATH or next to pavo ({last_err}) — \
374         install it (`cargo xtask install` or `cargo install --path crates/pavo-tui`) \
375         or run `cargo run -p pavo-tui`"
376    )))
377}

384/// [`cmd_drivers`] over an explicit host — split for tests.
385fn cmd_drivers_host(host: &mut Host, json: bool) -> Result<(), Fail> {
386    let kinds: Vec<String> = host.drivers().iter().map(|d| d.kind.clone()).collect();
387    let mut rows = Vec::new();
388    for kind in kinds {
389        match host.manifest(&kind) {
390            Ok(m) => rows.push(json!({
391                "driver": kind,
392                "title": m.title,
393                "version": m.version,
394                "views": m.views.len(),
395                "renderers": m.renderers.len(),
396                "contributes": contrib_counts(&m.contributes),
397            })),
398            Err(e) => rows.push(json!({
399                "driver": kind,
400                "manifest_error": e.to_string(),
401            })),
402        }
403    }
404    if json {
405        println!(
406            "{}",
407            serde_json::to_string(&rows).map_err(|e| Fail::err(e.to_string()))?
408        );
409        return Ok(());
410    }
411    if rows.is_empty() {
412        println!("no pavo-capable drivers found (PATH + $TAMIAS_HOME/drivers)");
413        return Ok(());
414    }
415    for row in &rows {
416        if let Some(err) = row.get("manifest_error").and_then(Value::as_str) {
417            println!(
418                "{}  —  manifest error: {err}",
419                row["driver"].as_str().unwrap_or("?")
420            );
421            continue;
422        }
423        let counts = &row["contributes"];
424        let contrib = ["navbar", "dashboard", "statusbar", "routes"]
425            .iter()
426            .filter(|slot| counts.get(**slot).and_then(Value::as_u64).unwrap_or(0) > 0)
427            .map(|slot| format!("{slot}:{}", counts[*slot].as_u64().unwrap_or(0)))
428            .collect::<Vec<_>>()
429            .join(" ");
430        println!(
431            "{}  {}  v{}  views:{}  renderers:{}  {}",
432            row["driver"].as_str().unwrap_or("?"),
433            row["title"].as_str().unwrap_or("?"),
434            row["version"].as_str().unwrap_or("?"),
435            row["views"].as_u64().unwrap_or(0),
436            row["renderers"].as_u64().unwrap_or(0),
437            if contrib.is_empty() {
438                "contributes: —".to_string()
439            } else {
440                format!("contributes: {contrib}")
441            }
442        );
443    }
444    Ok(())
445}

447/// Per-slot contribution counts for the `drivers` summary (shared
448/// with `serve`'s `/api/drivers`).
449pub(crate) fn contrib_counts(c: &pavo_protocol::Contributes) -> Value {
450    let mut map = Map::new();
451    for (name, entries) in [
452        ("navbar", &c.navbar),
453        ("dashboard", &c.dashboard),
454        ("statusbar", &c.statusbar),
455        ("routes", &c.routes),
456    ] {
457        map.insert(name.to_string(), json!(entries.len()));
458    }
459    let extra: Map<String, Value> = c
460        .extra
461        .iter()
462        .map(|(k, v)| (k.clone(), json!(v.len())))
463        .collect();
464    map.insert("extra".to_string(), Value::Object(extra));
465    Value::Object(map)
466}

519/// [`cmd_chrome`] over an explicit host — split for tests.
520fn cmd_chrome_host(host: &mut Host, json: bool) -> Result<(), Fail> {
521    let chrome = host.chrome();
522    if json {
523        let slot = |entries: &[pavo_core::SlotEntry]| -> Vec<Value> {
524            entries
525                .iter()
526                .map(|e| {
527                    json!({
528                        "driver": e.driver,
529                        "driver_title": e.driver_title,
530                        "id": e.contribution.id,
531                        "title": e.contribution.title,
532                        "route": e.contribution.route,
533                        "priority": e.contribution.priority,
534                        "icon": e.contribution.icon,
535                    })
536                })
537                .collect()
538        };
539        let extra: Map<String, Value> = chrome
540            .extra
541            .iter()
542            .map(|(k, v)| (k.clone(), Value::Array(slot(v))))
543            .collect();
544        let doc = json!({
545            "navbar": slot(&chrome.navbar),
546            "dashboard": slot(&chrome.dashboard),
547            "statusbar": slot(&chrome.statusbar),
548            "routes": slot(&chrome.routes),
549            "extra": Value::Object(extra),
550        });
551        println!(
552            "{}",
553            serde_json::to_string_pretty(&doc).map_err(|e| Fail::err(e.to_string()))?
554        );
555        return Ok(());
556    }
557    let mut any = false;
558    for (name, entries) in [
559        ("navbar", &chrome.navbar),
560        ("dashboard", &chrome.dashboard),
561        ("statusbar", &chrome.statusbar),
562        ("routes", &chrome.routes),
563    ] {
564        println!("{name}:");
565        if entries.is_empty() {
566            println!("  (none)");
567            continue;
568        }
569        any = true;
570        for e in entries {
571            let c = &e.contribution;
572            println!("  {}  {}  → {}  [{}]", c.id, c.title, c.route, e.driver);
573        }
574    }
575    for (slot_name, entries) in &chrome.extra {
576        println!("{slot_name}:");
577        any = true;
578        for e in entries {
579            let c = &e.contribution;
580            println!("  {}  {}  → {}  [{}]", c.id, c.title, c.route, e.driver);
581        }
582    }
583    if !any {
584        println!("no chrome contributions");
585    }
586    Ok(())
587}

603/// The `view` verb's format dispatch — returns the exact bytes to
604/// print (split from [`cmd_view_host`] so tests can assert on
605/// output).
606///
607/// Precedence: renderer plugins are consulted for **every** format
608/// before the built-in interpretation runs — `text`/`dot`/`mermaid`
609/// substitute claimed nodes' RenderModels per tree path and keep the
610/// built-in rendering for unclaimed nodes. `json` is exempt (the raw
611/// `ViewResult` dump is host-owned). Any *other* format string is a
612/// plugin-only format: every claimed node's RenderModel is emitted;
613/// no claims → usage error naming the built-ins.
614fn view_output(
615    host: &Host,
616    resolved: &ResolvedView,
617    target: &str,
618    format: &str,
619) -> Result<String, Fail> {
620    match format {
621        "json" => serde_json::to_string_pretty(&resolved.result)
622            .map(|s| format!("{s}\n"))
623            .map_err(|e| Fail::err(e.to_string())),
624        "text" => {
625            // Renderer plugins that claim `text` (or wildcard cli
626            // formats) substitute their RenderModel per node; the
627            // rest render built-in.
628            let renders = host.render_view_plugins(resolved, "cli", Some("text"));
629            Ok(render::render_text_plugins(
630                &resolved.result.view.root,
631                &resolved.result.data,
632                &renders,
633            ))
634        }
635        "dot" => {
636            // Plugins claiming `dot` (or wildcard cli formats) take
637            // precedence per node; unclaimed graphs still emit
638            // digraph blocks.
639            let renders = host.render_view_plugins(resolved, "cli", Some("dot"));
640            render::render_dot_plugins(&resolved.result.view.root, &resolved.result.data, &renders)
641                .map_err(|e| Fail::err(format!("{target}: {e}")))
642        }
643        "mermaid" => {
644            let renders = host.render_view_plugins(resolved, "cli", Some("mermaid"));
645            render::render_mermaid_plugins(
646                &resolved.result.view.root,
647                &resolved.result.data,
648                &renders,
649            )
650            .map_err(|e| Fail::err(format!("{target}: {e}")))
651        }
652        other => {
653            // Plugin format: route every claimed node through its
654            // driver's `render` op and concatenate the models.
655            let renders = host.render_view_plugins(resolved, "cli", Some(other));
656            if renders.is_empty() {
657                return Err(Fail::err(format!(
658                    "{target}: no renderer plugin handles --format '{other}' \
659                     (built-ins: {})",
660                    BUILTIN_FORMATS.join(", ")
661                )));
662            }
663            // Deterministic order: sort by path ("" root first, then
664            // tree order).
665            let mut keyed: Vec<(&String, &pavo_protocol::RenderModel)> = renders.iter().collect();
666            keyed.sort_by(|a, b| a.0.cmp(b.0));
667            let mut out = String::new();
668            for (_, model) in keyed {
669                out.push_str(&render::render_model_cli(model));
670            }
671            Ok(out)
672        }
673    }
674}

711/// [`cmd_submit`] over an explicit host — split for tests.
712fn cmd_submit_host(
713    host: &mut Host,
714    target: &str,
715    form: Option<&str>,
716    values: Params,
717    json: bool,
718) -> Result<(), Fail> {
719    let result = host.submit(target, form, values)?;
720    if json {
721        println!(
722            "{}",
723            serde_json::to_string_pretty(&result).map_err(|e| Fail::err(e.to_string()))?
724        );
725    } else {
726        // Text mode: message first, then data.
727        if let Some(message) = &result.message {
728            println!("{message}");
729        }
730        match &result.data {
731            Some(data) => println!(
732                "{}",
733                serde_json::to_string_pretty(data).map_err(|e| Fail::err(e.to_string()))?
734            ),
735            None if result.message.is_none() && result.errors.is_empty() => {
736                println!("(empty result)");
737            }
738            None => {}
739        }
740    }
741    // Per-field validation errors mean the driver *rejected* the
742    // submit (the wire still answered ok:true). Exit 1 in both
743    // modes so CI can gate on it; `--json` callers also get the
744    // field details in the payload on stdout.
745    if !result.errors.is_empty() {
746        let detail = result
747            .errors
748            .iter()
749            .map(|(k, v)| format!("{k}: {}", v.as_str().unwrap_or("invalid")))
750            .collect::<Vec<_>>()
751            .join("; ");
752        return Err(Fail {
753            code: 1,
754            msg: format!("submit rejected — {detail}"),
755        });
756    }
757    Ok(())
758}

760/// Parse repeated `--param k=v` args into a [`Params`] map. Values
761/// parse as JSON when possible (`k=42`, `k=true`, `k={…}`), else
762/// become strings. Missing `=` or an empty key is a usage error.
763fn parse_params(args: &[String]) -> Result<Params, Fail> {
764    let mut params = Params::new();
765    for arg in args {
766        let Some((key, raw)) = arg.split_once('=') else {
767            return Err(Fail::usage(format!(
768                "--param expects key=value, got '{arg}'"
769            )));
770        };
771        if key.is_empty() {
772            return Err(Fail::usage(format!("--param has an empty key: '{arg}'")));
773        }
774        let value = serde_json::from_str(raw).unwrap_or_else(|_| Value::String(raw.to_string()));
775        params.insert(key.to_string(), value);
776    }
777    Ok(params)
778}
