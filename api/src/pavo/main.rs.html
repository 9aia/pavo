2//! pavo — the CLI surface of the viz host.
3//!
4//! Verbs:
5//!
6//! - `drivers` — list discovered `tamias-driver-*` executables that
7//!   speak pavo (manifest version, view count, contributes summary).
8//! - `views [--driver <name>]` — every view as a `<driver>:<view>`
9//!   ref with title/route.
10//! - `chrome` — the merged contribution slots (navbar/dashboard/
11//!   statusbar/routes/sidebar/headerActions/notifications/jobs +
12//!   forward-compat extras).
13//! - `view <driver>:<view> [--param k=v …] [--format …]` — resolve a
14//!   deep link and render it: `text` (default) summarizes the
15//!   ViewNode tree + bound data, `json` dumps the raw
16//!   [`ViewResult`], `dot`/`mermaid` emit the view's graph nodes.
17//! - `query <driver> <query> [--param k=v …]` — raw `query` op
18//!   passthrough.
19//! - `submit <driver>:<view> [--form <id>] --param k=v …` — the
20//!   `submit` wire op: send a form's field values to its driver and
21//!   print the `SubmitResult` (message/errors/data; `--json` dumps
22//!   the full payload including any updated view).
23//! - `jobs` / `notifications` — the aggregated platform feeds: every
24//!   provider driver's `jobs`/`notifications` wire op, tagged with
25//!   its driver (`--json` dumps the flattened rows).
26//! - `serve [--port N]` — the localhost HTTP bridge for the web host:
27//!   `/api/*` JSON endpoints over the same [`Host`] plus static
28//!   serving of the Nuxt build (`web/.output/public`). First stdout
29//!   line is `listening http://127.0.0.1:<port>`.
30//! - `open [<driver>:<view>]` — hand off to `pavo-tui` (the ratatui
31//!   surface) with the deep link, or the bare dashboard when no
32//!   target is given. On a non-TTY it falls back to `pavo view`
33//!   text output for the given target.
34//! - `manifest` / `schema` — the CLI's own capability manifest and
35//!   verb schema (sibling convention: tamias/kuda).
36//!
37//! `--json` is a per-subcommand flag (sibling convention) on the
38//! listing/query verbs; `view --format json` is its machine-readable
39//! mode. Errors go to stderr; exit codes are CI-meaningful: `0` ok,
40//! `1` view/driver error, `2` usage (bad `--param`, malformed view
41//! ref, bad format — clap reports the last itself).
42//!
43//! Discovery is environment-driven like pavo-core's
44//! [`Host::discover`]: drivers are found on `PATH` and under
45//! `$TAMIAS_HOME/drivers` — set `PATH`/`TAMIAS_HOME` to point the CLI
46//! at fixture drivers (integration tests do exactly this).

180/// A CLI failure: exit code + human message (stderr).
181struct Fail {
182    code: u8,
183    msg: String,
184}
185
186impl Fail {
187    /// View/driver/runtime error → exit 1.
188    fn err(msg: impl Into<String>) -> Self {
189        Self {
190            code: 1,
191            msg: msg.into(),
192        }
193    }
194
195    /// Usage error → exit 2.
196    fn usage(msg: impl Into<String>) -> Self {
197        Self {
198            code: 2,
199            msg: msg.into(),
200        }
201    }
202}
203
204impl From<PavoError> for Fail {
205    fn from(e: PavoError) -> Self {
206        // A malformed `<driver>:<view>` ref is a usage error; driver
207        // and resolution failures are runtime errors.
208        let code = match &e {
209            PavoError::BadViewRef(_) => 2,
210            _ => 1,
211        };
212        Self {
213            code,
214            msg: e.to_string(),
215        }
216    }
217}

219fn main() -> ExitCode {
220    let cli = Cli::parse();
221    match run(cli) {
222        Ok(code) => ExitCode::from(code),
223        Err(f) => {
224            eprintln!("pavo: {}", f.msg);
225            ExitCode::from(f.code)
226        }
227    }

230/// `Ok` carries the process exit code (only `open` can produce a
231/// non-zero one — it propagates `pavo-tui`'s status).
232fn run(cli: Cli) -> Result<u8, Fail> {
233    match cli.command {
234        None => {
235            println!("pavo {}", env!("CARGO_PKG_VERSION"));
236            Ok(0)
237        }
238        Some(Command::Drivers { json }) => cmd_drivers(json).map(|_| 0),
239        Some(Command::Views { driver, json }) => cmd_views(driver.as_deref(), json).map(|_| 0),
240        Some(Command::Chrome { json }) => cmd_chrome(json).map(|_| 0),
241        Some(Command::View {
242            target,
243            params,
244            format,
245        }) => cmd_view(&target, &params, &format).map(|_| 0),
246        Some(Command::Query {
247            driver,
248            query,
249            params,
250            json,
251        }) => cmd_query(&driver, &query, &params, json).map(|_| 0),
252        Some(Command::Submit {
253            target,
254            form,
255            params,
256            json,
257        }) => cmd_submit(&target, form.as_deref(), &params, json).map(|_| 0),
258        Some(Command::Jobs { json }) => cmd_jobs(json).map(|_| 0),
259        Some(Command::Notifications { json }) => cmd_notifications(json).map(|_| 0),
260        Some(Command::Serve { port }) => serve::run(port)
261            .map(|_| 0)
262            .map_err(|e| Fail::err(format!("serve: {e}"))),
263        Some(Command::Open { target }) => cmd_open(target.as_deref()),
264        Some(Command::Manifest) => {
265            println!(
266                "{}",
267                json!({
268                    "name": "pavo",
269                    "version": env!("CARGO_PKG_VERSION"),
270                    "verbs": [
271                        "drivers", "views", "chrome", "view", "query",
272                        "submit", "jobs", "notifications", "serve",
273                        "open", "manifest", "schema"
274                    ],
275                })
276            );
277            Ok(0)
278        }
279        Some(Command::Schema) => {
280            println!(
281                "{}",
282                serde_json::to_string_pretty(&schema_doc())
283                    .map_err(|e| Fail::err(format!("schema serialization: {e}")))?
284            );
285            Ok(0)
286        }
287    }
288}

357/// `pavo open` — hand the deep link to `pavo-tui` on a TTY, else
358/// degrade to `view` text output. Returns `pavo-tui`'s exit status.
359fn cmd_open(target: Option<&str>) -> Result<u8, Fail> {
360    if std::io::stdout().is_terminal() {
361        return spawn_tui(target);
362    }
363    let Some(target) = target else {
364        return Err(Fail::usage(
365            "pavo open needs a terminal for the TUI — pass <driver>:<view> \
366             (or use `pavo view`) for text output",
367        ));
368    };
369    eprintln!("pavo: not a TTY — rendering text output instead of the TUI");
370    cmd_view(target, &[], "text").map(|_| 0)
371}

373/// Candidate `pavo-tui` binaries: the sibling of this executable
374/// (covers `cargo run`/target-dir installs) then `PATH`.
375fn tui_candidates() -> Vec<std::path::PathBuf> {
376    let mut out = Vec::new();
377    if let Ok(exe) = std::env::current_exe() {
378        if let Some(dir) = exe.parent() {
379            out.push(dir.join("pavo-tui"));
380        }
381    }
382    out.push(std::path::PathBuf::from("pavo-tui"));
383    out
384}
385
386/// Spawn `pavo-tui` (inheriting stdio) with the optional deep link
387/// and propagate its exit status.
388fn spawn_tui(target: Option<&str>) -> Result<u8, Fail> {
389    let mut last_err = String::new();
390    for cand in tui_candidates() {
391        let mut cmd = std::process::Command::new(&cand);
392        if let Some(t) = target {
393            cmd.arg(t);
394        }
395        match cmd.status() {
396            Ok(status) => {
397                // Signals yield no code; clamp exotic values to 1.
398                let code = status
399                    .code()
400                    .and_then(|c| u8::try_from(c).ok())
401                    .unwrap_or(1);
402                return Ok(code);
403            }
404            Err(e) => last_err = format!("{}: {e}", cand.display()),
405        }
406    }
407    Err(Fail::err(format!(
408        "pavo-tui not found on PATH or next to pavo ({last_err}) — \
409         install it (`cargo xtask install` or `cargo install --path crates/pavo-tui`) \
410         or run `cargo run -p pavo-tui`"
411    )))
412}

419/// [`cmd_drivers`] over an explicit host — split for tests.
420fn cmd_drivers_host(host: &mut Host, json: bool) -> Result<(), Fail> {
421    let kinds: Vec<String> = host.drivers().iter().map(|d| d.kind.clone()).collect();
422    let mut rows = Vec::new();
423    for kind in kinds {
424        match host.manifest(&kind) {
425            Ok(m) => rows.push(json!({
426                "driver": kind,
427                "title": m.title,
428                "version": m.version,
429                "views": m.views.len(),
430                "renderers": m.renderers.len(),
431                "contributes": contrib_counts(&m.contributes),
432            })),
433            Err(e) => rows.push(json!({
434                "driver": kind,
435                "manifest_error": e.to_string(),
436            })),
437        }
438    }
439    if json {
440        println!(
441            "{}",
442            serde_json::to_string(&rows).map_err(|e| Fail::err(e.to_string()))?
443        );
444        return Ok(());
445    }
446    if rows.is_empty() {
447        println!("no pavo-capable drivers found (PATH + $TAMIAS_HOME/drivers)");
448        return Ok(());
449    }
450    for row in &rows {
451        if let Some(err) = row.get("manifest_error").and_then(Value::as_str) {
452            println!(
453                "{}  —  manifest error: {err}",
454                row["driver"].as_str().unwrap_or("?")
455            );
456            continue;
457        }
458        let counts = &row["contributes"];
459        let contrib = [
460            "navbar",
461            "dashboard",
462            "statusbar",
463            "routes",
464            "sidebar",
465            "headerActions",
466            "notifications",
467            "jobs",
468        ]
469        .iter()
470        .filter(|slot| counts.get(**slot).and_then(Value::as_u64).unwrap_or(0) > 0)
471        .map(|slot| format!("{slot}:{}", counts[*slot].as_u64().unwrap_or(0)))
472        .collect::<Vec<_>>()
473        .join(" ");
474        println!(
475            "{}  {}  v{}  views:{}  renderers:{}  {}",
476            row["driver"].as_str().unwrap_or("?"),
477            row["title"].as_str().unwrap_or("?"),
478            row["version"].as_str().unwrap_or("?"),
479            row["views"].as_u64().unwrap_or(0),
480            row["renderers"].as_u64().unwrap_or(0),
481            if contrib.is_empty() {
482                "contributes: —".to_string()
483            } else {
484                format!("contributes: {contrib}")
485            }
486        );
487    }
488    Ok(())
489}

491/// Serialize host feed entries (`jobs`/`notifications`) for `--json`
492/// output and the serve bridge's `/api/jobs` + `/api/notifications`:
493/// each item's fields flattened into one object with `driver` /
494/// `driver_title` attribution — the host fields win over any
495/// same-named item fields (including `extra` catch-all keys).
496pub(crate) fn feed_rows<T: serde::Serialize>(
497    entries: &[pavo_core::FeedEntry<T>],
498) -> Result<Vec<Value>, serde_json::Error> {
499    entries
500        .iter()
501        .map(|e| {
502            let mut v = serde_json::to_value(&e.item)?;
503            if let Value::Object(ref mut map) = v {
504                map.insert("driver".to_string(), json!(e.driver));
505                map.insert("driver_title".to_string(), json!(e.driver_title));
506            }
507            Ok(v)
508        })
509        .collect()
510}

512/// Per-slot contribution counts for the `drivers` summary (shared
513/// with `serve`'s `/api/drivers`). Slot keys use the wire spellings —
514/// `headerActions` is camelCase like the manifest field.
515pub(crate) fn contrib_counts(c: &pavo_protocol::Contributes) -> Value {
516    let mut map = Map::new();
517    for (name, entries) in [
518        ("navbar", &c.navbar),
519        ("dashboard", &c.dashboard),
520        ("statusbar", &c.statusbar),
521        ("routes", &c.routes),
522        ("sidebar", &c.sidebar),
523        ("headerActions", &c.header_actions),
524        ("notifications", &c.notifications),
525        ("jobs", &c.jobs),
526    ] {
527        map.insert(name.to_string(), json!(entries.len()));
528    }
529    let extra: Map<String, Value> = c
530        .extra
531        .iter()
532        .map(|(k, v)| (k.clone(), json!(v.len())))
533        .collect();
534    map.insert("extra".to_string(), Value::Object(extra));
535    Value::Object(map)
536}

545    let mut views = host.list_views();
546    if let Some(d) = driver {
547        views.retain(|v| v.driver == d);
548    }

550        let rows: Vec<Value> = views
551            .iter()
552            .map(|v| {
553                json!({
554                    "driver": v.driver,
555                    "driver_title": v.driver_title,
556                    "id": v.id,
557                    "namespaced": v.namespaced,
558                    "title": v.title,
559                    "route": v.route,
560                    "description": v.description,
561                    "queries": v.queries,
562                })
563            })
564            .collect();
565        println!(
566            "{}",
567            serde_json::to_string(&rows).map_err(|e| Fail::err(e.to_string()))?
568        );
569        return Ok(());
570    }
571    if views.is_empty() {
572        println!("no views found");
573        return Ok(());
574    }
575    for v in &views {
576        println!("{:<24} {:<28} {}", v.namespaced, v.title, v.route);
577        if let Some(desc) = &v.description {
578            println!("{:<24}   {}", "", desc);
579        }
580    }
581    Ok(())
582}

641        ("jobs", &chrome.jobs),

643        println!("{name}:");
644        if entries.is_empty() {
645            println!("  (none)");
646            continue;
647        }
648        any = true;
649        for e in entries {
650            let c = &e.contribution;
651            println!("  {}  {}  → {}  [{}]", c.id, c.title, c.route, e.driver);
652        }
653    }
654    for (slot_name, entries) in &chrome.extra {
655        println!("{slot_name}:");
656        any = true;
657        for e in entries {
658            let c = &e.contribution;
659            println!("  {}  {}  → {}  [{}]", c.id, c.title, c.route, e.driver);
660        }
661    }

664    }

682/// The `view` verb's format dispatch — returns the exact bytes to
683/// print (split from [`cmd_view_host`] so tests can assert on
684/// output).
685///
686/// Precedence: renderer plugins are consulted for **every** format
687/// before the built-in interpretation runs — `text`/`dot`/`mermaid`
688/// substitute claimed nodes' RenderModels per tree path and keep the
689/// built-in rendering for unclaimed nodes. `json` is exempt (the raw
690/// `ViewResult` dump is host-owned). Any *other* format string is a
691/// plugin-only format: every claimed node's RenderModel is emitted;
692/// no claims → usage error naming the built-ins.
693fn view_output(
694    host: &Host,
695    resolved: &ResolvedView,
696    target: &str,
697    format: &str,
698) -> Result<String, Fail> {
699    match format {
700        "json" => serde_json::to_string_pretty(&resolved.result)
701            .map(|s| format!("{s}\n"))
702            .map_err(|e| Fail::err(e.to_string())),
703        "text" => {
704            // Renderer plugins that claim `text` (or wildcard cli
705            // formats) substitute their RenderModel per node; the
706            // rest render built-in.
707            let renders = host.render_view_plugins(resolved, "cli", Some("text"));
708            Ok(render::render_text_plugins(
709                &resolved.result.view.root,
710                &resolved.result.data,
711                &renders,
712            ))
713        }
714        "dot" => {
715            // Plugins claiming `dot` (or wildcard cli formats) take
716            // precedence per node; unclaimed graphs still emit
717            // digraph blocks.
718            let renders = host.render_view_plugins(resolved, "cli", Some("dot"));
719            render::render_dot_plugins(&resolved.result.view.root, &resolved.result.data, &renders)
720                .map_err(|e| Fail::err(format!("{target}: {e}")))
721        }
722        "mermaid" => {
723            let renders = host.render_view_plugins(resolved, "cli", Some("mermaid"));
724            render::render_mermaid_plugins(
725                &resolved.result.view.root,
726                &resolved.result.data,
727                &renders,
728            )
729            .map_err(|e| Fail::err(format!("{target}: {e}")))
730        }
731        other => {
732            // Plugin format: route every claimed node through its
733            // driver's `render` op and concatenate the models.
734            let renders = host.render_view_plugins(resolved, "cli", Some(other));
735            if renders.is_empty() {
736                return Err(Fail::err(format!(
737                    "{target}: no renderer plugin handles --format '{other}' \
738                     (built-ins: {})",
739                    BUILTIN_FORMATS.join(", ")
740                )));
741            }
742            // Deterministic order: sort by path ("" root first, then
743            // tree order).
744            let mut keyed: Vec<(&String, &pavo_protocol::RenderModel)> = renders.iter().collect();
745            keyed.sort_by(|a, b| a.0.cmp(b.0));
746            let mut out = String::new();
747            for (_, model) in keyed {
748                out.push_str(&render::render_model_cli(model));
749            }
750            Ok(out)
751        }
752    }
753}

790/// [`cmd_submit`] over an explicit host — split for tests.
791fn cmd_submit_host(
792    host: &mut Host,
793    target: &str,
794    form: Option<&str>,
795    values: Params,
796    json: bool,
797) -> Result<(), Fail> {
798    let result = host.submit(target, form, values)?;
799    if json {
800        println!(
801            "{}",
802            serde_json::to_string_pretty(&result).map_err(|e| Fail::err(e.to_string()))?
803        );
804    } else {
805        // Text mode: message first, then data.
806        if let Some(message) = &result.message {
807            println!("{message}");
808        }
809        match &result.data {
810            Some(data) => println!(
811                "{}",
812                serde_json::to_string_pretty(data).map_err(|e| Fail::err(e.to_string()))?
813            ),
814            None if result.message.is_none() && result.errors.is_empty() => {
815                println!("(empty result)");
816            }
817            None => {}
818        }
819    }
820    // Per-field validation errors mean the driver *rejected* the
821    // submit (the wire still answered ok:true). Exit 1 in both
822    // modes so CI can gate on it; `--json` callers also get the
823    // field details in the payload on stdout.
824    if !result.errors.is_empty() {
825        let detail = result
826            .errors
827            .iter()
828            .map(|(k, v)| format!("{k}: {}", v.as_str().unwrap_or("invalid")))
829            .collect::<Vec<_>>()
830            .join("; ");
831        return Err(Fail {
832            code: 1,
833            msg: format!("submit rejected — {detail}"),
834        });
835    }
836    Ok(())
837}

903/// Parse repeated `--param k=v` args into a [`Params`] map. Values
904/// parse as JSON when possible (`k=42`, `k=true`, `k={…}`), else
905/// become strings. Missing `=` or an empty key is a usage error.
906fn parse_params(args: &[String]) -> Result<Params, Fail> {
907    let mut params = Params::new();
908    for arg in args {
909        let Some((key, raw)) = arg.split_once('=') else {
910            return Err(Fail::usage(format!(
911                "--param expects key=value, got '{arg}'"
912            )));
913        };
914        if key.is_empty() {
915            return Err(Fail::usage(format!("--param has an empty key: '{arg}'")));
916        }
917        let value = serde_json::from_str(raw).unwrap_or_else(|_| Value::String(raw.to_string()));
918        params.insert(key.to_string(), value);
919    }
920    Ok(params)
921}
