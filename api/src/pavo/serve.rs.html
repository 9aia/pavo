1//! `pavo serve` — the localhost HTTP bridge: pavo-core's [`Host`] on
2//! one side, the Nuxt web app (or any JSON client) on the other.
3//!
4//! Hand-rolled HTTP/1.1 over [`std::net::TcpListener`], thread-per-
5//! connection, std-only (see [`crate::http`] for the wire bits). One
6//! [`Host`] is shared across threads behind an `Arc` — it is `Sync`:
7//! lazy manifest fetches, client spawns, and the view cache are
8//! internally locked, and each driver's `serve` client carries its own
9//! mutex so requests to *different* drivers run in parallel while
10//! same-driver requests stay serialized on the wire.
11//!
12//! Routes:
13//!
14//! - `GET /api/health` → `{ok:true}`
15//! - `GET /api/drivers` → per-driver manifest summary
16//! - `GET /api/views` → every `<driver>:<view>` entry
17//! - `GET /api/chrome` → merged chrome slots (navbar/dashboard/
18//!   statusbar/routes/sidebar/headerActions/notifications/jobs +
19//!   forward-compat extras)
20//! - `GET /api/jobs` → the aggregated job feed (`contributes.jobs`
21//!   providers' `jobs` op, items tagged with their driver)
22//! - `GET /api/notifications` → the aggregated notification feed
23//! - `POST /api/call?driver=<kind>` → cross-driver wire proxy: the
24//!   JSON body is a wire [`Request`] forwarded verbatim to that
25//!   driver's `serve`; its `result` comes back in the envelope
26//! - `GET /api/view?ref=<driver>:<view>` (+ `param.<k>=<v>`) → ViewResult
27//! - `GET /api/raw?ref=…` → same as `/api/view`
28//! - `GET /api/query?driver=<d>&query=<q>` (+ `param.<k>=<v>`) → QueryResult
29//! - `GET /api/submit?ref=<d>:<v>[&form=<id>][&values=<json>]` → SubmitResult
30//! - `GET /api/renderers` → the renderer-plugin registry (kind claims)
31//! - `GET /api/render?kind=<k>&node=<json>&data=<json>&surface=<s>[&format=<f>]`
32//!   → RenderModel — routes to the claiming driver's `render` op
33//! - `GET /*` → static files from the Nuxt output (`web/.output/public`,
34//!   `web/dist`, or `$PAVO_WEB_DIR`), else a built-in landing page.
35//!   `/view/<ref>` deep links take params as path `k=v` segments
36//!   (`/view/demo:board/id=T-1`) or `param.*`/bare query keys — the
37//!   Nuxt page speaks the same grammar (see `web/app/utils/view-link.ts`).
38//!
39//! All `/api/*` responses use the `{"ok":…,"result"/"error"}` envelope
40//! with `Access-Control-Allow-Origin: *` — this is a localhost dev
41//! tool, not a hardened server. GET/HEAD everywhere except
42//! `POST /api/call` (the wire proxy needs a body).

55/// Run the bridge until killed. Binds `127.0.0.1:port`; port `0`
56/// picks a free port. The first stdout line is always
57/// `listening http://127.0.0.1:<port>` so wrappers/tests can learn it.
58///
59/// # Errors
60/// [`io::Error`] if the listener can't bind.
61pub fn run(port: u16) -> io::Result<()> {
62    let listener = TcpListener::bind(("127.0.0.1", port))?;
63    let bound = listener.local_addr()?.port();
64    println!("listening http://127.0.0.1:{bound}");
65    use std::io::Write;
66    let _ = std::io::stdout().flush();
67
68    let host = Arc::new(Host::discover());
69    let web_root = find_web_root();
70    serve(listener, host, web_root)
71}

73/// The accept loop, split from [`run`] so tests can bind their own
74/// listener (and learn the port) without parsing stdout.
75fn serve(listener: TcpListener, host: Arc<Host>, web_root: Option<PathBuf>) -> io::Result<()> {
76    for stream in listener.incoming() {
77        match stream {
78            Ok(mut stream) => {
79                let host = Arc::clone(&host);
80                let root = web_root.clone();
81                std::thread::spawn(move || {
82                    let _ = handle_connection(&mut stream, &host, root.as_deref());
83                });
84            }
85            Err(e) => {
86                eprintln!("pavo serve: accept error: {e}");
87            }
88        }
89    }
90    Ok(())
91}

93/// Serve one connection (keep-alive loop). Errors are logged, never
94/// fatal — one bad client can't take the bridge down.
95fn handle_connection(
96    stream: &mut std::net::TcpStream,
97    host: &Arc<Host>,
98    web_root: Option<&Path>,
99) -> io::Result<()> {
100    loop {
101        match http::read_request(stream) {
102            Ok(None) => return Ok(()),
103            Ok(Some(req)) => {
104                let (response, close) = route(&req, host, web_root);
105                let close = close || req.wants_close();
106                let _ = response.write(stream, close);
107                if close {
108                    return Ok(());
109                }
110            }
111            Err(ReadError::Http { status, message }) => {
112                let resp = Response::raw(
113                    status,
114                    "text/plain; charset=utf-8",
115                    message.into_bytes(),
116                    false,
117                );
118                let _ = resp.write(stream, true);
119                return Ok(());
120            }
121            Err(ReadError::Io(e)) => return Err(e),
122        }
123    }
124}

126/// Route one request → (response, close_after). `close_after` is set
127/// for HEAD and for errors where keep-alive state is uncertain.
128fn route(req: &Request, host: &Arc<Host>, web_root: Option<&Path>) -> (Response, bool) {
129    let (path, query) = http::split_target(&req.target);
130    let path = http::percent_decode(path);
131    let pairs = http::parse_query(query);
132
133    // HEAD behaves like GET with an empty body.
134    let head_only = req.method.eq_ignore_ascii_case("HEAD");
135
136    if !path.starts_with("/api/") {
137        if req.method != "GET" && !head_only {
138            return (api_error(405, "method not allowed", false), true);
139        }
140        let resp = static_response(&path, web_root, &pairs);
141        let resp = if head_only {
142            Response {
143                body: Vec::new(),
144                ..resp
145            }
146        } else {
147            resp
148        };
149        return (resp, false);
150    }
151
152    // /api/* — JSON envelope, CORS on. POST is accepted only for
153    // /api/call (the cross-driver wire proxy needs a body);
154    // everything else stays GET/HEAD.
155    let post_call = req.method == "POST" && path == "/api/call";
156    if req.method != "GET" && !head_only && !post_call {
157        let mut resp = api_error(405, "method not allowed", true);
158        let allow = if path == "/api/call" {
159            "GET, HEAD, POST"
160        } else {
161            "GET, HEAD"
162        };
163        resp.extra.push(("allow".to_string(), allow.to_string()));
164        return (resp, false);
165    }
166
167    let resp = api_route(path.as_str(), &pairs, host, &req.body);
168    let resp = if head_only {
169        Response {
170            body: Vec::new(),
171            ..resp
172        }
173    } else {
174        resp
175    };
176    (resp, false)
177}

181fn api_route(path: &str, pairs: &[(String, String)], host: &Host, body: &[u8]) -> Response {
182    let endpoint = path.trim_start_matches("/api/");
183
184    let result: Result<Value, ApiFail> = match endpoint {
185        "health" => Ok(json!({})),
186        "drivers" => api_drivers(host),
187        "views" => api_views(host),
188        "chrome" => api_chrome(host),
189        "jobs" => api_jobs(host),
190        "notifications" => api_notifications(host),
191        "call" => api_call(host, pairs, body),
192        "view" | "raw" => api_view(host, pairs),
193        "query" => api_query(host, pairs),
194        "submit" => api_submit(host, pairs),
195        "renderers" => api_renderers(host),
196        "render" => api_render(host, pairs),
197        _ => Err(ApiFail::new(404, format!("unknown endpoint '{endpoint}'"))),
198    };
199
200    match result {
201        Ok(result) => Response::json(200, &json!({"ok": true, "result": result}), true),
202        Err(f) => Response::json(f.status, &json!({"ok": false, "error": f.message}), true),
203    }
204}

243fn api_drivers(host: &Host) -> Result<Value, ApiFail> {
244    let kinds: Vec<String> = host.drivers().iter().map(|d| d.kind.clone()).collect();
245    let mut rows = Vec::new();
246    for kind in kinds {
247        match host.manifest(&kind) {
248            Ok(m) => rows.push(json!({
249                "driver": kind,
250                "title": m.title,
251                "version": m.version,
252                "view_count": m.views.len(),
253                "contributes": crate::contrib_counts(&m.contributes),
254            })),
255            Err(e) => rows.push(json!({
256                "driver": kind,
257                "manifest_error": e.to_string(),
258            })),
259        }
260    }
261    Ok(json!(rows))
262}

284fn api_chrome(host: &Host) -> Result<Value, ApiFail> {
285    let chrome = host.chrome();
286    let slot = |entries: &[pavo_core::SlotEntry]| -> Vec<Value> {
287        entries
288            .iter()
289            .map(|e| {
290                json!({
291                    "driver": e.driver,
292                    "driver_title": e.driver_title,
293                    "id": e.contribution.id,
294                    "title": e.contribution.title,
295                    "route": e.contribution.route,
296                    "priority": e.contribution.priority,
297                    "icon": e.contribution.icon,
298                })
299            })
300            .collect()
301    };
302    let extra: serde_json::Map<String, Value> = chrome
303        .extra
304        .iter()
305        .map(|(k, v)| (k.clone(), Value::Array(slot(v))))
306        .collect();
307    Ok(json!({
308        "navbar": slot(&chrome.navbar),
309        "dashboard": slot(&chrome.dashboard),
310        "statusbar": slot(&chrome.statusbar),
311        "routes": slot(&chrome.routes),
312        "sidebar": slot(&chrome.sidebar),
313        // Wire spelling — matches `contributes.headerActions`.
314        "headerActions": slot(&chrome.header_actions),
315        "notifications": slot(&chrome.notifications),
316        "jobs": slot(&chrome.jobs),
317        "extra": Value::Object(extra),
318    }))
319}

339/// `POST /api/call?driver=<kind>` — the cross-driver escape hatch.
340/// The request body is a wire [`WireRequest`] JSON object
341/// (`{"op":"query",…}`, `{"op":"jobs"}`, …) forwarded verbatim to
342/// that driver's `serve` session; the driver's `result` payload
343/// comes back under the usual `{"ok":true,"result":…}` envelope.
344/// The driver name is validated by [`Host::client`] — unknown kinds
345/// 404, dep-quarantined drivers surface their [`PavoError::Deps`].
346fn api_call(host: &Host, pairs: &[(String, String)], body: &[u8]) -> Result<Value, ApiFail> {
347    let driver = http::query_get(pairs, "driver")
348        .ok_or_else(|| ApiFail::new(400, "missing 'driver' param"))?;
349    let request: WireRequest = serde_json::from_slice(body)
350        .map_err(|e| ApiFail::new(400, format!("body isn't a wire Request JSON object: {e}")))?;
351    let client = host.client(driver)?;
352    let mut guard = client
353        .lock()
354        .map_err(|_| ApiFail::new(500, "driver client lock poisoned"))?;
355    guard.request::<Value>(&request).map_err(ApiFail::from)
356}

358/// Resolve the `driver:view` target: `?ref=` wins, else `?driver=` +
359/// `?view=` (bare driver = its first view, same as the CLI).
360fn view_target(pairs: &[(String, String)]) -> Result<String, ApiFail> {
361    if let Some(r) = http::query_get(pairs, "ref") {
362        return Ok(r.to_string());
363    }
364    match (
365        http::query_get(pairs, "driver"),
366        http::query_get(pairs, "view"),
367    ) {
368        (Some(d), Some(v)) => Ok(format!("{d}:{v}")),
369        (Some(d), None) => Ok(d.to_string()),
370        _ => Err(ApiFail::new(
371            400,
372            "missing view ref — pass ?ref=<driver>:<view> or ?driver=&view=",
373        )),
374    }
375}

385    let driver = http::query_get(pairs, "driver")

394/// `GET /api/submit` — the `submit` wire op over HTTP. `ref` (or
395/// `driver`+`view`) names the target like `/api/view`; `form` is the
396/// optional form-node id; field values come from a JSON object in
397/// `?values={…}` or, absent that, the `param.<k>=<v>` convention
398/// `/api/view` uses.
399fn api_submit(host: &Host, pairs: &[(String, String)]) -> Result<Value, ApiFail> {
400    let target = view_target(pairs)?;
401    let form = http::query_get(pairs, "form");
402    let values: Params = match http::query_get(pairs, "values") {
403        Some(raw) => serde_json::from_str(raw)
404            .map_err(|e| ApiFail::new(400, format!("'values' isn't a JSON object: {e}")))?,
405        None => http::collect_params(pairs).into_iter().collect(),
406    };
407    let result = host.submit(&target, form, values)?;
408    serde_json::to_value(&result).map_err(|e| ApiFail::new(500, e.to_string()))
409}

433/// `GET /api/render` — route one node to its claiming renderer
434/// driver. `node`/`data` are URL-encoded JSON (`data` defaults to
435/// `{}`); `surface` is required (`cli`/`tui`/`web`); `format` is the
436/// optional format hint.
437fn api_render(host: &Host, pairs: &[(String, String)]) -> Result<Value, ApiFail> {
438    let kind =
439        http::query_get(pairs, "kind").ok_or_else(|| ApiFail::new(400, "missing 'kind' param"))?;
440    let surface = http::query_get(pairs, "surface")
441        .ok_or_else(|| ApiFail::new(400, "missing 'surface' param"))?;
442    let node_raw = http::query_get(pairs, "node").unwrap_or("{}");
443    let node: Value = serde_json::from_str(node_raw)
444        .map_err(|e| ApiFail::new(400, format!("'node' isn't valid JSON: {e}")))?;
445    let data: Params = match http::query_get(pairs, "data") {
446        Some(raw) => serde_json::from_str(raw)
447            .map_err(|e| ApiFail::new(400, format!("'data' isn't a JSON object: {e}")))?,
448        None => Params::new(),
449    };
450    let format = http::query_get(pairs, "format");
451    let model = host.render_by_kind(kind, &node, &data, surface, format)?;
452    serde_json::to_value(&model).map_err(|e| ApiFail::new(500, e.to_string()))
453}

457/// Locate the web root: `$PAVO_WEB_DIR`, then `web/.output/public` /
458/// `web/dist` relative to the cwd, then the same under the repo root
459/// (derived from the crate manifest dir — works from any cwd inside
460/// a pavo checkout).
461fn find_web_root() -> Option<PathBuf> {
462    // Explicit override wins outright — set-but-missing means "no web
463    // root" (tests use this to force the built-in landing page).
464    if let Some(dir) = std::env::var_os("PAVO_WEB_DIR") {
465        let p = PathBuf::from(dir);
466        return p.is_dir().then_some(p);
467    }
468    let mut candidates: Vec<PathBuf> = Vec::new();
469    for base in [
470        PathBuf::from("."),
471        Path::new(env!("CARGO_MANIFEST_DIR")).join("../.."),
472    ] {
473        candidates.push(base.join("web/.output/public"));
474        candidates.push(base.join("web/dist"));
475    }
476    candidates.into_iter().find(|p| p.is_dir())
477}

479fn mime_for(path: &Path) -> &'static str {
480    match path.extension().and_then(|e| e.to_str()) {
481        Some("html") | Some("htm") => "text/html; charset=utf-8",
482        Some("css") => "text/css; charset=utf-8",
483        Some("js") | Some("mjs") => "text/javascript; charset=utf-8",
484        Some("json") | Some("map") => "application/json; charset=utf-8",
485        Some("svg") => "image/svg+xml",
486        Some("png") => "image/png",
487        Some("jpg") | Some("jpeg") => "image/jpeg",
488        Some("gif") => "image/gif",
489        Some("webp") => "image/webp",
490        Some("ico") => "image/x-icon",
491        Some("woff") => "font/woff",
492        Some("woff2") => "font/woff2",
493        Some("ttf") => "font/ttf",
494        Some("wasm") => "application/wasm",
495        Some("txt") | Some("md") => "text/plain; charset=utf-8",
496        _ => "application/octet-stream",
497    }
498}

500/// Resolve `path` under `root` safely — rejects `..` escapes and
501/// non-UTF8 tricks. Returns the file path or None.
502fn resolve_static(root: &Path, path: &str) -> Option<PathBuf> {
503    let rel = path.trim_start_matches('/');
504    if rel.split('/').any(|seg| seg == "..") {
505        return None;
506    }
507    let candidate = root.join(rel);
508    // Reject symlink escapes too — canonicalize both sides.
509    let canon = candidate.canonicalize().ok()?;
510    let root_canon = root.canonicalize().ok()?;
511    if !canon.starts_with(&root_canon) {
512        return None;
513    }
514    if canon.is_file() {
515        return Some(canon);
516    }
517    if canon.is_dir() {
518        let index = canon.join("index.html");
519        if index.is_file() {
520            return Some(index);
521        }
522    }
523    None
524}

526/// `GET /*` handler: static file if a web root exists, else the
527/// built-in landing page (and a minimal `/view/<ref>` page).
528fn static_response(path: &str, web_root: Option<&Path>, pairs: &[(String, String)]) -> Response {
529    if let Some(root) = web_root {
530        if let Some(file) = resolve_static(root, path) {
531            let body = std::fs::read(&file).unwrap_or_default();
532            return Response::raw(200, mime_for(&file), body, false);
533        }
534        // SPA-ish fallback: page-like paths (no extension) get the
535        // Nuxt SPA shell — `200.html` when the static export emitted
536        // one (its payload isn't pinned to a prerendered route, unlike
537        // `index.html`), else `index.html`.
538        let last = path.rsplit('/').next().unwrap_or("");
539        if !last.contains('.') {
540            for fallback in ["/200.html", "/index.html"] {
541                if let Some(file) = resolve_static(root, fallback) {
542                    let body = std::fs::read(&file).unwrap_or_default();
543                    return Response::raw(200, mime_for(&file), body, false);
544                }
545            }
546        }
547        return Response::raw(
548            404,
549            "text/plain; charset=utf-8",
550            format!("not found: {path}").into_bytes(),
551            false,
552        );
553    }
554
555    // No web build — the built-in mini UI keeps the bridge useful.
556    match path {
557        "/" => landing_page(),
558        p if p.starts_with("/view/") => {
559            let rest = p.trim_start_matches("/view/");
560            view_page_stub(rest, pairs)
561        }
562        _ => Response::raw(
563            404,
564            "text/plain; charset=utf-8",
565            "not found — no web build (run `npm run build` in web/)"
566                .as_bytes()
567                .to_vec(),
568            false,
569        ),
570    }
571}

573fn html_escape(s: &str) -> String {
574    s.replace('&', "&amp;")
575        .replace('<', "&lt;")
576        .replace('>', "&gt;")
577        .replace('"', "&quot;")

670/// Split a `/view/` path remainder into (ref, params) — the first
671/// `/`-separated segment is the `driver:view` ref, each later segment
672/// is `key=value` (a bare `key` means an empty value). Query keys are
673/// params too: bare `?k=v` first, then `param.*` on top, so query
674/// wins over path and `param.*` wins over bare on collision — the
675/// same grammar the Nuxt page normalizes into `param.<k>=<v>`.
676fn parse_view_path(rest: &str, pairs: &[(String, String)]) -> (String, Params) {
677    let mut segments = rest.split('/');
678    let reference = segments.next().unwrap_or("").to_string();
679    let mut params = Params::new();
680    for seg in segments {
681        if seg.is_empty() {
682            continue;
683        }
684        let (k, v) = seg.split_once('=').unwrap_or((seg, ""));
685        if !k.is_empty() {
686            params.insert(k.to_string(), http::param_json(v));
687        }
688    }
689    // Bare query keys (`?k=v`) — everything not in the `param.`
690    // namespace.
691    for (k, v) in pairs {
692        if !k.is_empty() && !k.starts_with("param.") {
693            params.insert(k.clone(), http::param_json(v));
694        }
695    }
696    // `param.*` last → highest precedence.
697    for (k, v) in http::collect_params(pairs) {
698        params.insert(k, v);
699    }
700    (reference, params)
701}

703/// Minimal pre-build view page: fetches nothing — renders the
704/// ViewResult JSON inline so the bridge is debuggable without Nuxt.
705///
706/// `rest` is the percent-decoded path after `/view/`; see
707/// [`parse_view_path`] for the deep-link grammar (same as the Nuxt
708/// page: path `k=v` segments + `param.*`/bare query keys).
709fn view_page_stub(rest: &str, pairs: &[(String, String)]) -> Response {
710    // Discover a fresh host (same reasoning as landing_page).
711    let host = Host::discover();
712    let (reference, params) = parse_view_path(rest, pairs);
713    let body_inner = match host.resolve_view(&reference, params) {
714        Ok(resolved) => format!(
715            "<p><a href=\"/\">← all views</a></p><h1>{}</h1><pre>{}</pre>",
716            html_escape(&resolved.namespaced),
717            html_escape(&serde_json::to_string_pretty(&resolved.result).unwrap_or_default()),
718        ),
719        Err(e) => format!(
720            "<p><a href=\"/\">← all views</a></p><h1>{}</h1><p><b>error:</b> {}</p>",
721            html_escape(&reference),
722            html_escape(&e.to_string()),
723        ),
724    };
725    html_page(&reference, &body_inner)
726}

985        let tmp = tempfile::tempdir().unwrap();
986        let bin = tmp.path().join("bin");

996        // Keep the tempdir alive for the process lifetime.

1000        let opts = pavo_core::discovery::DiscoveryOptions {
1001            path: Some(std::ffi::OsString::from(&bin)),
1002            tamias_home: Some(home),
1003            ..Default::default()
1004        };
1005        Arc::new(Host::discover_with(opts))

1283    /// until `release` exists; every other query answers `"fast"`.

1285        let manifest = bare_manifest("gate");
1286        let wire = format!("{{\"ok\":true,\"result\":{manifest}}}");

1507    /// whole reply (server closes or we hit EOF).
1508    fn roundtrip(port: u16, bytes: &[u8]) -> String {

1511        c.write_all(bytes).unwrap();

1518    /// Read one HTTP response (head + content-length body) from an
1519    /// open keep-alive connection.
1520    fn read_http_response(c: &mut TcpStream) -> String {
1521        let mut out = Vec::new();
1522        let mut buf = [0u8; 4096];
1523        let head_end;
1524        loop {
1525            let n = c.read(&mut buf).unwrap();
1526            assert!(n > 0, "connection closed mid-response");
1527            out.extend_from_slice(&buf[..n]);
1528            if let Some(p) = find_bytes(&out, b"\r\n\r\n") {
1529                head_end = p + 4;
1530                break;
1531            }
1532        }
1533        let head = String::from_utf8_lossy(&out[..head_end]).into_owned();
1534        let len: usize = head
1535            .lines()
1536            .find_map(|l| {
1537                l.to_ascii_lowercase()
1538                    .strip_prefix("content-length:")
1539                    .and_then(|v| v.trim().parse().ok())
1540            })
1541            .unwrap_or(0);
1542        while out.len() < head_end + len {
1543            let n = c.read(&mut buf).unwrap();
1544            assert!(n > 0, "connection closed mid-body");
1545            out.extend_from_slice(&buf[..n]);
1546        }
1547        String::from_utf8_lossy(&out[..head_end + len]).into_owned()
1548    }
